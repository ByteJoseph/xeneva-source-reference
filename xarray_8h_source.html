27/*
28 * The bottom two bits of the entry determine how the XArray interprets
29 * the contents:
30 *
31 * 00: Pointer entry
32 * 10: Internal entry
33 * x1: Value entry or tagged pointer
34 *
35 * Attempting to store internal entries in the XArray is a bug.
36 *
37 * Most internal entries are pointers to the next node in the tree.
38 * The following internal entries have a special meaning:
39 *
40 * 0-62: Sibling entries
41 * 256: Retry entry
42 * 257: Zero entry
43 *
44 * Errors are also represented as internal entries, but use the negative
45 * space (-4094 to -2). They're never stored in the slots array; only
46 * returned by the normal API.
47 */
48
49#define BITS_PER_XA_VALUE (BITS_PER_LONG - 1)

101/*
102 * DCL delta -- `uintptr_t` where mainline writes `unsigned long`.
103 *
104 * A tagged entry is a real address with its tag in the bottom two bits, so
105 * the integer in the middle of the round trip has to be pointer width.
106 * mainline gets that from `unsigned long` being 64 bits on arm64; this
107 * target is aarch64-unknown-windows, which is LLP64 (verified:
108 * sizeof(long) == 4), so mainline's spelling keeps the address's low 32
109 * bits and hands back a pointer into the low 4 GB that nothing maps.
110 * xa_tag_pointer()/xa_untag_pointer() here, and xa_mk_node()/xa_to_node()
111 * further down, are the two round-trip pairs; xa_is_node() compares a whole
112 * entry, so it widens too. `~(uintptr_t)3` rather than mainline's `~3UL`
113 * because the 32-bit mask would clear out the address half again.
114 *
115 * The casts that only read bits stay `unsigned long`: the tag in
116 * xa_pointer_tag(), xa_is_internal() and xa_is_value() lives in the low
117 * bits at either width, and the value entries (xa_mk_value()) are not
118 * addresses.
119 */
120static inline void *xa_tag_pointer(void *p, unsigned long tag)
121{
122 return (void *)((uintptr_t)p | tag);
123}
124
135static inline void *xa_untag_pointer(void *entry)
136{
137 return (void *)((uintptr_t)entry & ~(uintptr_t)3);
138}
139
150static inline unsigned int xa_pointer_tag(void *entry)
151{
152 return (unsigned long)entry & 3UL;
153}

155/*
156 * xa_mk_internal() - Create an internal entry.
157 * @v: Value to turn into an internal entry.
158 *
159 * Internal entries are used for a number of purposes. Entries 0-255 are
160 * used for sibling entries (only 0-62 are used by the current code). 256
161 * is used for the retry entry. 257 is used for the reserved / zero entry.
162 * Negative internal entries are used to represent errnos. Node pointers
163 * are also tagged as internal entries in some situations.
164 *
165 * Context: Any context.
166 * Return: An XArray internal entry corresponding to this value.
167 */
168static inline void *xa_mk_internal(unsigned long v)
169{
170 return (void *)((v << 2) | 2);
171}

173/*
174 * xa_to_internal() - Extract the value from an internal entry.
175 * @entry: XArray entry.
176 *
177 * Context: Any context.
178 * Return: The value which was stored in the internal entry.
179 */
180static inline unsigned long xa_to_internal(const void *entry)
181{
182 return (unsigned long)entry >> 2;
183}

185/*
186 * xa_is_internal() - Is the entry an internal entry?
187 * @entry: XArray entry.
188 *
189 * Context: Any context.
190 * Return: %true if the entry is an internal entry.
191 */
192static inline bool xa_is_internal(const void *entry)
193{
194 return ((unsigned long)entry & 3) == 2;
195}

1162/*
1163 * The xarray is constructed out of a set of 'chunks' of pointers. Choosing
1164 * the best chunk size requires some tradeoffs. A power of two recommends
1165 * itself so that we can walk the tree based purely on shifts and masks.
1166 * Generally, the larger the better; as the number of slots per level of the
1167 * tree increases, the less tall the tree needs to be. But that needs to be
1168 * balanced against the memory consumption of each node. On a 64-bit system,
1169 * xa_node is currently 576 bytes, and we get 7 of them per 4kB page. If we
1170 * doubled the number of slots per node, we'd get only 3 nodes per 4kB page.
1171 */
1172#ifndef XA_CHUNK_SHIFT
1173#define XA_CHUNK_SHIFT (IS_ENABLED(CONFIG_BASE_SMALL) ? 4 : 6)
1174#endif
1175#define XA_CHUNK_SIZE (1UL << XA_CHUNK_SHIFT)
1176#define XA_CHUNK_MASK (XA_CHUNK_SIZE - 1)
1177#define XA_MAX_MARKS 3
1178#define XA_MARK_LONGS BITS_TO_LONGS(XA_CHUNK_SIZE)
1179
1180/*
1181 * @count is the count of every non-NULL element in the ->slots array
1182 * whether that is a value entry, a retry entry, a user pointer,
1183 * a sibling entry or a pointer to the next level of the tree.
1184 * @nr_values is the count of every element in ->slots which is
1185 * either a value entry or a sibling of a value entry.
1186 */
1187struct xa_node {
1188 unsigned char shift; /* Bits remaining in each slot */
1189 unsigned char offset; /* Slot offset in parent */
1190 unsigned char count; /* Total entry count */
1191 unsigned char nr_values; /* Value entry count */
1192 struct xa_node __rcu *parent; /* NULL at top of tree */
1193 struct xarray *array; /* The array we belong to */
1194 union {
1195 struct list_head private_list; /* For tree user */
1196 struct rcu_head rcu_head; /* Used when freeing node */
1197 };
1198 void __rcu *slots[XA_CHUNK_SIZE];
1199 union {
1200 unsigned long tags[XA_MAX_MARKS][XA_MARK_LONGS];
1201 unsigned long marks[XA_MAX_MARKS][XA_MARK_LONGS];
1202 };
1203};

1356/*
1357 * The xa_state is opaque to its users. It contains various different pieces
1358 * of state involved in the current operation on the XArray. It should be
1359 * declared on the stack and passed between the various internal routines.
1360 * The various elements in it should not be accessed directly, but only
1361 * through the provided accessor functions. The below documentation is for
1362 * the benefit of those working on the code, not for users of the XArray.
1363 *
1364 * @xa_node usually points to the xa_node containing the slot we're operating
1365 * on (and @xa_offset is the offset in the slots array). If there is a
1366 * single entry in the array at index 0, there are no allocated xa_nodes to
1367 * point to, and so we store %NULL in @xa_node. @xa_node is set to
1368 * the value %XAS_RESTART if the xa_state is not walked to the correct
1369 * position in the tree of nodes for this operation. If an error occurs
1370 * during an operation, it is set to an %XAS_ERROR value. If we run off the
1371 * end of the allocated nodes, it is set to %XAS_BOUNDS.
1372 */

1376 unsigned char xa_shift;
1377 unsigned char xa_sibs;
1378 unsigned char xa_offset;
1379 unsigned char xa_pad; /* Helps gcc generate better code */
1380 struct xa_node *xa_node;
1381 struct xa_node *xa_alloc;
1382 xa_update_node_t xa_update;
1383 struct list_lru *xa_lru;
1384};
