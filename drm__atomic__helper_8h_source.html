249static inline bool drm_atomic_plane_enabling(struct drm_plane_state *old_plane_state,
250 struct drm_plane_state *new_plane_state)
251{
252 /*
253 * When enabling a plane, CRTC and FB should always be set together.
254 * Anything else should be considered a bug in the atomic core, so we
255 * gently warn about it.
256 */
257 WARN_ON((!new_plane_state->crtc && new_plane_state->fb) ||
258 (new_plane_state->crtc && !new_plane_state->fb));
259
260 return !old_plane_state->crtc && new_plane_state->crtc;
261}

275static inline bool
276drm_atomic_plane_disabling(struct drm_plane_state *old_plane_state,
277 struct drm_plane_state *new_plane_state)
278{
279 /*
280 * When disabling a plane, CRTC and FB should always be NULL together.
281 * Anything else should be considered a bug in the atomic core, so we
282 * gently warn about it.
283 */
284 WARN_ON((new_plane_state->crtc == NULL && new_plane_state->fb != NULL) ||
285 (new_plane_state->crtc != NULL && new_plane_state->fb == NULL));
286
287 return old_plane_state->crtc && !new_plane_state->crtc;
288}
