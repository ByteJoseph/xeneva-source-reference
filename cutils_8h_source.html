366{
367 if (c >= '0' && c <= '9')

401static inline double fromfp16(uint16_t v)
402{
403 double d;
404 uint32_t v1;
405 v1 = v & 0x7fff;
406 if (unlikely(v1 >= 0x7c00))
407 v1 += 0x1f8000; /* NaN or infinity */
408 d = uint64_as_float64(((uint64_t)(v >> 15) << 63) | ((uint64_t)v1 << (52 - 10)));
409 return d * 0x1p1008;
410}
411
412static inline uint16_t tofp16(double d)
413{
414 uint64_t a, addend;
415 uint32_t v, sgn;
416 int shift;
417
418 a = float64_as_uint64(d);
419 sgn = a >> 63;
420 a = a & 0x7fffffffffffffff;
421 if (unlikely(a > 0x7ff0000000000000)) {
422 /* nan */
423 v = 0x7c01;
424 } else if (a < 0x3f10000000000000) { /* 0x1p-14 */
425 /* subnormal f16 number or zero */
426 if (a <= 0x3e60000000000000) { /* 0x1p-25 */
427 v = 0x0000; /* zero */
428 } else {
429 shift = 1051 - (a >> 52);
430 a = ((uint64_t)1 << 52) | (a & (((uint64_t)1 << 52) - 1));
431 addend = ((a >> shift) & 1) + (((uint64_t)1 << (shift - 1)) - 1);
432 v = (a + addend) >> shift;
433 }
434 } else {
435 /* normal number or infinity */
436 a -= 0x3f00000000000000; /* adjust the exponent */
437 /* round */
438 addend = ((a >> (52 - 10)) & 1) + (((uint64_t)1 << (52 - 11)) - 1);
439 v = (a + addend) >> (52 - 10);
440 /* overflow ? */
441 if (unlikely(v > 0x7c00))
442 v = 0x7c00;
443 }
444 return v | (sgn << 15);
445}
