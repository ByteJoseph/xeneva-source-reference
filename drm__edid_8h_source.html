385static inline const char *drm_edid_decode_mfg_id(u16 mfg_id, char vend[4])
386{
387 vend[0] = '@' + ((mfg_id >> 10) & 0x1f);
388 vend[1] = '@' + ((mfg_id >> 5) & 0x1f);
389 vend[2] = '@' + ((mfg_id >> 0) & 0x1f);
390 vend[3] = '\0';
391
392 return vend;
393}
