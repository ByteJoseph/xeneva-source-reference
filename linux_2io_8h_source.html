44/*
45 * ioread32be()/iowrite32be() -- 32-bit big-endian MMIO.
46 *
47 * Only mem32be_serial_in/out() use them (8250_port.c:386-396), the register
48 * access mode for SoCs whose UART register block sits on a big-endian bus.
49 * On this little-endian aarch64 target a big-endian device word is a byte
50 * swap of the natural load, which is exactly what mainline's
51 * be32_to_cpu/cpu_to_be32 pair reduces to -- so the swap is spelled out with
52 * __builtin_bswap32 rather than pulling in <linux/byteorder> for two calls.
53 *
54 * Worth knowing: the *absence* of the swap would still compile and would
55 * still produce a UART that never answers, because every 8250 register read
56 * would come back byte-reversed. That failure mode -- transmits, never
57 * receives -- is the same one the comment at the top of <linux/termios.h>
58 * warns about for flag tables.
59 */
60static inline u32 ioread32be(const volatile void* addr)
61{
62 return (u32)__builtin_bswap32(readl(addr));
63}
64
65static inline void iowrite32be(u32 v, volatile void* addr)
66{
67 writel(__builtin_bswap32(v), addr);
68}
