20/*
21 * struct _ddebug_class_map and DECLARE_DYNDBG_CLASSMAP -- the one piece of
22 * the dyndbg surface that has to declare a real object instead of discarding
23 * its arguments.
24 *
25 * drm_crtc_helper.c:53 spells the macro *unguarded*, and only drm_print.c's
26 * copy sits inside `#if defined(CONFIG_DRM_USE_DYNAMIC_DEBUG)` (drm_print.c:
27 * 58-72). So the expansion has to put a name to something, or that line
28 * decays to a list of bare identifiers and clang reports "type specifier
29 * missing" at the first one.
30 *
31 * `static` is not an oversight: when CONFIG_DRM_USE_DYNAMIC_DEBUG *is*
32 * defined, drm_print.c:60 expands this same macro, and two extern definitions
33 * of one symbol would not link. It also keeps drm_crtc_helper.c's copy from
34 * being reported as unused in a build where the map is written but never read.
35 *
36 * The variadic tail -- the ten "DRM_UT_*" strings -- is dropped. They exist
37 * in mainline so the dyndbg control file can name a class as text, and there
38 * is no control file here; the class *values* are what the drm core
39 * compares against, and they come from enum drm_debug_category, not from
40 * these strings.
41 */
42struct _ddebug_class_map {
