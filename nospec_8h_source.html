4/*
5 * DCL <linux/nospec.h> -- array_index_nospec(), the bounds-check clamp.
6 *
7 * Only the part a ported driver actually calls is here. mainline's header
8 * also carries barrier_nospec() and the arch_prctl / arch_seccomp speculation
9 * prctls; nothing in DCL uses those, and an uncompiled definition is worse
10 * than an absent one (it reads as supported).
11 *
12 * ### The one real difference from mainline: the second BUILD_BUG_ON is gone
13 *
14 * mainline's macro ends with:
15 *
16 * BUILD_BUG_ON(sizeof(_i) > sizeof(long));
17 * BUILD_BUG_ON(sizeof(_s) > sizeof(long));
18 *
19 * The first guard is kept. The second cannot survive on this target:
20 * aarch64-unknown-windows is LLP64, so sizeof(long) is 4 while sizeof(size_t)
21 * is 8 -- and `_s` is routinely a size_t, `sizeof(vi->data)` in virtio-rng.c
22 * being the current instance. mainline can only afford that check because on
23 * LP64 long and size_t are the same width; here it would make every use of
24 * the macro a compile error rather than catch anything.
25 *
26 * The truncation it guards against is real but harmless for a length: the
27 * value is a struct member's size, always far below 4 GB, and it is only ever
28 * compared against an index of the same object. So `_s` is cast to
29 * unsigned long explicitly rather than silently converted -- the narrowing is
30 * written down at the call site of the mask function, where a future reader
31 * will see it.
32 *
33 * If a _s value could ever exceed 4 GB, the right fix is to make the whole
34 * clamp size_t-wide and drop array_index_mask_nospec's unsigned long
35 * parameters, not to pretend the guard is still there.
36 *
37 * ### BITS_PER_LONG is 32 here, and that is load-bearing
38 *
39 * the mask is a sign-bit trick: it must shift the sign bit of a
40 * `unsigned long` into place, so the shift count has to be
41 * sizeof(unsigned long)*8 - 1. kernel.h defines BITS_PER_LONG exactly that
42 * way (it was a hardcoded 64 until bringing this file in, which would have
43 * shifted a 32-bit value by 63 and produced mask 0 -- silently reading
44 * element 0 of every array, forever).
45 */

49/*
50 * Tell the compiler the value of `var` is not available, so it cannot fold
51 * a load out of the clamp on the strength of an earlier bounds check. This
52 * is mainline's spelling: an empty asm with an output operand, which keeps
53 * the value in a register and discards the compiler's knowledge of it. It is
54 * a compiler barrier only -- no instruction is emitted.
55 */
56#ifndef OPTIMIZER_HIDE_VAR
57#define OPTIMIZER_HIDE_VAR(var) __asm__ __volatile__("" : "=r"(var))

72static inline unsigned long array_index_mask_nospec(unsigned long index,
73 unsigned long size)
74{
75 /*
76 * Always compute the mask even if the compiler can prove the index is
77 * in range: it cannot prove it is in range *under speculation*, which
78 * is the whole threat model. OPTIMIZER_HIDE_VAR stops it from folding
79 * the answer away anyway.
80 */
81 OPTIMIZER_HIDE_VAR(index);
82 return ~(unsigned long)(index | (size - 1UL - index)) >>
83 (BITS_PER_LONG - 1);
84}
