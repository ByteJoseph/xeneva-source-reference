4/*
5 * DCL <linux/cleanup.h> -- scope guards, in mainline's call syntax.
6 *
7 * Modern mainline tty/serial code does not say mutex_lock() ... mutex_unlock()
8 * a hundred lines apart. It says:
9 *
10 * guard(mutex)(&port->mutex) { ... }
11 * scoped_guard(spinlock_irq, &tty->ctrl.lock) { ... }
12 * scoped_cond_guard(mutex_intr, return -ERESTARTSYS, &port->mutex) { ... }
13 *
14 * There are 120 such sites across serial_core.c, 8250_port.c, n_tty.c and
15 * tty_io.c, which makes this header load-bearing rather than ornamental:
16 * without it, none of them parse.
17 *
18 * Mainline builds them on __cleanup()/DEFINE_FREE()/IS_ERR() and a
19 * lock_ptr/lock_err protocol spread across <linux/compiler.h>, <linux/err.h>
20 * and <linux/args.h>. DCL keeps the *call syntax* -- which is what driver text
21 * depends on -- and implements it directly on the one piece of machinery both
22 * sides genuinely need, clang's __attribute__((cleanup)).
23 *
24 * The macros are deliberately spelled DCL_LOCK_GUARD*, not mainline's
25 * DEFINE_GUARD_*: adopting mainline's names while changing their argument
26 * lists would be a trap for whoever next ports serial_core.h. When that
27 * happens the few guard definitions in it get rewritten to these, and the
28 * call sites -- the part that matters -- stay byte-identical to upstream.
29 *
30 * The three forms:
31 *
32 * guard(name)(args) { ... }
33 * Acquire on the way in, release when the enclosing scope ends -- on
34 * fall-through, on `break`, on `return`, on `goto`, whichever comes
35 * first. That last part is the entire point: a `return` in the middle
36 * of a critical section still unlocks, which is exactly what a manual
37 * lock()/unlock() pair gets wrong once a function grows an error path.
38 *
39 * scoped_guard(name, args) { ... }
40 * The same, with the lifetime bound to the next compound statement.
41 *
42 * scoped_cond_guard(name, fail, args) { ... }
43 * For locks that can fail to be acquired. If acquisition failed the
44 * guard holds lock == NULL, so the body is skipped and `fail` runs
45 * instead -- and because nothing was ever locked, the destructor has
46 * nothing to release. `fail` is normally `return -ERESTARTSYS`.
47 *
48 * Guard *classes* (mutex, spinlock_irq, rwsem_read, uart_port_lock_irq, ...)
49 * live beside the primitives they wrap -- <linux/spinlock.h>, <linux/mutex.h>,
50 * <linux/rwsem.h>, later serial_core.h -- because a guard is only meaningful
51 * next to the lock it takes.
52 *
53 * Not ported: CLASS()/DEFINE_CLASS()/DEFINE_FREE()/ACQUIRE(). Nothing in the
54 * nine tty/serial files uses them (only PCI_DEVICE_CLASS, an unrelated PCI
55 * macro, resembles them). Add them when something does, rather than carrying
56 * a speculative surface.
57 */
58

66/*
67 * DCL_LOCK_GUARD_TYPES(name, type, release) -- the half of a guard that both
68 * variants own: the struct carrying the held lock, NULL while it is not
69 * held, and the destructor that releases only a lock that is held.
70 *
71 * That NULL-means-not-held rule is the whole reason DCL_LOCK_GUARD_COND can
72 * report a failed acquire at all, so it is written once rather than twice --
73 * a destructor that disagreed with its own ctor would unlock a lock the
74 * caller never took, in code that runs on scope exit where nobody looks.
75 *
76 * Only the acquisition differs between the two macros, which is why each
77 * still spells its own: unconditional for DCL_LOCK_GUARD, an int expression
78 * that can fail for DCL_LOCK_GUARD_COND.
79 */
80#define DCL_LOCK_GUARD_TYPES(name, type, release) \
81 typedef struct { type* lock; } dclg_##name; \
82 static inline void dclg_##name##_release(dclg_##name* g) \
83 { \
84 type* _T = g->lock; \
85 if (_T) { release; } \
86 }

88/*
89 * DCL_LOCK_GUARD(name, type, acquire, release)
90 * For locks that cannot fail to be taken (spinlocks, the rwsem read side,
91 * task_lock): `acquire` runs unconditionally. `acquire` and `release` are
92 * statements written against a local `_T` of type `type *`.
93 *
94 * DCL_LOCK_GUARD_COND(name, type, acquire, release)
95 * Same shape, except `acquire` must be an *int expression* that is 0 on
96 * success (mutex_lock_interruptible's contract). A nonzero result leaves
97 * lock == NULL, which is how scoped_cond_guard() tells "failed" apart
98 * from "held" -- and how the destructor avoids unlocking a lock that was
99 * never taken.
100 */
101#define DCL_LOCK_GUARD(name, type, acquire, release) \
102 DCL_LOCK_GUARD_TYPES(name, type, release) \
103 static inline dclg_##name dclg_##name##_ctor(type* p) \
104 { \
105 dclg_##name g; \
106 type* _T = p; \
107 (void)_T; \
108 acquire; \
109 g.lock = p; \
110 return g; \
111 }
112
113#define DCL_LOCK_GUARD_COND(name, type, acquire, release) \
114 DCL_LOCK_GUARD_TYPES(name, type, release) \
115 static inline dclg_##name dclg_##name##_ctor(type* p) \
116 { \
117 dclg_##name g; \
118 type* _T = p; \
119 int _r = (acquire); \
120 (void)_T; \
121 g.lock = (_r == 0) ? p : 0; \
122 return g; \
123 }

125/*
126 * Mainline's spelling of the two forms that are actually instantiated.
127 *
128 * These exist because of Vendored/drivers/tty/serial/8250/8250.h, which says
129 * DEFINE_GUARD(serial8250_rpm, struct uart_8250_port *,
130 * serial8250_rpm_get(_T), serial8250_rpm_put(_T));
131 * and vendored files are never rewritten to suit DCL (Vendored/README.md
132 * rule 1) -- so when mainline text says DEFINE_GUARD, BaseHdr has to answer
133 * to that name. The DCL_* spellings stay for everything written by hand.
134 *
135 * The header above worried that adopting mainline's names "while changing
136 * their argument lists would be a trap". These do NOT change the list:
137 * DEFINE_GUARD(_name, _type, _lock, _unlock) has mainline's order, and the
138 * one real difference from DCL_LOCK_GUARD is that mainline passes a *pointer*
139 * type (`struct uart_port *`) and names the local `_T`, where DCL passes the
140 * pointee (`struct mutex`) and makes the local `type *_T`. Two shapes, each
141 * self-consistent, so neither spelling quietly means the other's.
142 *
143 * Not ported here, same rule as above: DEFINE_GUARD_COND and
144 * DEFINE_LOCK_GUARD_1_COND. serial_core.h's two _try definitions were
145 * removed with it -- nothing in 8250_port.c, serial_core.c or 8250.h takes a
146 * _try guard, and a conditional acquire with no caller is machinery nobody
147 * can test. The 3-argument forms are what to port if one shows up.
148 */

150/*
151 * DEFINE_GUARD(_name, _type, _lock, _unlock)
152 * _type is the POINTER type (mainline passes `struct uart_port *`), so
153 * the guard struct holds it directly and `_T` in the expressions is that
154 * pointer -- which is what makes `uart_port_unlock(_T)` read correctly.
155 */
156#define DEFINE_GUARD(_name, _type, _lock, _unlock) \
157 typedef struct { _type lock; } dclg_##_name; \
158 static inline void dclg_##_name##_release(dclg_##_name* g) \
159 { \
160 _type _T = g->lock; \
161 if (_T) { _unlock; } \
162 } \
163 static inline dclg_##_name dclg_##_name##_ctor(_type p) \
164 { \
165 dclg_##_name g; \
166 _type _T = p; \
167 (void)_T; \
168 _lock; \
169 g.lock = p; \
170 return g; \
171 }

173/*
174 * DEFINE_LOCK_GUARD_1(_name, _type, _lock, _unlock, ...)
175 * For guards whose lock routine carries state that outlives the call --
176 * uart_port_lock_irqsave() writes a saved-flags word that
177 * uart_port_unlock_irqrestore() reads back. That word has to live in the
178 * guard object, not on the ctor's stack, which is the entire reason this
179 * form takes a trailing declaration list (`unsigned long flags`).
180 *
181 * `_type` is the BARE type here (mainline passes `struct uart_port`), so
182 * the member is `_type *lock` and the expressions address into the guard:
183 * `_T->lock` and `_T->flags`. _T is therefore the guard, and it is
184 * written by the release function too -- both sides see the same struct
185 * because the ctor returns it by value and __attribute__((cleanup))
186 * passes the stored copy back.
187 */
188#define DEFINE_LOCK_GUARD_1(_name, _type, _lock, _unlock, ...) \
189 typedef struct { _type* lock; __VA_ARGS__; } dclg_##_name; \
190 static inline void dclg_##_name##_release(dclg_##_name* _T) \
191 { \
192 if (_T->lock) { _unlock; } \
193 } \
194 static inline dclg_##_name dclg_##_name##_ctor(_type* p) \
195 { \
196 dclg_##_name g = { .lock = p }; \
197 dclg_##_name* _T = &g; \
198 (void)_T; \
199 _lock; \
200 return g; \
201 }

203/*
204 * guard(name)(args);
205 * An anonymous guard variable. The type name and the __COUNTER__ suffix
206 * keep two guards in one function from colliding, which is why the
207 * argument has to be pasted before it meets the counter (DCL_UID).
208 */
209#define guard(name) \

211 __cleanup(dclg_##name##_release) = dclg_##name##_ctor

213/*
214 * scoped_guard(name, args) { ... }
215 * The guard lives in the for-init, so its scope *is* the loop: leaving
216 * the body -- normally, by break, or by returning out of it -- runs the
217 * cleanup. `_dclg_done` starts NULL so the body runs exactly once; the
218 * increment arm trips it, which is what makes a `continue` end the loop
219 * rather than re-acquire the lock.
220 */
221#define scoped_guard(name, args...) \
222 for (dclg_##name _dclg __cleanup(dclg_##name##_release) \
223 = dclg_##name##_ctor(args), *_dclg_done = 0; \
224 !_dclg_done; _dclg_done = (void*)1)
225
226/*
227 * scoped_cond_guard(name, fail, args) { ... }
228 * Same loop, but the macro ends on an `if/else` so the caller's compound
229 * statement becomes the *else* branch: taken only when the lock was
230 * actually acquired. On failure `fail` runs inside the if -- usually a
231 * `return`, which fires the cleanup over a NULL lock and therefore
232 * releases nothing, exactly right.
233 */
234#define scoped_cond_guard(name, fail, args...) \
235 for (dclg_##name _dclg __cleanup(dclg_##name##_release) \
236 = dclg_##name##_ctor(args), *_dclg_done = 0; \
237 !_dclg_done; _dclg_done = (void*)1) \
238 if (_dclg.lock == 0) { fail; } else
