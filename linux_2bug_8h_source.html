104 * Since detected data corruption should stop operation on the affected
105 * structures. Return value must be checked and sanely acted on by caller.
106 */
107/*
108 * WARN_ONCE(cond, fmt, ...) -- report a condition once per call site and
109 * return its truth, mainline's "warn but do not flood" spelling.
110 *
111 * The consumers are the DRM set's WARN_ONCE uses, reached three ways: the
112 * files that include this header directly, the ones that include it through
113 * <linux/kernel.h> -> <linux/bitmap.h> -> <linux/bitops.h> (which includes
114 * it at its bottom for exactly this), and drm_print.h:783's drm_WARN_ONCE,
115 * which folds a device prefix into it. drm_drv.c:579 is the shape that
116 * makes the return value load-bearing:
117 *
118 * if (drm_WARN_ONCE(dev, !recovery, "invalid recovery method %u\n", opt))
119 * break;
120 *
121 * -- the loop over set recovery method bits must not continue on an unknown
122 * bit, and the warning is not an error (unlike BUG). It is also why this
123 * cannot be a DCL shim function: fmt and __VA_ARGS__ are spliced into the
124 * report at the call site, which only a macro can do.
125 *
126 * The `static int` inside the statement expression is one object per
127 * expansion, which is what "once" means here; a per-frame call such as the
128 * vkms or vblank paths would otherwise print on every scanout. The body is
129 * kernel.h's WARN_ON_ONCE with the fmt form of WARN added, so the UART sink
130 * and the file:line trailer stay identical to every other warning this tree
131 * prints. Return is the condition's truth (mainline's contract), not
132 * WARN_ON_ONCE's `!cond`.
133 */
134#ifndef WARN_ONCE
135#define WARN_ONCE(cond, fmt, ...) \
136 ({ \
137 static int __warned_once; \
138 int __wret = !!(cond); \
139 if (__wret && !__warned_once) { \
140 __warned_once = 1; \
141 UARTDebugOut("[WARN] " fmt " (%s:%d)\r\n", \
142 ##__VA_ARGS__, __FILE__, __LINE__); \
143 } \
144 __wret; \
145 })
146#endif
