361/*
362 * BITS_PER_LONG -- bits in an `unsigned long`, which is NOT the same as bits
363 * in a pointer on this target.
364 *
365 * aarch64-unknown-windows is LLP64: __SIZEOF_LONG__ is 4 while
366 * __SIZEOF_SIZE_T__ and pointers are 8. This was hardcoded to 64, and every
367 * consumer of it inherits the mistake:
368 *
369 * - bitops.h's test_bit/set_bit compute the word index as (nr)/BITS_PER_LONG
370 * and the shift as (nr)%BITS_PER_LONG. With 64, bits 32..63 land in word 0
371 * instead of word 1, and `1UL << 32..63` is a shift by >= the width of a
372 * 32-bit type, which is undefined. Nothing hit it yet only because
373 * tty_struct_flags runs 0..11 and TTY_PORT_* runs 0..5, so every index is
374 * < 32 and the two spellings happen to agree.
375 * - GENMASK(h,l) shifts `~0UL` -- a 32-bit value -- by BITS_PER_LONG-1-h,
376 * i.e. by 62-h with h small: also undefined, and in practice a 0.
377 *
378 * It is computed rather than typed as 32 so it stays true if the target ever
379 * changes; this is exactly how BaseHdr/bordoisila_bits.h spells it, and that
380 * header's GENMASK had to be correct for the iMX8MP PLL register fields.
381 */
382/*
383 * 32, not (sizeof(long) * BITS_PER_BYTE): same number, but a number a #if can
384 * read. tty_ldsem.c:36 does `#if BITS_PER_LONG == 64` and rejected the
385 * expression form outright ("function-like macro 'sizeof' is not defined").
386 *
387 * It is genuinely 32 here and not a guess -- this is aarch64 under the LLP64
388 * ABI, where long is 4 bytes -- and 32 is what the reader wants: the only
389 * #if consumer sizes the active-count mask in struct ld_semaphore against
390 * atomic_long_t, whose counter is a long (atomic.h:141), so the word and the
391 * mask have to agree on 32 and the 64-bit arm would not.
392 *
393 * mainline's value on aarch64 is 64 because its long is 64 there; nothing
394 * about that portability transfers, since DCL's `long` is not the machine
395 * word. Consumers that want the machine word spell it as size_t.
396 */
397#define BITS_PER_LONG 32

732/*
733 * sized_strscpy(dst, src, count) -- copy a NUL-terminated string into a
734 * bounded buffer. The result is discarded by every caller in the tree, but
735 * the contract still has to be mainline's, because it is the *return* that
736 * tells a future caller whether the string fit:
737 *
738 * >= 0 bytes copied, excluding the terminator
739 * -E2BIG src did not fit in count bytes; dst is still NUL-terminated
740 *
741 * Byte loop rather than memcpy, and deliberately so: the terminator is
742 * written as part of the copy, so there is no window in which dst holds an
743 * unterminated run -- which is the whole point of strscpy against strncpy,
744 * whose mainline comment says the same thing in more words. count == 0 is
745 * -E2BIG because nothing could be stored, matching upstream.
746 *
747 * The name is mainline's for this spelling: mainline v7.2 has the 3-argument
748 * form here and builds strscpy() on top of it -- see the macros below.
749 *
750 * mainline: include/linux/string.h -> lib/string.c; see the note above this
751 * block for why the definition sits in this file instead.
752 */
753static inline long sized_strscpy(char* dst, const char* src, size_t count)
754{
755 size_t i;
756
757 if (count == 0)
758 return -E2BIG;
759
760 for (i = 0; i + 1 < count && src[i]; i++)
761 dst[i] = src[i];
762
763 if (src[i])
764 {
765 dst[0] = '\0';
766 return -E2BIG;
767 }
768
769 dst[i] = '\0';
770 return (long)i;
771}

773/*
774 * strscpy(dst, src[, count]) -- mainline's two-form spelling of the above.
775 *
776 * mainline dispatches on the argument count rather than overloading, because
777 * C cannot overload and this tree has both shapes in it at once:
778 *
779 * virtio_console.c:1635 strscpy(port->name, ..., sizeof(port->name));
780 * serial_core.c:2485 strscpy(address, "*unknown*", sizeof(address));
781 * drm_client.c:253 strscpy(fb->comm, client->name, TASK_COMM_LEN);
782 * drm_modes.c:2322 strscpy(cmdline_mode->name, mode->name,
783 * sizeof(cmdline_mode->name));
784 * drm_framebuffer.c:886 strscpy(fb->comm, current->comm);
785 *
786 * The last line is what made this necessary. Against the plain 3-argument
787 * function this file used to carry, it reported "too few arguments to
788 * function call, expected 3, have 2" -- and adding a default parameter was
789 * not the fix, because mainline's rule is that the 2-argument form is only
790 * valid when dst *is* an array whose size is known, and the count then comes
791 * from sizeof(dst). The four 3-argument sites pass a count that is not the
792 * whole array (a TASK_COMM_LEN into a longer field), which is the other half
793 * of why the argument cannot simply be dropped.
794 *
795 * The dispatch is mainline's in shape and different in mechanism, and the
796 * difference is worth recording: mainline selects with COUNT_ARGS() over a
797 * padded argument list, which needs the __ARG_PLACEHOLDER machinery for the
798 * empty case. Checked rather than assumed, clang counts an empty __VA_ARGS__
799 * as *one* empty argument, so the padded-list version selects the 3-argument

801 * missing.
802 *
803 * __VA_OPT__ is direct about the same question. __strscpy_sel returns its
804 * middle argument; the first __VA_OPT__ supplies "the count, then the
805 * 3-argument arm" only when a count was actually written, and when it was
806 * not, the slot that __VA_OPT__ left empty is precisely the middle one --
807 * which lands on __strscpy_2. __strscpy_2 is variadic so the trailing empty
808 * __VA_ARGS__ that strscpy() passes has somewhere to go.
809 *
810 * Both arms are always parsed, so dst is sizeof'd even on the 3-argument
811 * path; that is harmless and is the price of picking the arm after expansion
812 * rather than before.
813 */
814#define __strscpy_2(dst, src, ...) sized_strscpy(dst, src, sizeof(dst))
