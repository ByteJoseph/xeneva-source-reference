61static inline int cr_add_point(CharRange *cr, uint32_t v)
62{
63 if (cr->len >= cr->size) {
64 if (cr_realloc(cr, cr->len + 1))
65 return -1;
66 }
67 cr->points[cr->len++] = v;
68 return 0;
69}

71static inline int cr_add_interval(CharRange *cr, uint32_t c1, uint32_t c2)
72{
73 if ((cr->len + 2) > cr->size) {
74 if (cr_realloc(cr, cr->len + 2))
75 return -1;
76 }
77 cr->points[cr->len++] = c1;
78 cr->points[cr->len++] = c2;
79 return 0;
80}

140static inline int lre_is_space_byte(uint8_t c) {

144static inline int lre_is_id_start_byte(uint8_t c) {

153}

188#ifdef CONFIG_ALL_UNICODE
189 return lre_is_id_continue(c);
190#else
191 return !lre_is_space_non_ascii(c);
192#endif
