70struct dma_fence {
71 union {
72 spinlock_t *extern_lock;
73 spinlock_t inline_lock;
74 };
75 const struct dma_fence_ops __rcu *ops;
76 /*
77 * We clear the callback list on kref_put so that by the time we
78 * release the fence it is unused. No one should be adding to the
79 * cb_list that they don't themselves hold a reference for.
80 *
81 * The lifetime of the timestamp is similarly tied to both the
82 * rcu freelist and the cb_list. The timestamp is only set upon
83 * signaling while simultaneously notifying the cb_list. Ergo, we
84 * only use either the cb_list of timestamp. Upon destruction,
85 * neither are accessible, and so we can use the rcu. This means
86 * that the cb_list is *only* valid until the signal bit is set,
87 * and to read either you *must* hold a reference to the fence,
88 * and not just the rcu_read_lock.
89 *
90 * Listed in chronological order.
91 */
92 union {
93 struct list_head cb_list;
94 /* @cb_list replaced by @timestamp on dma_fence_signal() */
95 ktime_t timestamp;
96 /* @timestamp replaced by @rcu on dma_fence_release() */
97 struct rcu_head rcu;
98 };
99 u64 context;
100 u64 seqno;
101 unsigned long flags;
102 struct kref refcount;
103 int error;
104};

352static inline struct dma_fence *
353dma_fence_get_rcu_safe(struct dma_fence __rcu **fencep)
354{
355 do {
356 struct dma_fence *fence;
357
358 fence = rcu_dereference(*fencep);
359 if (!fence)
360 return NULL;
361
362 if (!dma_fence_get_rcu(fence))
363 continue;
364
365 /* The atomic_inc_not_zero() inside dma_fence_get_rcu()
366 * provides a full memory barrier upon success (such as now).
367 * This is paired with the write barrier from assigning
368 * to the __rcu protected fence pointer so that if that
369 * pointer still matches the current fence, we know we
370 * have successfully acquire a reference to it. If it no
371 * longer matches, we are holding a reference to some other
372 * reallocated pointer. This is possible if the allocator
373 * is using a freelist like SLAB_TYPESAFE_BY_RCU where the
374 * fence remains valid for the RCU grace period, but it
375 * may be reallocated. When using such allocators, we are
376 * responsible for ensuring the reference we get is to
377 * the right fence, as below.
378 */
379 if (fence == rcu_access_pointer(*fencep))
380 return rcu_pointer_handoff(fence);
381
382 dma_fence_put(fence);
383 } while (1);
384}

476/*
477 * dma_fence_test_signaled_flag - Only check whether a fence is signaled yet.
478 * @fence: the fence to check
479 *
480 * This function just checks whether @fence is signaled, without interacting
481 * with the fence in any way. The user must, therefore, ensure through other
482 * means that fences get signaled eventually.
483 *
484 * This function uses test_bit(), which is thread-safe. Naturally, this function
485 * should be used opportunistically; a fence could get signaled at any moment
486 * after the check is done.
487 *
488 * Return: true if signaled, false otherwise.
489 */
490static inline bool
491dma_fence_test_signaled_flag(struct dma_fence *fence)
492{
493 return test_bit(DMA_FENCE_FLAG_SIGNALED_BIT, &fence->flags);
494}

510static inline bool
511dma_fence_is_signaled_locked(struct dma_fence *fence)
512{
513 const struct dma_fence_ops *ops;
514
515 if (dma_fence_test_signaled_flag(fence))
516 return true;
517
518 rcu_read_lock();
519 ops = rcu_dereference(fence->ops);
520 if (ops && ops->signaled && ops->signaled(fence)) {
521 rcu_read_unlock();
522 dma_fence_signal_locked(fence);
523 return true;
524 }
525 rcu_read_unlock();
526
527 return false;
528}

575static inline bool __dma_fence_is_later(struct dma_fence *fence, u64 f1, u64 f2)
576{
577 /* This is for backward compatibility with drivers which can only handle
578 * 32bit sequence numbers. Use a 64bit compare when the driver says to
579 * do so.
580 */
581 if (test_bit(DMA_FENCE_FLAG_SEQNO64_BIT, &fence->flags))
582 return f1 > f2;
583
584 return (int)(lower_32_bits(f1) - lower_32_bits(f2)) > 0;
585}

595static inline bool dma_fence_is_later(struct dma_fence *f1,
596 struct dma_fence *f2)
597{
598 if (WARN_ON(f1->context != f2->context))

628static inline struct dma_fence *dma_fence_later(struct dma_fence *f1,
629 struct dma_fence *f2)
630{
631 if (WARN_ON(f1->context != f2->context))
632 return NULL;
633
634 /*
635 * Can't check just DMA_FENCE_FLAG_SIGNALED_BIT here, it may never
636 * have been set if enable_signaling wasn't called, and enabling that
637 * here is overkill.
638 */
639 if (dma_fence_is_later(f1, f2))
640 return dma_fence_is_signaled(f1) ? NULL : f1;
641 else
642 return dma_fence_is_signaled(f2) ? NULL : f2;
643}

703static inline ktime_t dma_fence_timestamp(struct dma_fence *fence)
704{
705 if (WARN_ON(!test_bit(DMA_FENCE_FLAG_SIGNALED_BIT, &fence->flags)))
706 return ktime_get();
707
708 while (!test_bit(DMA_FENCE_FLAG_TIMESTAMP_BIT, &fence->flags))
709 cpu_relax();
710
711 return fence->timestamp;
712}
