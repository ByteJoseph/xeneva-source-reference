4/*
5 * DCL <linux/kref.h> -- reference counting for tty_struct and tty_driver.
6 *
7 * Both of those are handed around as bare pointers by tty_io.c (a cdev open
8 * finds a driver, a driver finds a port, a port finds a tty) and freed from
9 * whichever path happens to run last. kref is the contract that says "last
10 * one out turns the lights off".
11 *
12 * Xeneva has no real concurrency in these paths -- <linux/mutex.h> records
13 * that plainly -- so the counter is atomic for mainline's reasons (the
14 * arithmetic must be indivisible on SMP) rather than because anyone here can
15 * actually race it. Keeping the operations atomic anyway costs a ldxr/stxr on
16 * aarch64 and avoids a header whose semantics differ from mainline's the day
17 * the second core turns on.
18 *
19 * kref_get_unless_zero() is included because tty_port_get() (tty_port.h:164)
20 * is written around it: it must decline to take a reference on an object
21 * already at zero, or a hangup path resurrects a port being destroyed.
22 */
23

60/*
61 * kref_put() -- mainline returns int (1 if this call released) because
62 * tty_release_struct() and friends branch on it. The release function takes
63 * the kref, so callers derive their container with container_of().
64 */
65static inline int kref_put(struct kref* kref, void (*release)(struct kref* kref))
66{
67 if (atomic_dec_and_test(&kref->refcount)) {
68 release(kref);
69 return 1;
70 }

72}
