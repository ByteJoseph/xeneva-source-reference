601static inline css_error set_clip(
602 css_computed_style *style, uint8_t type,
603 css_computed_clip_rect *rect)
604{
605 uint32_t *bits;
606
607 bits = &style->i.bits[CLIP_INDEX];
608
609 /*
610 26bits: tt tttr rrrr bbbb blll llTR BLyy:
611 units: top | right | bottom | left
612 opcodes: top | right | bottom | left | type
613 */
614 *bits = (*bits & ~CLIP_MASK) |
615 ((type & 0x3) << CLIP_SHIFT);
616
617 if (type == CSS_CLIP_RECT) {
618 *bits |= (((rect->top_auto ? 0x20 : 0) |
619 (rect->right_auto ? 0x10 : 0) |
620 (rect->bottom_auto ? 0x8 : 0) |
621 (rect->left_auto ? 0x4 : 0)) << CLIP_SHIFT);
622
623 *bits |= (((rect->tunit << 5) | rect->runit)
624 << (CLIP_SHIFT + 16));
625
626 *bits |= (((rect->bunit << 5) | rect->lunit)
627 << (CLIP_SHIFT + 6));
628
629 style->i.clip_a = rect->top;
630 style->i.clip_b = rect->right;
631 style->i.clip_c = rect->bottom;
632 style->i.clip_d = rect->left;
633 }
634
635 return CSS_OK;
636}

845 case CSS_COMPUTED_CONTENT_URI:

854 break;

856 c->data.counters.name =
857 lwc_string_ref(c->data.counters.name);
858 c->data.counters.sep =
859 lwc_string_ref(c->data.counters.sep);
860 break;
861 default:
862 break;
863 }
864 }

866 style->content = content;
867
868 /* Free existing array */
869 if (oldcontent != NULL) {
870 for (c = oldcontent;
871 c->type != CSS_COMPUTED_CONTENT_NONE; c++) {
872 switch (c->type) {
873 case CSS_COMPUTED_CONTENT_STRING:
874 lwc_string_unref(c->data.string);
875 break;
876 case CSS_COMPUTED_CONTENT_URI:
877 lwc_string_unref(c->data.uri);
878 break;
879 case CSS_COMPUTED_CONTENT_ATTR:
880 lwc_string_unref(c->data.attr);
881 break;

883 lwc_string_unref(c->data.counter.name);
884 break;
885 case CSS_COMPUTED_CONTENT_COUNTERS:
886 lwc_string_unref(c->data.counters.name);
887 lwc_string_unref(c->data.counters.sep);
888 break;
889 default:
890 break;
891 }
892 }
893
894 if (oldcontent != content)

896 }

898 return CSS_OK;
