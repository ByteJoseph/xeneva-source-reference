4/*
5 * DCL <linux/lockdep.h> -- the lock-dependency API, as no-ops.
6 *
7 * lockdep's whole job is to notice at *runtime* that a program acquires locks
8 * in two different orders and might therefore deadlock -- it tracks a graph of
9 * held locks while running. DCL has no lock graph, and its locks are no-ops
10 * (see <linux/spinlock.h> and <linux/mutex.h>), so there is no acquisition
11 * order to record in the first place.
12 *
13 * The reason this header exists rather than being skipped is that ported code
14 * *names* lockdep: tty_buffer.c:618 calls
15 *
16 * lockdep_set_subclass(&port->buf.lock, TTY_LOCK_SLAVE);
17 *
18 * as its final line, and 8250_port.c includes this header outright. A call
19 * with no declaration is an implicit-function-declaration error, so the call
20 * has to parse. What the call means upstream -- "this lock is being taken
21 * nested, tell lockdep it is a different instance so it does not report a
22 * recursion" -- has no referent when locks do not recurse, so the arguments
23 * are evaluated for side effects and dropped.
24 *
25 * lockdep_init_map_type() and friends take variadic arguments including
26 * format strings, hence the `...` form rather than a fixed parameter list:
27 * mainline's call sites pass different numbers of them.
28 */

38/*
39 * The `lock` argument is dropped rather than evaluated, which is what
40 * mainline's own !CONFIG_LOCKDEP arm does (lockdep.h:349, `do { (void)(name);
41 * (void)(key); } while (0)`). The difference matters because mainline's
42 * seqlock.h is a byte-port here and its __seqcount_init() (:48) and
43 * do_write_seqcount_begin_nested() (:478) both address `&s->dep_map` -- a
44 * member seqcount_t only has when CONFIG_DEBUG_LOCK_ALLOC is on, which it is
45 * not. Evaluating the first parameter type-checked that member access and
46 * turned every dma-resv.c inclusion into "no member named 'dep_map' in
47 * 'struct seqcount'". A macro parameter that never appears in its
48 * replacement list is not expanded at all, so the expression never reaches C.
49 *
50 * mainline evaluates name and key and so does this: both are live parameters
51 * of __seqcount_init(), so the (void) casts also keep them from warning.
52 *
53 * seqcount_acquire/seqcount_release are mainline's own two-step names
54 * (lockdep.h:528/:530), spelled the same way as the rwsem quartet below for
55 * the same reason -- the lockdep_map they would take does not exist here.
56 */

79/*
80 * The three names the drm core adds to that family. drm_auth.c:66 is the
81 * interesting one:
82 *
83 * lockdep_assert_once(lockdep_is_held(&fpriv->master_lookup_lock) ||
84 * lockdep_is_held(&dev->master_mutex));
85 *
86 * and drm_syncobj.c:385/:1020 hand lockdep_is_held() to
87 * rcu_dereference_protected()'s condition slot. mainline answers "does this
88 * task hold this lock" out of the held-lock graph it maintains while running
89 * (lockdep.h:252); DCL has no graph and its locks are no-ops, so nothing is
90 * ever held and no order can be violated -- `1` is the answer that makes
91 * every one of these assertions pass, which is what an assertion that can
92 * only fail with lockdep is supposed to do. The RCU condition is a hint
93 * DCL's rcu_dereference_protected() drops anyway (rcupdate.h).
94 *
95 * lockdep_assert_once() *does* evaluate its argument, unlike its neighbours
96 * above that take no value and unlike mainline's !CONFIG_LOCKDEP form
97 * (`do { } while (0)`, lockdep.h:390): the argument is the expression under
98 * assertion, and evaluating it is what keeps the lock it names used -- the
99 * same reason lockdep_set_subclass() above evaluates its arguments. The
100 * compiler then folds it away, since lockdep_is_held() is the constant 1.
101 * A macro rather than an extern declaration (which is what upstream's
102 * !CONFIG_LOCKDEP build uses) follows lockdep_match_class() above, and both
103 * files that reference it reach this header, so no definition is needed.
104 */
105#define lockdep_is_held(lock) (1)
106#define lockdep_assert_once(cond) do { (void)(cond); } while (0)

109/*
110 * The rwsem acquire/release quartet -- tty_ldsem.c:300-310, which brackets
111 * every ldsem_up/_down with them:
112 *
113 * rwsem_acquire_read(&sem->dep_map, subclass, 0, _RET_IP_);
114 * count = atomic_long_add_return(LDSEM_READ_BIAS, &sem->count);
115 * ...
116 * rwsem_release(&sem->dep_map, _RET_IP_);
117 *
118 * mainline's versions are real when CONFIG_LOCKDEP and CONFIG_PROVE_LOCKING
119 * are on and no-ops otherwise, so this is the arm every non-debug build
120 * already takes. What is *not* borrowed is mainline's `(void)(l)` argument
121 * handling: these four take a `struct lockdep_map *` that struct ld_semaphore
122 * does not carry (tty_ldisc.h), and evaluating it would mean adding a member
123 * whose only purpose is to be addressed and forgotten. Dropping the
124 * parameter instead means the expression never reaches C at all -- a macro
125 * parameter that does not appear in the replacement list is not expanded --
126 * which also leaves _RET_IP_ (a lockdep return-address macro) as a bare
127 * identifier that never has to be defined.
128 *
129 * Argument order is mainline's: lockdep_map, subclass[, read], ip.
130 */

144 * `&connector_list_iter_dep_map`, and mainline only *defines* that object
145 * inside `#ifdef CONFIG_LOCKDEP` (drm_connector.c:1032-1036). The two call

147 * mainline's no-op spelling of these macros discards its arguments before

152 * macros must not be written `(void)(l)`: evaluating the argument would
