2 * DCL rcupdate.h -- no-op RCU for single-CPU Xeneva.
3 *
4 * mainline drm/xarray call the read-side + grace-period API; here there is no
5 * preemption and no callback queue, so rcu_dereference degrades to a plain
6 * load and call_rcu runs the callback immediately. Immediate execution keeps
7 * the xa_node free path honest: mainline never touches an object after
8 * call_rcu() on it, which is exactly the contract a synchronous free needs.

54/*
55 * RCU_INITIALIZER(), unrcu_pointer(), rcu_pointer_handoff() -- mainline's
56 * three annotation helpers (rcupdate.h:525/:491/:788), with the sparse
57 * markers left out because they are empty here: __rcu is above and __force
58 * is kernel.h:239, so `(typeof(*(v)) __force __rcu *)(v)` and
59 * `(typeof(*(v)) *)(v)` are the same cast. What remains of the cast is its
60 * one real job -- the type check -- so it is kept rather than dropped.
61 *
62 * RCU_INITIALIZER() adds the annotation and unrcu_pointer() removes it, and
63 * both have exactly one caller: dma-fence-chain.c:65, the cmpxchg() in
64 * dma_fence_chain_alloc(), which reads
65 *
66 * tmp = unrcu_pointer(cmpxchg(&chain->prev, RCU_INITIALIZER(prev),
67 * RCU_INITIALIZER(replacement)));
68 *
69 * and reaches this header via dma-fence-chain.h -> dma-fence.h. The first
70 * argument of that cmpxchg is `struct dma_fence __rcu **`, so the casts have
71 * to land on that same pointer type or cmpxchg's argument check rejects
72 * them; `typeof(*(v))` is the pointee with the annotation already stripped,
73 * which is why the pointer spelling is the right one to cast to.
74 *
75 * rcu_pointer_handoff()'s one caller is dma-fence.h:380, in
76 * dma_fence_get_rcu_safe(): it marks the moment a reference stops being
77 * RCU-protected and becomes an owned one, which is a statement about the
78 * caller, not a change to the pointer.
79 */
80#define RCU_INITIALIZER(v) ((typeof(*(v)) *)(v))
81#define unrcu_pointer(p) ((typeof(*(p)) *)(p))
82#define rcu_pointer_handoff(p) (p)
