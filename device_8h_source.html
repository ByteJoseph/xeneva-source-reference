31/*
32 * struct device_attribute and DEVICE_ATTR_RW().
33 *
34 * mainline spells device_attribute with __SYSFS_FUNCTION_ALTERNATIVE to get
35 * a const-correct show() alongside show_const(); DCL writes the plain two
36 * members, because nothing in this tree takes the address of a const variant
37 * and the alternative exists to serve a different property system. The field
38 * *names* -- attr, show, store -- are mainline's, and they matter: 8250_port.c
39 * reads the attribute back as &dev_attr_rx_trig_bytes.attr, so a differently
40 * named member would not compile rather than silently misbehave.
41 */
42struct device_attribute {
43 struct attribute attr;

45 char* buf);
46 ssize_t (*store)(struct device* dev, struct device_attribute* attr,
47 const char* buf, size_t count);

69/*
70 * dev_get_drvdata() / dev_set_drvdata().
71 *
72 * mainline stores the pointer in `dev->driver_data`. DCL's struct device
73 * (in <linux/fs.h>) is an offset-forged blob -- `kobj_name` at 0, `devt` at
74 * 708, `_tail[88]` -- reproduced field-for-field so DCL and Xeneva's own
75 * devfs agree about where `devt` lives. Growing it by a pointer would move
76 * nothing *and* break that agreement in the direction that matters: Xeneva
77 * allocates the object with its own sizeof, and a write at offset 800 of an
78 * 800-byte allocation is a heap overrun, not a warning.
79 *
80 * So the accessors exist and are honest about what they answer. Nobody sets
81 * drvdata anywhere in this tree -- there is no dev_set_drvdata call in DCL,
82 * in 8250_port.c or in serial_core.c -- and the readers are all sysfs show
83 * callbacks (8250_port.c:3026/3071, serial_core.c:2824-2898) reached only
84 * when a sysfs file exists. It does not: DCL/linux_cdev_shim.c's
85 * sysfs_create_group() is a deliberate no-op with the comment "no sysfs tree;
86 * success keeps add_port() on the happy path", so uart_add_one_port() succeeds
87 * and creates no attributes, and these functions are unreachable.
88 *
89 * NULL is therefore the truthful answer today, not a placeholder waiting to
90 * be filled in -- see the stage-4 note in DCL_TODO.md: when tty_io.c brings
91 * device registration over, this is where a real slot has to be found for
92 * drvdata, and the choice is between extending struct device (and reconciling
93 * it with Xeneva's sizeof) and a side table. Neither is needed until
94 * something sets it.
95 */
96static inline void* dev_get_drvdata(const struct device* dev)
97{
98 (void)dev;
99 return NULL;
100}

111/*
112 * DEVICE_ATTR(_name, _mode, _show, _store) -- the four-argument spelling
113 * mainline defines in <linux/device.h>. The three one-argument spellings
114 * above cover everything else in this tree (8250 and serial_core use only
115 * _RW/_RO); virtio_console.c:1250 is the first caller that names its own
116 * mode -- `static DEVICE_ATTR(name, S_IRUGO, show_port_name, NULL);` -- so
117 * this exists for it.
118 *
119 * It is a *declaration* macro exactly like DEVICE_ATTR_RW above: the
120 * expansion has to carry the struct keyword, because the call site writes
121 * `static` and expects the expansion to start at the type. The mode comes
122 * from <linux/sysfs.h>, which this header already includes, and `store` is
123 * allowed to be NULL (it is, at that one call site) because __ATTR puts it
124 * straight into a struct member.
125 */
126#define DEVICE_ATTR(_name, _mode, _show, _store) \
127 struct device_attribute dev_attr_##_name = __ATTR(_name, _mode, \
128 _show, _store)

167/*
168 * dev_name() -- the device's name as devfs set it. DCL's struct device has
169 * the kobject name as its first member (kobj_name at offset 0, the same
170 * slot mainline's dev->kobj.name occupies), and linux_cdev_shim.c:449 fills
171 * it in, so this reads a live value rather than fabricating one.
172 *
173 * The NULL arm is belt-and-braces: the one caller (serial_core.c:2490) already
174 * guards `port->dev ? dev_name(port->dev) : ""`, but `%s` handed a NULL on a
175 * formatter with no "(null)" convention is a fault, and a name that is absent
176 * is better reported as empty than as a crash.
177 */
178static inline const char* dev_name(const struct device* dev)
179{
180 if (!dev || !dev->kobj_name)
181 return "";
182 return dev->kobj_name;
183}

185/*
186 * device_find_child() -- search a device's children for one the predicate
187 * accepts. DCL has no child-device tree: devices are created flat by
188 * device_create() and never linked under a parent, so the answer is always
189 * "none".
190 *
191 * The consequence is visible one line later at both call sites (serial_core.c
192 * :2305 and :2383): `tty_dev` comes back NULL, so the wake-arming arm at
193 * :2306 is skipped (it tests tty_dev first) and the one at :2383 calls
194 * device_may_wakeup(NULL), which is written to tolerate it. Both paths then
195 * fall through to put_device(tty_dev) with NULL, which the shim ignores.
196 *
197 * Returning NULL rather than a fabricated device is what keeps
198 * serial_match_port() -- whose job is to compare dev_t values against a real
199 * child -- from ever being handed something it would have to make sense of.
200 */
201static inline struct device* device_find_child(
202 struct device* parent, void* data,
203 int (*match)(struct device* dev, const void* data))
204{
205 (void)parent;
206 (void)data;
207 (void)match;
208 return NULL;
