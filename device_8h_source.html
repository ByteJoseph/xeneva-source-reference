63/*
64 * struct device_attribute and DEVICE_ATTR_RW().
65 *
66 * mainline spells device_attribute with __SYSFS_FUNCTION_ALTERNATIVE to get
67 * a const-correct show() alongside show_const(); DCL writes the plain two
68 * members, because nothing in this tree takes the address of a const variant
69 * and the alternative exists to serve a different property system. The field
70 * *names* -- attr, show, store -- are mainline's, and they matter: 8250_port.c
71 * reads the attribute back as &dev_attr_rx_trig_bytes.attr, so a differently
72 * named member would not compile rather than silently misbehave.
73 */
74struct device_attribute {
75 struct attribute attr;
76 ssize_t (*show)(struct device* dev, struct device_attribute* attr,
77 char* buf);
78 ssize_t (*store)(struct device* dev, struct device_attribute* attr,
79 const char* buf, size_t count);
80};

87 * static DEVICE_ATTR_RW(rx_trig_bytes);

125 * DEVICE_ATTR(_name, _mode, _show, _store) -- the four-argument spelling
126 * mainline defines in <linux/device.h>. The three one-argument spellings
127 * above cover everything else in this tree (8250 and serial_core use only
128 * _RW/_RO); virtio_console.c:1250 is the first caller that names its own

180/*
181 * dev_name() -- the device's name as devfs set it. DCL's struct device has
182 * the kobject name as its first member (kobj_name at offset 0, the same
183 * slot mainline's dev->kobj.name occupies), and linux_cdev_shim.c:449 fills
184 * it in, so this reads a live value rather than fabricating one.
185 *
186 * The NULL arm is belt-and-braces: the one caller (serial_core.c:2490) already
187 * guards `port->dev ? dev_name(port->dev) : ""`, but `%s` handed a NULL on a
188 * formatter with no "(null)" convention is a fault, and a name that is absent
189 * is better reported as empty than as a crash.

194 return "";
195 return dev->kobj_name;

198/*
199 * device_find_child() -- search a device's children for one the predicate
200 * accepts. DCL has no child-device tree: devices are created flat by
201 * device_create() and never linked under a parent, so the answer is always
202 * "none".
203 *
204 * The consequence is visible one line later at both call sites (serial_core.c
205 * :2305 and :2383): `tty_dev` comes back NULL, so the wake-arming arm at
206 * :2306 is skipped (it tests tty_dev first) and the one at :2383 calls
207 * device_may_wakeup(NULL), which is written to tolerate it. Both paths then
208 * fall through to put_device(tty_dev) with NULL, which the shim ignores.
209 *
210 * Returning NULL rather than a fabricated device is what keeps
211 * serial_match_port() -- whose job is to compare dev_t values against a real
212 * child -- from ever being handed something it would have to make sense of.
213 */
214static inline struct device* device_find_child(
215 struct device* parent, void* data,
216 int (*match)(struct device* dev, const void* data))
217{
218 (void)parent;
219 (void)data;
220 (void)match;
221 return NULL;
222}
