4/*
5 * DCL <linux/dma-mapping.h> -- the DMA-buffer fence markers, and nothing else.
6 *
7 * mainline's header is 27 KB: the dma_alloc_coherent family, the dmam_alloc
8 * family, dma_map_sg, dma_sync_single_*, the attrs and direction enums, the
9 * DMA_MASK checks. None of it has a caller in this tree, and all of it would
10 * need a DMA layer behind it -- which Xeneva does not have. What a ported
11 * source reaches for today is a pair of macros that fence a buffer inside a
12 * struct, so that is what this is. If a driver needs real allocation, that is
13 * the moment to decide whether the linear map is the answer and to write the
14 * rest of the header then.
15 *
16 * The two macros are mainline's, including the quirk that matters:
17 *
18 * #define __dma_from_device_group_begin(GROUP) \
19 * __cacheline_group_begin(GROUP) ____dma_from_device_aligned
20 *
21 * NO trailing semicolon (the caller's `;` terminates the member), and GROUP
22 * may be empty -- virtio-rng.c:34 and :38 write `__dma_from_device_group_begin();`
23 * with nothing inside the parens. See the note in <linux/cache.h>.
24 *
25 * ARCH_HAS_DMA_MINALIGN is not defined for this target, so
26 * ____dma_from_device_aligned is the empty spelling mainline falls back to
27 * too -- no extra alignment. That is not an oversight: virtio-rng.c guards
28 * its buffer with `#if SMP_CACHE_BYTES < 32` and sizes it itself, so the
29 * cacheline (64) already governs the layout. ARCH_DMA_MINALIGN itself is
30 * defined now, in <asm/cache.h>, for drm_managed.c's struct drmres -- but it
31 * is ARCH_HAS_DMA_MINALIGN that the #ifdef below tests, and that one stays
32 * undefined, so the layout here does not move.
33 */

58/*
59 * dma_alloc_coherent() / dma_free_coherent() -- the one pair the fence macros
60 * above were standing in front of.
61 *
62 * mainline's header declares a whole family (the dmam_ allocators, dma_map_sg,
63 * the direction enum, the mask checks) and this comment used to say none of it
64 * had a caller. virtio_console.c:439 is a caller:
65 *
66 * buf->buf = dma_alloc_coherent(buf->dev, buf_size, &buf->dma, GFP_KERNEL);
67 *
68 * -- the rproc-serial arm of alloc_buf(), the arm DCL folds away (see
69 * CONFIG_REMOTEPROC in <linux/autoconf.h>). It has to *compile*, and it has to
70 * link if the compiler ever declines to fold it, so the two declarations are
71 * here and the bodies are in DCL/linux_mm_shim.c.
72 *
73 * What they do: dma_alloc_coherent() is a plain kmalloc with the address
74 * handed back as the dma_addr_t as well. That is sound rather than convenient
75 * -- Xeneva's virtio path takes a *virtual* address and translates it at kick
76 * time (see <linux/scatterlist.h> on why buf is a linear-map pointer), so the
77 * only thing this tree ever does with buf->dma is hand it straight back to
78 * dma_free_coherent() at :373, and a token that round-trips is a token that
79 * works. There is no streaming mapping here to need a bus address, and no
80 * IOMMU to need an address the CPU cannot reach.
81 *
82 * The parameter types are mainline's, so a caller written against mainline
83 * compiles without a cast: `struct device *` (incomplete is fine -- nothing
84 * here dereferences it), size_t, `dma_addr_t *`, gfp_t.
85 */

87 dma_addr_t* dma_handle, gfp_t gfp);

129/*
130 * dma_map_sg() -- map `nents` sg entries for the device and report how many
131 * DMA segments they became.
132 *
133 * The answer is `nents` unchanged, which is the identity mapping: with no
134 * IOMMU and no bounce buffers, an entry is already contiguous and already
135 * addressable, so nothing coalesces. That is the same reasoning
136 * dma_alloc_coherent() above uses for handing the virtual address back as the
137 * dma_addr_t -- and drm_need_swiotlb() in drm_cache.c answering false is the
138 * other half of it. The @attrs word is mainline's and, at this mapping
139 * granularity, has nothing to act on: DMA_ATTR_SKIP_CPU_SYNC in particular
140 * only defers a sync that never happens here.
141 *
142 * Declared here, defined in DCL/linux_mm_shim.c beside the two allocators.
143 */

149/*
150 * dma_map_sgtable() -- mainline's wrapper, and mainline's body verbatim: map
151 * the list, and on success rewrite nents to the count that survived. The
152 * -EINVAL arm is mainline's for the "mapped to nothing" case, which cannot
153 * happen against the dma_map_sg() above but is kept so a real mapper could
154 * be dropped in without the contract changing.
155 *
156 * Inline rather than a shim because the whole of it is the wrapper -- there is
157 * no state of its own, and mainline keeps it in the header too.
158 */
159static inline int dma_map_sgtable(struct device* dev, struct sg_table* sgt,
160 enum dma_data_direction dir,
161 unsigned long attrs)
162{
163 int nents;
164
165 (void)attrs; /* see the note above: nothing here reads it */
166
167 nents = dma_map_sg(dev, sgt->sgl, sgt->nents, dir);
168 if (nents <= 0)
169 return nents < 0 ? nents : -EINVAL;
170
171 sgt->nents = (unsigned int)nents;
172 return 0;
173}
