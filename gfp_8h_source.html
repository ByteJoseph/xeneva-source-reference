4/*
5 * DCL <linux/gfp.h> -- allocation flags. mainline's is 16.8 KB and brings
6 * mmzone.h -> nodemask.h -> numa.h -> percpu behind it, i.e. the entire NUMA
7 * surface, in exchange for two constants.
8 *
9 * DCL already owns both, in <linux/kernel.h>:
10 * :55 typedef unsigned int gfp_t;
11 * :111 #define GFP_KERNEL 0
12 * :112 #define GFP_ATOMIC 1
13 * and <linux/slab.h> already accepts mainline's two-argument kmalloc(size,
14 * gfp) by dropping the flag (`#define kmalloc(size, flags) dcl_kmalloc(size)`).
15 * So there is nothing to *define* here -- this header exists so a mainline
16 * source that writes `#include <linux/gfp.h>` resolves, and it routes rather
17 * than redefines: a second gfp_t and a second GFP_KERNEL in one translation
18 * unit would be a redefinition error, and mainline's GFP_KERNEL is a bitmask
19 * of __GFP_RECLAIM|__GFP_IO|__GFP_FS that nothing here interprets.
20 *
21 * Routing the two constants is not the same as having no vocabulary, though,
22 * and the include below is the difference: mainline's gfp.h includes
23 * gfp_types.h, which carries the individual __GFP_* flags. drm_managed.h
24 * needs one of them -- drmm_kzalloc() and drmm_kzalloc_array() write
25 * `flags | __GFP_ZERO`. Without the include those two inline functions fail
26 * in the *header* that declares them, so no drm caller ever reaches the
27 * -ENOSYS path gfpflags_allow_blocking() sits beside.
28 */
29#include <linux/kernel.h>
30#include <linux/gfp_types.h>
