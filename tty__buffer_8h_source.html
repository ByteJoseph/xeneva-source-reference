4/*
5 * DCL <linux/tty_buffer.h> -- struct tty_buffer and the byte/flag accessors.
6 *
7 * Ported from mainline include/linux/tty_buffer.h (v7.2), field for field.
8 *
9 * The layout is the part that cannot be reinterpreted: `data[]` is a flexible
10 * array holding *both* the characters and, when flags are in use, a parallel
11 * flag byte per character immediately after them -- flag_buf_ptr() finds the
12 * flag array by adding b->size (the character capacity) to char_buf_ptr().
13 * So `size` is not "how much is in use", it is the fixed stride separating
14 * the two arrays inside one allocation, and tty_buffer_reset() setting
15 * size = 0 on the sentinel is what makes an empty buffer claim no space at
16 * all.
17 *
18 * `used`/`commit`/`read`/`lookahead` are the producer/consumer cursor set:
19 * used -- bytes the driver has written (not yet visible to the ldisc)
20 * commit -- bytes made visible, published by tty_flip_buffer_commit()
21 * read -- bytes the ldisc has consumed
22 * lookahead-- the ldisc's next candidate position when a consumer stopped
23 * short (tty_buffer.c:404 walks the chain from here)
24 * The commit/read gap is what tty_flip_buffer_push() exists to close.
25 *
26 * The union is the point of the llist: a buffer is either on the live queue
27 * (->next) or on the free list (->free), never both, so the two share storage.
28 */

60/*
61 * struct tty_bufhead embeds a `sentinel` tty_buffer, and struct tty_buffer
62 * ends in a flexible array -- so a struct with a variable-sized member sits in
63 * the middle of another struct. That is a GNU extension and clang says so
64 * (-Wgnu-variable-sized-type-not-at-end), once per translation unit, because
65 * it comes from a header.
66 *
67 * mainline's layout is exactly this (include/linux/tty_buffer.h v7.2): the
68 * sentinel is what an empty buffer head points at, `size` reads 0 after
69 * tty_buffer_reset(&buf->sentinel, 0), and the fields after it (`free`,
70 * `mem_used`, `mem_limit`, `tail`) are what the sentinel's zero-length data[]
71 * leaves room for. It is deliberate, it is the whole design of the empty
72 * buffer, and it cannot be moved to the end without changing what
73 * tty_buffer_find() compares against. So the layout is kept and the notice is
74 * scoped to this struct: it says "extension", not "bug".
75 */
76#if defined(__clang__)
77#pragma clang diagnostic push
78#pragma clang diagnostic ignored "-Wgnu-variable-sized-type-not-at-end"
79#endif
