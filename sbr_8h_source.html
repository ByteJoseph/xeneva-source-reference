139/* do y <<= n, clipping to range [-2^30, 2^30 - 1] (i.e. output has one guard bit) */
140#define CLIP_2N_SHIFT30(y, n) { \
141 int sign = (y) >> 31; \
142 if (sign != (y) >> (30 - (n))) { \
143 (y) = sign ^ (0x3fffffff); \
144 } else { \
145 (y) = (y) << (n); \
146 } \
147}
148
149/*
150#define CLIP_2N(y, n) { \
151 int sign = (y) >> 31; \
152 if (sign != ((y) >> (n))) { \
153 (y) = sign ^ ((1 << (n)) - 1); \
154 } \
155}
156*/
