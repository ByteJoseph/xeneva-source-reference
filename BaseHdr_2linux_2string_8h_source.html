5 * DCL <linux/string.h> -- memcpy/memset/memcmp and friends.
6 *
7 * These are Xeneva's own routines (BaseHdr/string.h), not mainline's lib/:
8 * the tty layer copies into and out of flip buffers with them, and a second
9 * implementation would only be a second thing to get right. Mainline includes
10 * <linux/string.h> for the same reason -- to see the prototypes -- so the
11 * header exists to resolve the include and nothing more.
12 *
13 * tty_buffer.c's use is exactly memcpy() into char_buf_ptr(), memset() over
14 * the flags byte for a run of same-flagged characters, and the zeroing of
15 * consumed bytes in receive_buf() so a buffer can be handed on.
16 *
17 * The DRM/KMS stage then made this header carry real content, all of it for
18 * callers that reach no other string header:
19 *
20 * - strnchr, str_has_prefix, mem_is_zero and strtomem_pad, which mainline
21 * keeps here. drm_modes.c parses the video= command line with the first
22 * two, drm_edid.c asks whether a 128-byte block is all zero, and
23 * drm_connector.c:614-615 copies an HDMI vendor/product string into a
24 * fixed-width, possibly non-NUL-terminated field.
25 *
26 * - str_yes_no, re-exported from <linux/string_choices.h> rather than
27 * copied. drm_edid.c:6319 prints an HDMI capability as yes/no and

29 * carrier in this tree. Copying the body would collide with the header

57/*
58 * strnchr(s, count, c) -- strchr() bounded to `count` bytes.
59 *
60 * The difference from strchr() is the whole reason drm_modes.c uses it
61 * (drm_modes.c:2302, :2390, :2392, :2396): the string being searched is a
62 * slice of a larger command-line buffer -- the connector name before the
63 * ':' or '-' or '@' separators -- and it is *not* NUL-terminated at the
64 * point the caller cares about. `name_end`/`options_off` are how long that
65 * slice is. A plain strchr() would run past the slice into the rest of the
66 * option string and report the separator belonging to a later field, so
67 * `video=HDMI-A-1:1280x720@60` would be read as a name that itself contains
68 * '@'. mainline's body is this loop in lib/string.c; count is decremented
69 * before the NUL test so `count == 0` answers NULL rather than reading s.
70 */
71static inline char *strnchr(const char *s, size_t count, int c)
72{
73 for (; count-- && *s != '\0'; ++s)
74 if (*s == (char)c)
75 return (char *)s;
76 return NULL;
77}

79/*
80 * str_has_prefix(str, prefix) -- strlen(prefix) if str starts with prefix,
81 * else 0.
82 *
83 * The return has to be the *length*, not a boolean, for the one call site:
84 * drm_modes.c:2318 matches a command-line name against each named mode with
85 *
86 * ret = str_has_prefix(name, mode->name);
87 * if (ret != name_end)
88 * continue;
89 *
90 * where `name` is the unbounded slice and `name_end` its measured length, so
91 * the comparison is "starts with mode->name *and* is exactly that long". A
92 * strncmp-style 0/1 answer would compare `1 != name_end` and every named
93 * mode would be rejected except a one-character one. strncmp is bounded by
94 * len, so a short `str` (NUL before len) also answers 0, as mainline
95 * documents.
96 */
97static inline size_t str_has_prefix(const char *str, const char *prefix)
98{
99 size_t len = strlen(prefix);
100
101 return strncmp(str, prefix, len) == 0 ? len : 0;
102}

126/*
127 * strtomem_pad(dest, src, pad) -- copy the NUL-terminated `src` into the
128 * fixed-width array `dest`, filling whatever is left with `pad`.
129 *
130 * drm_connector.c:614-615 is the call:
131 *
132 * strtomem_pad(connector->hdmi.vendor, vendor, 0);
133 * strtomem_pad(connector->hdmi.product, product, 0);
134 *
135 * where the fields are `unsigned char vendor[DRM_CONNECTOR_HDMI_VENDOR_LEN]
136 * __nonstring` (drm_connector.h:1933) -- HDMI 2.1 wants them fixed-width with
137 * no NUL, and a bytes-typed readout must see the padding rather than
138 * whatever the struct held before. The source strings come from userspace
139 * through the connector init path and are not guaranteed to be shorter than
140 * the field, so the copy must stop at sizeof(dest) without reading the
141 * source past it -- the loop below reads src[i] only while i < dest_len,
142 * which is strnlen(src, dest_len)'s guarantee without needing a strnlen DCL
143 * does not carry.
144 *
145 * The helper is a real inline and the macro supplies sizeof(dest) and the
146 * compile-time array check: dest is an array, not a pointer, and the size is
147 * not an argument (same contract as mainline's strtomem_pad).
148 * __must_be_array(dest) is the check -- its _Static_assert stops the build
149 * for a pointer, where sizeof would silently be the pointer's size and the
150 * copy would truncate -- but its *value* is not part of the length here.
151 * mainline adds it (it folds to 0 on GNU/Linux), and on this target it does
152 * not: aarch64-unknown-windows gives an empty struct -- the shape
153 * __BUILD_BUG_ON_ZERO_MSG folds to -- size 4, so
154 * `sizeof(dest) + __must_be_array(dest)` made every copy run 4 bytes past
155 * the field until the check was split onto its own line. The do/while is
156 * mainline's shape too; both call sites are statements.
157 */
158static inline void __dcl_strtomem_pad(void *dest, size_t dest_len,
159 const char *src, char pad)
160{
161 unsigned char *d = (unsigned char *)dest;
162 size_t i = 0;
163
164 while (i < dest_len && src[i] != '\0') {
165 d[i] = (unsigned char)src[i];
166 i++;
167 }
168 while (i < dest_len)
169 d[i++] = (unsigned char)pad;
170}
