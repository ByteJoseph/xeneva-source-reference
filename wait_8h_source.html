4/*
5 * DCL <linux/wait.h> -- wait queues, with an honest answer to the one thing
6 * this port cannot do: sleep.
7 *
8 * There is no native equivalent in Xeneva and no scheduler to block in, so
9 * the queue is built for real -- the lists, the entries, the wake functions
10 * and the bookkeeping are all genuine -- while the *waiting* is done by
11 * polling, bounded. The distinction matters: what must not happen is a driver
12 * that believes it slept and then loops on a condition that only ever changes
13 * while it is spinning.
14 *
15 * Three call sites decide the design (the measured blocking surface of the
16 * nine files: 7 wait-queue ops, 5 schedule()):
17 *
18 * n_tty.c:2232 / :2354 -- blocking read and write, via
19 * DEFINE_WAIT_FUNC(wait, woken_wake_function) + add_wait_queue + a loop
20 * that ends in wait_woken(). Both are stage 5's EAGAIN-first cut; until
21 * that lands, an O_NONBLOCK fd returns -EAGAIN before reaching here
22 * (n_tty checks tty_io_nonblock() first), and a blocking fd with no
23 * data spins in wait_woken() rather than sleeping. The queue itself
24 * must therefore be correct -- it is what the cut will hang off.
25 *
26 * serial_core.c:3194 -- wait_event(state->remove_wait,
27 * !atomic_read(&state->refcount)) during port removal. The condition is
28 * normally already true (refcount drops to zero on the line above), so
29 * the bounded poll sees it on the first test and returns.
30 *
31 * serial_core.c:1205 / tty_io.c:3118 -- DECLARE_WAITQUEUE and
32 * init_waitqueue_head at construction, no waiting at all.
33 *
34 * wait_event() polls for DCL_WAIT_EVENT_MS and, if the condition still has
35 * not come true, says so through dcl_wait_event_timeout() before falling
36 * through. Mainline would block indefinitely there; falling through is the
37 * wrong answer dressed as the right one, which is exactly why it is logged
38 * instead of being quiet -- an unconditional block is a hung kernel and an
39 * unconditional fall-through is a corrupt one, and neither should look like
40 * a normal boot.
41 *
42 * add_wait_queue/remove_wait_queue are real list operations with no lock
43 * held across them (DCL's spinlocks are no-ops -- see <linux/spinlock.h>),
44 * so a wake arriving concurrently with an add is not excluded. That matches
45 * the locking contract of everything else here.
46 */

182/*
183 * _dcl_wait_core(wq_head, condition) -- the one polling loop every
184 * wait_event* macro below is built from: DCL_WAIT_EVENT_MS passes of
185 * mdelay(1) while the condition is false, dcl_wait_event_timeout() on the
186 * way out, then the condition read one last time.
187 *
188 * Spelled once because the four macros that use it differ only in what they
189 * make of an expiry -- nothing, -ERESTARTSYS, 0, or the caller's own
190 * timeout -- which is what each of their comments is about. Four copies of
191 * the loop would let a change to the budget or the tick reach one variant
192 * and not the others, and a caller cannot see that until a port stops
193 * draining.
194 *
195 * Evaluates to 1 when the condition held and 0 when the budget ran out, so
196 * the caller picks its own result instead of re-testing the condition: it is
197 * read once per millisecond and once here, which is the side-effect-free
198 * contract the note on wait_event() describes.
199 */
200#define _dcl_wait_core(wq_head, condition) \
201 ({ \
202 unsigned int _dcl_i = DCL_WAIT_EVENT_MS; \
203 int _dcl_ok; \
204 (void)(wq_head); \
205 while (!(condition) && _dcl_i) { \
206 _dcl_i--; \
207 mdelay(1); \
208 } \
209 _dcl_ok = !!(condition); \
210 if (!_dcl_ok) \
211 dcl_wait_event_timeout(__func__); \
212 _dcl_ok; \
213 })

215/*
216 * wait_event(wq, condition) -- poll, bounded.
217 *
218 * The condition is evaluated once per millisecond and once at the end; it
219 * must therefore be side-effect free, which mainline's own wait_event()
220 * requires too (it evaluates it on every wake as well).
221 */
222#define wait_event(wq_head, condition) \
223 do { \
224 (void)_dcl_wait_core(wq_head, condition); \
225 } while (0)

228/*
229 * wait_event_interruptible(wq, condition) -- tty_ioctl.c:484, which does
230 *
231 * retval = wait_event_interruptible(tty->write_wait, !tty_chars_in_buffer(tty));
232 * if (retval < 0)
233 * return retval;
234 *
235 * so the value is the whole contract: 0 means "the buffer drained", negative
236 * means "give up and unwind", and a 0 on timeout would let the caller proceed
237 * to write with bytes still queued -- the exact thing the wait was for.
238 *
239 * DCL has no signal delivery to make a wait *interruptible* (signal_pending()
240 * is 0, kernel.h), so the failure it can report is the one wait_event() above
241 * already reports: DCL_WAIT_EVENT_MS expired with the condition still false.
242 * That is spelled -ERESTARTSYS because that is the value this call site's
243 * `retval < 0` branch is written against and the value mainline returns when
244 * an interrupt does what a timeout does here. The polling shape -- one
245 * millisecond per pass, a bounded number of them, dcl_wait_event_timeout() on
246 * the way out -- is wait_event()'s, deliberately: two different wait loops in
247 * one header that disagreed about when they gave up would be worse than two
248 * that agreed and differed only in their result type.
249 */
250#define wait_event_interruptible(wq_head, condition) \
251 ({ \
252 _dcl_wait_core(wq_head, condition) \
253 ? 0 : -ERESTARTSYS; \
254 })

256/*
257 * wait_event_interruptible_timeout(wq, condition, timeout) --
258 * tty_ioctl.c:184 in tty_wait_until_sent(), and its result is read three
259 * different ways on the lines after:
260 *
261 * timeout = wait_event_interruptible_timeout(tty->write_wait,
262 * !tty_chars_in_buffer(tty), timeout);
263 * if (timeout <= 0) return; -- gave up
264 * if (timeout == MAX_SCHEDULE_TIMEOUT)
265 * timeout = 0; -- the caller passed "wait forever"
266 *
267 * Hence returning the *original* timeout on success rather than a remaining
268 * one. mainline returns jiffies left, and here that would matter only if a
269 * caller compared for equality -- which this one does, against
270 * MAX_SCHEDULE_TIMEOUT, where the true remaining time is so close to the
271 * original that mainline's own arithmetic lands on the same value. Returning
272 * the input preserves that equality for the infinite case, and for the finite
273 * case the caller only tests `<= 0`, so an optimistic remainder cannot be
274 * observed. 0 is the timeout, and -ERESTARTSYS never happens: there is no
275 * signal, so the two exits are "condition met" and "bounded wait expired".
276 *
277 * The `timeout` argument is in jiffies on mainline; DCL has no jiffy clock on
278 * this path and mdelay() is what wait_event() already spends, so the bounded
279 * wait is DCL_WAIT_EVENT_MS like every other one here, and timeout's value
280 * only ever travels through to the caller's own comparison.
281 */
282#define wait_event_interruptible_timeout(wq_head, condition, timeout) \
283 ({ \
284 long _dcl_t = (timeout); \
285 _dcl_wait_core(wq_head, condition) \
286 ? _dcl_t : 0; \
287 })

308/*
309 * prepare_to_wait() / finish_wait() -- the two calls that bracket a wait
310 * loop, at tty_port.c:542 and :569.
311 *
312 * The queue work is the part that has to be real, and it is: the entry goes
313 * on the queue only if it is not already there, which is what mainline does
314 * and what stops a loop that re-prepares each pass from queueing itself a
315 * second time -- and it comes off in finish_wait(), which is what stops a
316 * later wake_up_interruptible() from walking an entry whose stack frame has
317 * already been returned through. Both reuse the add/remove this header
318 * already declares (bodies in DCL/linux_irq_shim.c) rather than open-coding
319 * list operations, so there is one answer for where on the queue an entry
320 * goes instead of two that could disagree.
321 *
322 * The state work is the no-op set_current_state from <linux/sched.h>, and
323 * that is the same bargain wait_event() above strikes: there is no task state
324 * to change because no task blocks here. The queue is genuine so the wake-up
325 * is genuine; the *waiting* is the bounded poll the caller does after this
326 * returns. What would be wrong is an entry left queued and a caller that
327 * believed it had slept.
328 *
329 * mainline: kernel/sched/wait.c.
330 */
331static inline void prepare_to_wait(struct wait_queue_head* wq_head,
332 struct wait_queue_entry* wq_entry,
333 int state)
334{
335 if (list_empty(&wq_entry->entry))
336 add_wait_queue(wq_head, wq_entry);
337 set_current_state(state);
338}

348/*
349 * wait_event_freezable(wq_head, condition) -- the freezer-aware wait.
350 *
351 * virtio_console.c:752 and :784 both write it, inside wait_port_readable()
352 * and wait_port_writable(), where mainline's version additionally lets a
353 * suspend in progress abort the wait with -ERESTARTSYS.
354 *
355 * DCL has no refrigerator: nothing freezes a task, so the half of mainline's
356 * contract that responds to freezing has nothing to respond *to*. The other
357 * half -- poll the condition, give up after a bounded budget -- is what
358 * wait_event() above already does, and this is that loop and nothing else,
359 * deliberately: two waits in one header that gave up at different times would
360 * be a difference a caller could not see until a port stopped draining.
361 *
362 * The shape (one millisecond per pass, DCL_WAIT_EVENT_MS of them,
363 * dcl_wait_event_timeout() on the way out) is wait_event()'s, so a reader who
364 * knows one of them knows both.
365 *
366 * It returns an int, which is the part the previous note here got wrong when
367 * it said "returns nothing, because mainline's does not either" -- mainline's
368 * __wait_event_freezable() does return one, and both call sites treat the
369 * value as the whole contract:
370 *
371 * ret = wait_event_freezable(port->waitqueue, !will_read_block(port));
372 * if (ret < 0)
373 * return ret;
374 *
375 * A `do {} while (0)` there is not a value to assign, so the call would not
376 * compile at all, which is how the mistake surfaced. The two arms are the
377 * ones wait_event_interruptible() above already chose and for the same
378 * reason: 0 when the condition came true, -ERESTARTSYS when the bounded
379 * budget expired first. A 0 on the second arm would walk past the wait with
380 * the port still empty and hand read() a success with nothing in it; the
381 * callers' `ret < 0` branch is written for exactly the case that produces.
382 */
383#define wait_event_freezable(wq_head, condition) \
384 ({ \
385 _dcl_wait_core(wq_head, condition) \
386 ? 0 : -ERESTARTSYS; \
387 })
