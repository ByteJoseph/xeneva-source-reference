39/*
40 * __cacheline_group_begin/__end -- mainline's spelling, byte for byte:
41 *
42 * #define __cacheline_group_begin(GROUP) \
43 * __u8 __cacheline_group_begin__##GROUP[0]
44 *
45 * It is a zero-length array member, which is a GNU extension and a legal
46 * member declaration here, used to fence a run of fields inside a struct so a
47 * checker can assert they sit in their own cacheline. No section is emitted
48 * (mainline's alternative definition, used when the toolchain supports it,
49 * puts them in .data.cacheline_group.<GROUP>; DCL has no such checker).
50 *
51 * Two things a caller gets wrong on the first try, so they are written down:
52 *
53 * - the macros take NO trailing semicolon. The source writes
54 * `__dma_from_device_group_begin();` and that `;` is the member's.
55 * Emitting one inside the macro would leave a stray declaration.
56 * - GROUP may be empty. virtio-rng.c's struct virtrng_info calls both with
57 * `()`, so GROUP pastes to nothing and the member is plain
58 * `__cacheline_group_begin__[0]`. That is why this cannot be a variadic
59 * or no-argument macro: it has to accept exactly one, possibly-empty token.
60 */
61#ifndef __cacheline_group_begin
62#define __cacheline_group_begin(GROUP) __u8 __cacheline_group_begin__##GROUP[0]
63#endif
