162static inline double JS_VALUE_GET_FLOAT64(JSValue v)
163{
164 union {
165 JSValue v;
166 double d;
167 } u;
168 u.v = v;
169 u.v += (uint64_t)JS_FLOAT64_TAG_ADDEND << 32;
170 return u.d;

173#define JS_NAN (0x7ff8000000000000 - ((uint64_t)JS_FLOAT64_TAG_ADDEND << 32))

175static inline JSValue __JS_NewFloat64(JSContext *ctx, double d)
176{
177 union {
178 double d;
179 uint64_t u64;
180 } u;
181 JSValue v;
182 u.d = d;
183 /* normalize NaN */
184 if (js_unlikely((u.u64 & 0x7fffffffffffffff) > 0x7ff0000000000000))
185 v = JS_NAN;
186 else
187 v = u.u64 - ((uint64_t)JS_FLOAT64_TAG_ADDEND << 32);
188 return v;
189}

596{
597 int32_t val;

600 uint64_t u;
601 } u, t;

603 u.d = d;

610 }
