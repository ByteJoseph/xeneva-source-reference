117/*
118 * dev_err_probe(dev, err, fmt, ...) -- "report this probe failure, unless it
119 * is -EPROBE_DEFER, which is normal".
120 *
121 * The whole behaviour is that one exception: mainline suppresses the
122 * -EPROBE_DEFER message because a deferred probe is the driver asking to be
123 * called again, not a fault, and printing every one would bury real errors.
124 * The value comes back unchanged either way -- the caller at serial_core.c:3575
125 * does `return dev_err_probe(...)`, so what matters is that the *original*
126 * error reaches the caller, not what was printed.
127 *
128 * Statement expression because it has both jobs in one call: evaluate the
129 * error once, print conditionally, yield the error. `dev` is dropped by
130 * dev_err below exactly as it is for every other dev_* in this header.
131 */
132#define dev_err_probe(dev, err, fmt, ...) \
133 ({ \
134 int __e = (err); \

137 __e; \
