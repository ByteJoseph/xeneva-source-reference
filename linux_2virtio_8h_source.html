336/*
337 * module_virtio_driver(drv) -- register a static struct virtio_driver at
338 * boot. This is how virtio-rng.c ends: `module_virtio_driver(virtio_rng_driver);`
339 *
340 * mainline expands it through module_driver() into an init function *and* a
341 * matching exit function. DCL has the init half only, and the reason the exit
342 * half is dropped rather than stubbed is mechanical: <linux/module.h>'s
343 * module_exit() is an empty macro here (DCL has no unload path), so a
344 * `static void __exit drv##_exit(void)` would be a function nothing ever
345 * references -- -Wunused-function on every build of every driver that uses
346 * this macro. An unload path that cannot run is worse than no unload path.
347 *
348 * The init half is real, though, and it is worth being precise about *how*:
349 * module.h maps module_init() to DCL_INITCALL(), which stores the function in
350 * the named global __dcl_initcall_<fn>. Nothing discovers or schedules it --
351 * DclRunInitcalls() (DCL/linux_mm_shim.c) calls that global by name. So
352 * bringing up a driver through this macro has two halves: it compiles here,
353 * and its line goes into DclRunInitcalls() there. The second half is not
354 * automatic, and forgetting it produces exactly the symptom a prebuilt .ko
355 * used to produce: everything builds, and probe is never called.
356 *
357 * Returns register_virtio_driver()'s int, which is what DclRunInitcalls()
358 * expects from an initcall.
359 */
360#define module_virtio_driver(drv) \
361 static int __init drv##_init(void) \
362 { \
363 return register_virtio_driver(&(drv)); \
364 } \
365 module_init(drv##_init)

399/*
400 * virtio_cread(vdev, type, field, ptr) -- copy `sizeof(*ptr)` bytes of config
401 * space, starting at offsetof(type, field), into *ptr.
402 *
403 * The previous version of this macro cast the offset address to `type *` and
404 * dereferenced it:
405 *
406 * *(ptr) = *((type *)(base + offsetof(type, field)));
407 *
408 * which yields a `type` -- the whole struct the field belongs to -- and
409 * assigns it to *ptr. virtio_console.c:1794/:1795 are exactly that shape
410 * gone wrong: `virtio_cread(vdev, struct virtio_console_config, cols, &cols)`
411 * with cols a u16, so the assignment was u16 = struct virtio_console_config
412 * and clang said so three times. Taking the caller's width instead of the
413 * container's is what mainline's `sizeof(virtio_cread_v)` does
414 * (linux/virtio_config.h:466), so this is the same read with mainline's
415 * arithmetic and DCL's copy -- the difference being that DCL has no
416 * might_sleep() to make, because nothing here can block.
417 *
418 * The `char *` cast matters: device_config_base is `void *`, and while clang
419 * permits void-pointer arithmetic as an extension it does so one element at a
420 * time, which for a one-byte element happens to be correct -- by accident.
421 */
422#define virtio_cread(vdev, type, field, ptr) \
423 do { \
424 memcpy((ptr), \
425 (char*)(vdev)->config->device_config_base + \
426 offsetof(type, field), \
427 sizeof(*(ptr))); \
428 } while (0)

520 * ones above; the signatures are mainline's, character for character, because
521 * a mismatch here is invisible to the linker (C does not mangle) and would

533 * has a call site in this driver *today* -- the four
