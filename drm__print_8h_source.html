339static inline struct drm_printer
340drm_coredump_printer(struct drm_print_iterator *iter)
341{
342 struct drm_printer p = {
343 .printfn = __drm_printfn_coredump,
344 .puts = __drm_puts_coredump,
345 .arg = iter,
346 };
347
348 /* Set the internal offset of the iterator to zero */
349 iter->offset = 0;
350
351 return p;
352}

364static inline bool drm_coredump_printer_is_full(struct drm_printer *p)
365{
366 struct drm_print_iterator *iterator = p->arg;
367
368 if (p->printfn != __drm_printfn_coredump)
369 return true;
370
371 return !iterator->remain;
372}
