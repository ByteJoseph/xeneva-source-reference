179/*
180 * The below helper functions use 2 operators with 3 different
181 * calling conventions. The operators are related like:
182 *
183 * comp(a->key,b) < 0 := less(a,b)
184 * comp(a->key,b) > 0 := less(b,a)
185 * comp(a->key,b) == 0 := !less(a,b) && !less(b,a)
186 *
187 * If these operators define a partial order on the elements we make no
188 * guarantee on which of the elements matching the key is found. See
189 * rb_find().
190 *
191 * The reason for this is to allow the find() interface without requiring an
192 * on-stack dummy object, which might not be feasible due to object size.
193 */
194
203static __always_inline struct rb_node *
204rb_add_cached(struct rb_node *node, struct rb_root_cached *tree,
205 bool (*less)(struct rb_node *, const struct rb_node *))
206{
207 struct rb_node **link = &tree->rb_root.rb_node;
208 struct rb_node *parent = NULL;
209 bool leftmost = true;
210
211 while (*link) {
212 parent = *link;
213 if (less(node, parent)) {
214 link = &parent->rb_left;
215 } else {
216 link = &parent->rb_right;
217 leftmost = false;
218 }
219 }
220
221 rb_link_node(node, parent, link);
222 rb_insert_color_cached(node, tree, leftmost);
223
224 return leftmost ? node : NULL;
225}

251static inline void
252rb_link_linked_node(struct rb_node *node, struct rb_node *parent, struct rb_node **link)
253{
254 if (!parent)
255 return;
256
257 struct rb_node_linked *nnew = __node_2_linked_node(node);
258 struct rb_node_linked *npar = __node_2_linked_node(parent);
259
260 if (link == &parent->rb_left) {
261 nnew->prev = npar->prev;
262 nnew->next = npar;
263 npar->prev = nnew;
264 if (nnew->prev)
265 nnew->prev->next = nnew;
266 } else {
267 nnew->next = npar->next;
268 nnew->prev = npar;
269 npar->next = nnew;
270 if (nnew->next)
271 nnew->next->prev = nnew;
272 }
273}
