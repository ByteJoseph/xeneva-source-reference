4/*
5 * DCL <linux/sched.h> -- task state, `current`, and schedule().
6 *
7 * Xeneva has no sleeping scheduler for kernel context: there is nothing to
8 * switch away to, so `schedule()` cannot do what mainline's does. Pretending
9 * otherwise would be the worst kind of wrongness here -- a driver that
10 * believes it slept would loop forever on a condition that never arrives.
11 * Instead the honest shape is kept and the sleeps are made *short and
12 * bounded*, which is what the tty code actually needs from them:
13 *
14 * schedule()
15 * Sleeps a millisecond. tty_open() retries on -EAGAIN with
16 * `schedule(); goto retry_open;` (tty_io.c:2127 and :2153) -- with a
17 * no-op that becomes a hard spin against a condition only an
18 * interrupt can change, so a millisecond of yield is the difference
19 * between a retry loop and a locked-up console.
20 *
21 * signal_pending() / fatal_signal_pending()
22 * Always 0. Xeneva has no signals to deliver, so no wait can be broken
23 * out of this way. That is recorded rather than papered over, because
24 * it has one real consequence: uart_wait_modem_status() (serial_core.c:
25 * 1200) loops until a modem-status interrupt or a signal, so TIOCMWAIT
26 * on a port that never raises one never returns. Closing that needs a
27 * bounded wait at the serial_core call site -- stage 4's list, not a
28 * lie in this header.
29 *
30 * current
31 * One task_struct per kernel context, with `comm` filled from the
32 * running thread. `current` is an object-like macro, exactly as on
33 * mainline, which means a struct member literally named `current`
34 * would not compile against this header. Nothing in the tty/serial
35 * set has one (the only `file->current` in the tree is Xeneva's own
36 * loader and FAT code, which do not include DCL headers).
37 *
38 * `signal->tty` is NULL: it is read by tty_io.c:2280 in tiocsti(), the
39 * TIOCSTI gate, and only there. NULL makes TIOCSTI answer -EPERM for a
40 * non-admin, which is the safe direction -- TIOCSTI injects keystrokes
41 * into another terminal and is disabled by default upstream for
42 * exactly that reason.
43 *
44 * task_lock()/task_unlock() are the one guard tty_io.c takes directly
45 * (guard(task_lock)(p) at :3046), so the class lives here with the primitive.
46 */

131 * for its `comm` (two deprecation warnings) and its `signal->tty` (one

138 * #undef first: <linux/mm.h> carries a fallback `current ((void*)0)` for the
139 * headers that need the name without a scheduler. Which of the two lands
140 * second depends on the include order of the translation unit, and an
141 * unconditional redefinition is a -Wmacro-redefined warning at best and, at
142 * worst, `((void*)0)->comm` if mm.h wins. Redefining unconditionally is the
143 * right behaviour -- dcl_current() is the true one and mm.h's is only ever a
144 * fallback -- so drop whatever is there and put this one in, in both orders.

149/*
150 * task_pid_vnr() -- drm_ioctl.c:199's drm_getclient() stores it into
151 * struct drm_client and hands that back to userspace.
152 *
153 * mainline answers "the task's pid as current's namespace sees it"; DCL has
154 * no pid table, and <linux/pid.h> records what follows from that for the
155 * accessors it owns (task_tgid() is NULL, task_pid_nr() is 0 for a debug line
156 * whose text is compiled away). A value userspace reads back is the other
157 * case, and the task is not anonymous: dcl_current() refreshes ->pid from the
158 * running Xeneva thread on every call (DCL/current_ident.c), so that real
159 * thread id is what this reports. The NULL arm cannot come from any caller
160 * (they all pass `current`); 0 is the one answer that touches nothing.
161 *
162 * A static inline in the shape of pid.h's accessors rather than a shim
163 * symbol, for the same reason those are: the body is one field read, and a
164 * link symbol for it would be a call to a function a header could have
165 * compiled away.
166 */
167static inline int task_pid_vnr(struct task_struct* tsk)
168{
169 return tsk ? tsk->pid : 0;
170}

172/*
173 * get_task_comm(buf, tsk) -- mainline's accessor for a task's comm, which
174 * mainline also declares in <linux/sched.h> (there as a macro over
175 * __get_task_comm()). One caller, the virtio-gpu debug-name ioctl, and the
176 * copy is bounded by the struct's own `comm` array so the terminator
177 * always fits. `comm` is exactly what dcl_current() refreshes from the
178 * running thread on every call, so a live task's name is real text; the
179 * shape is this header's static-inline style rather than a shim symbol,
180 * for the task_pid_vnr() reason above.
181 */
182static inline char* get_task_comm(char* buf, struct task_struct* tsk)
183{
184 unsigned int i;
185
186 for (i = 0; i + 1 < sizeof(tsk->comm) && tsk->comm[i]; i++)
187 buf[i] = tsk->comm[i];
188 buf[i] = 0;
189 return buf;
190}

194/*
195 * wake_up_process(tsk) -- tty_ldsem.c:97/:124/:245, where a waiter that has
196 * parked on the semaphore is told its turn came.
197 *
198 * mainline takes the task off the runqueue's sleep state and returns whether
199 * it found one. Here it returns 1 and does nothing else, and the reason it
200 * is safe for the semaphore to work at all is what schedule() is on this
201 * build: DCL/linux_irq_shim.c:350 parks in mdelay(1) and returns, so no task
202 * is ever *asleep* waiting for this -- the waiter in tty_ldsem.c's
203 * down_read_failed() loop re-evaluates its condition every millisecond
204 * whether or not anyone wakes it, and `wake_up_process` on a task that is
205 * already runnable is a no-op upstream too. A real wake-up would need a
206 * scheduler with a sleep state to wake out of.
207 *
208 * Returning 1 rather than void is mainline's signature: the callers discard
209 * it, but a decl that answered a different type would be a different symbol
210 * to a reader comparing this against tty_ldsem.c.
211 */
212static inline int wake_up_process(struct task_struct* tsk)
213{
214 (void)tsk;
215 return 1;
216}
