4/*
5 * DCL <linux/sched.h> -- task state, `current`, and schedule().
6 *
7 * Xeneva has no sleeping scheduler for kernel context: there is nothing to
8 * switch away to, so `schedule()` cannot do what mainline's does. Pretending
9 * otherwise would be the worst kind of wrongness here -- a driver that
10 * believes it slept would loop forever on a condition that never arrives.
11 * Instead the honest shape is kept and the sleeps are made *short and
12 * bounded*, which is what the tty code actually needs from them:
13 *
14 * schedule()
15 * Sleeps a millisecond. tty_open() retries on -EAGAIN with
16 * `schedule(); goto retry_open;` (tty_io.c:2127 and :2153) -- with a
17 * no-op that becomes a hard spin against a condition only an
18 * interrupt can change, so a millisecond of yield is the difference
19 * between a retry loop and a locked-up console.
20 *
21 * signal_pending() / fatal_signal_pending()
22 * Always 0. Xeneva has no signals to deliver, so no wait can be broken
23 * out of this way. That is recorded rather than papered over, because
24 * it has one real consequence: uart_wait_modem_status() (serial_core.c:
25 * 1200) loops until a modem-status interrupt or a signal, so TIOCMWAIT
26 * on a port that never raises one never returns. Closing that needs a
27 * bounded wait at the serial_core call site -- stage 4's list, not a
28 * lie in this header.
29 *
30 * current
31 * One task_struct per kernel context, with `comm` filled from the
32 * running thread. `current` is an object-like macro, exactly as on
33 * mainline, which means a struct member literally named `current`
34 * would not compile against this header. Nothing in the tty/serial
35 * set has one (the only `file->current` in the tree is Xeneva's own
36 * loader and FAT code, which do not include DCL headers).
37 *
38 * `signal->tty` is NULL: it is read by tty_io.c:2280 in tiocsti(), the
39 * TIOCSTI gate, and only there. NULL makes TIOCSTI answer -EPERM for a
40 * non-admin, which is the safe direction -- TIOCSTI injects keystrokes
41 * into another terminal and is disabled by default upstream for
42 * exactly that reason.
43 *
44 * task_lock()/task_unlock() are the one guard tty_io.c takes directly
45 * (guard(task_lock)(p) at :3046), so the class lives here with the primitive.
46 */

75/*
76 * The one DCL task. It is shared rather than per-thread: `current` is read
77 * for its `comm` (two deprecation warnings) and its `signal->tty` (one
78 * permission gate), never written by any of the nine files, so a single
79 * context whose `comm` is refreshed on each read answers truthfully without
80 * a task table that nothing else needs.

83/*
84 * #undef first: <linux/mm.h> carries a fallback `current ((void*)0)` for the
85 * headers that need the name without a scheduler. Which of the two lands
86 * second depends on the include order of the translation unit, and an
87 * unconditional redefinition is a -Wmacro-redefined warning at best and, at
88 * worst, `((void*)0)->comm` if mm.h wins. Redefining unconditionally is the
89 * right behaviour -- dcl_current() is the true one and mm.h's is only ever a
90 * fallback -- so drop whatever is there and put this one in, in both orders.
91 */

97/*
98 * wake_up_process(tsk) -- tty_ldsem.c:97/:124/:245, where a waiter that has
99 * parked on the semaphore is told its turn came.
100 *
101 * mainline takes the task off the runqueue's sleep state and returns whether
102 * it found one. Here it returns 1 and does nothing else, and the reason it
103 * is safe for the semaphore to work at all is what schedule() is on this
104 * build: DCL/linux_irq_shim.c:350 parks in mdelay(1) and returns, so no task
105 * is ever *asleep* waiting for this -- the waiter in tty_ldsem.c's
106 * down_read_failed() loop re-evaluates its condition every millisecond
107 * whether or not anyone wakes it, and `wake_up_process` on a task that is
108 * already runnable is a no-op upstream too. A real wake-up would need a
109 * scheduler with a sleep state to wake out of.
110 *
111 * Returning 1 rather than void is mainline's signature: the callers discard
112 * it, but a decl that answered a different type would be a different symbol
113 * to a reader comparing this against tty_ldsem.c.
114 */
115static inline int wake_up_process(struct task_struct* tsk)
116{
117 (void)tsk;
118 return 1;
119}

153 *
