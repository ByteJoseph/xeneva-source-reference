4/*
5 * DCL <linux/atomic.h> -- refcounted counters for the tty layer.
6 *
7 * DCL had no atomic API at all: nothing in Xeneva needed one, because its own
8 * code serialises at the process level. The tty core does need it, and for a
9 * reason that is worth stating -- tty_buffer.c uses `atomic_t` as a *turn
10 * counter*, not as a lock: `buf->priority` is incremented by
11 * tty_buffer_lock_exclusive() so flush_to_ldisc() can see that the line
12 * discipline has taken the buffer over and back off mid-loop (tty_buffer.c:482).
13 * That read has to be exact or a byte gets delivered twice or dropped.
14 *
15 * The operations below are real, on clang's __atomic builtins, so aarch64
16 * emits LDXR/STXR rather than a plain load/store. That is not belt-and-braces:
17 * tty_flip_buffer_push() runs from the UART interrupt handler while
18 * flush_to_ldisc() runs from whoever called queue_work() (see
19 * <linux/workqueue.h> -- DCL runs work inline), so these two genuinely do
20 * race on one core. Making them plain stores would have been the classic
21 * "works until it doesn't" bug.
22 *
23 * Signatures follow mainline: atomic_read() takes a const pointer, atomic_add()
24 * takes the value first, *_return() hands back the post-operation value, and
25 * atomic_add_unless() is the "add unless it already equals" form
26 * serial_core.c uses for its port refcount.
27 */
28

218/*
219 * atomic_long_try_cmpxchg(v, old, new) -- the conditional form, and the one
220 * tty_ldsem.c actually uses (five sites: :90, :114, :170, :245, :351), every
221 * one of them `if (atomic_long_try_cmpxchg(&sem->count, &count, ...))`
222 * inside a retry loop.
223 *
224 * Difference from the function above, and the reason both exist: the expected
225 * value goes in by pointer and is *rewritten* with what the word actually
226 * held when the exchange fails. The loops depend on that -- they compute the
227 * next value from `count` on each pass, so a stale count would re-apply the
228 * same adjustment forever. Returning the old value instead (as
229 * atomic_long_cmpxchg does) would leave `count` untouched and turn every
230 * retry into a repeat.
231 *
232 * __atomic_compare_exchange_n's second argument is exactly this contract, so
233 * the body is one call with `weak = 0`: the loops have their own retry
234 * condition and do not need a spurious failure, and seq_cst on both halves
235 * matches the rest of this header.
236 */
237static inline bool atomic_long_try_cmpxchg(atomic_long_t* v, long* old,
238 long new_v)
239{
240 return __atomic_compare_exchange_n(&v->counter, old, new_v, 0,
241 __ATOMIC_SEQ_CST, __ATOMIC_SEQ_CST);
242}
