7/*
8 * DCL delta -- `uintptr_t`, not mainline's `unsigned long`.
9 *
10 * This field is not a number: it is the parent pointer with the color bit
11 * folded into its bottom two bits, read back by rb_parent() and __rb_color().
12 * Its width therefore has to be pointer width. mainline gets that from
13 * `unsigned long` being 64-bit on arm64; this target is
14 * aarch64-unknown-windows, which is LLP64 (verified: sizeof(long) == 4), so
15 * `unsigned long` keeps the parent address's low 32 bits and every node
16 * above the 4 GB mark would walk to a parent in the low 4 GB that nothing
17 * maps -- the same narrowing serial_core.c's two address lines have, arriving
18 * through a struct field this time. The writers
19 * that fill the field are in rbtree.h and rbtree_augmented.h.
20 */
21struct rb_node {
22 uintptr_t __rb_parent_color;
23 struct rb_node *rb_right;
24 struct rb_node *rb_left;
25} __attribute__((aligned(sizeof(long))));
26/* The alignment might seem pointless, but allegedly CRIS needs it */

38/*
39 * Leftmost-cached rbtrees.
40 *
41 * We do not cache the rightmost node based on footprint
42 * size vs number of potential users that could benefit
43 * from O(1) rb_last(). Just not worth it, users that want
44 * this feature can always implement the logic explicitly.
45 * Furthermore, users that want to cache both pointers may
46 * find it a bit asymmetric, but that's ok.
47 */
48struct rb_root_cached {
49 struct rb_root rb_root;
50 struct rb_node *rb_leftmost;
51};
52
53/*
54 * Leftmost tree with links. This would allow a trivial rb_rightmost update,
55 * but that has been omitted due to the lack of users.
56 */
57struct rb_root_linked {
58 struct rb_root rb_root;
59 struct rb_node_linked *rb_leftmost;
60};
