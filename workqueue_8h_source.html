114/*
115 * Work runs inline: there is no kworker to run it later.
116 *
117 * These are functions, not statement-expressions, because mainline's callers
118 * use their return values -- tty_buffer.c:66 is literally
119 * `return queue_work(flip_wq ?: system_dfl_wq, &buf->work);`, which does not
120 * parse against a do{}while(0) macro. The bool is "work was queued"; since
121 * it runs to completion before we return, true is always the right answer.
122 *
123 * Inline execution is a real deviation from mainline and it is worth being
124 * precise about why it is safe here: tty_flip_buffer_push() calls this from
125 * interrupt context, so flush_to_ldisc() runs before tty_flip_buffer_push()
126 * returns rather than on a worker later. Its only lock is the buffer mutex,
127 * which in DCL is a no-op (<linux/mutex.h> says so plainly), so there is
128 * nothing to deadlock on. This is also what Linux itself did before 2.6.24,
129 * when tty_flip_buffer_push() called the flush directly. The behaviour that
130 * changes is latency, not correctness: a byte arrives at n_tty before the
131 * interrupt handler returns instead of after.
132 */
133static inline bool schedule_work(struct work_struct* work)
134{
135 __dcl_run_work(work);
136 return true;
137}

258/*
259 * mod_delayed_work() -- mainline re-arms work that may already be pending and
260 * reports whether it found it that way. There is nothing to re-arm: no timer
261 * wheel to move (see schedule_delayed_work above), so "run this, possibly
262 * later" and "run this now" are the only two states this tree has, and both
263 * are the second one. It forwards to queue_delayed_work() rather than
264 * duplicating the inline run, so the two spellings cannot drift apart.
265 *
266 * The true return is mainline's "yes, it was already pending". Both callers in
267 * the tree discard it, and it is the answer that matches what has just
268 * happened -- the work has run, which is the state a caller checking for true
269 * would go on to expect. Returning false would mean "nothing was queued", for
270 * a call that did queue something.
271 */
272static inline bool mod_delayed_work(struct workqueue_struct* wq,
273 struct delayed_work* dwork,
274 unsigned long delay) {
275 return queue_delayed_work(wq, dwork, delay);
276}
