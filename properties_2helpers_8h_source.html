63static inline css_error css__copy_lwc_string_array(
64 bool ref,
65 lwc_string *const*orig,
66 lwc_string ***copy_out)
67{
68 size_t count = 0;
69 lwc_string **copy = NULL;
70
71 if (orig != NULL) {
72 for (lwc_string *const*i = orig; (*i) != NULL; i++) {
73 count++;
74 }
75
76 copy = malloc((count + 1) * sizeof(*copy));
77 if (copy == NULL) {
78 return CSS_NOMEM;
79 }
80
81 if (ref) {
82 for (size_t i = 0; i < count; i++) {
83 copy[i] = lwc_string_ref(orig[i]);
84 }
85 copy[count] = NULL;
86 } else {
87 memcpy(copy, orig, (count + 1) * sizeof(*copy));
88 }
89 }
90
91 *copy_out = copy;
92 return CSS_OK;
93}

132static inline css_error css__copy_computed_content_item_array(
133 bool ref,
134 const css_computed_content_item *orig,
135 css_computed_content_item **copy_out)
136{
137 size_t count = 0;
138 css_computed_content_item *copy = NULL;
139
140 if (orig != NULL) {
141 for (const css_computed_content_item *i = orig;
142 i->type != CSS_COMPUTED_CONTENT_NONE; i++) {
143 count++;
144 }
145
146 copy = malloc((count + 1) * sizeof(*copy));
147 if (copy == NULL) {
148 return CSS_NOMEM;
149 }
150
151 if (ref) {
152 for (size_t i = 0; i < count; i++) {
153 switch (orig[i].type) {
154 case CSS_COMPUTED_CONTENT_STRING:
155 copy[i].data.string = lwc_string_ref(
156 orig[i].data.string);
157 break;
158 case CSS_COMPUTED_CONTENT_URI:
159 copy[i].data.uri = lwc_string_ref(
160 orig[i].data.uri);
161 break;
162 case CSS_COMPUTED_CONTENT_ATTR:
163 copy[i].data.attr = lwc_string_ref(
164 orig[i].data.attr);
165 break;
166 case CSS_COMPUTED_CONTENT_COUNTER:
167 copy[i].data.counter.name = lwc_string_ref(
168 orig[i].data.counter.name);
169 copy[i].data.counter.style =
170 orig[i].data.counter.style;
171 break;
172 case CSS_COMPUTED_CONTENT_COUNTERS:
173 copy[i].data.counters.name = lwc_string_ref(
174 orig[i].data.counters.name);
175 copy[i].data.counters.sep = lwc_string_ref(
176 orig[i].data.counters.sep);
177 copy[i].data.counters.style =
178 orig[i].data.counters.style;
179 break;
180 default:
181 break;
182 }
183 copy[i].type = orig[i].type;
184 }
185 copy[count].type = CSS_COMPUTED_CONTENT_NONE;
186 } else {
187 memcpy(copy, orig, (count + 1) * sizeof(*copy));
188 }
189 }
190
191 *copy_out = copy;
192 return CSS_OK;
193}
