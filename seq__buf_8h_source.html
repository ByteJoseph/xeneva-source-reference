74/*
75 * How much buffer is left on the seq_buf?

77static inline unsigned int
78seq_buf_buffer_left(struct seq_buf *s)
79{
80 if (seq_buf_has_overflowed(s))

108static inline const char *seq_buf_str(struct seq_buf *s)
109{
110 if (WARN_ON(s->size == 0))
111 return "";
112
113 if (seq_buf_buffer_left(s))
114 s->buffer[s->len] = 0;
115 else
116 s->buffer[s->size - 1] = 0;
117
118 return s->buffer;
119}

129static inline size_t seq_buf_get_buf(struct seq_buf *s, char **bufp)
130{
131 WARN_ON(s->len > s->size + 1);
132
133 if (s->len < s->size) {
134 *bufp = s->buffer + s->len;
135 return s->size - s->len;
136 }
137
138 *bufp = NULL;
139 return 0;
140}

151static inline void seq_buf_commit(struct seq_buf *s, int num)
152{
153 if (num < 0) {
154 seq_buf_set_overflow(s);
155 } else {
156 /* num must be negative on overflow */
157 BUG_ON(s->len + num > s->size);
158 s->len += num;
159 }
160}

170static inline int seq_buf_pop(struct seq_buf *s)
171{
172 if (!s->len)
173 return -1;
174
175 s->len--;
176 return (unsigned int)s->buffer[s->len];
177}
