5/*
6 * DCL <linux/virtio_config.h> -- a router, and four helpers mainline puts
7 * here that <linux/virtio.h> does not.
8 *
9 * ### Why this is not mainline's 699-line header
10 *
11 * mainline's <linux/virtio_config.h> carries its own copies of the three
12 * types DCL's <linux/virtio.h> already defines: `struct virtqueue_info`
13 * (:29), `struct virtio_config_ops` (:112) and, through the helpers, an
14 * opinion about `struct virtio_device` -- it reads `vdev->features_array`
15 * (:228, :239, :250) and declares `virtio_find_vqs()` with an
16 * `struct irq_affinity *` parameter (:293) that disagrees with the one in
17 * virtio.h. Two definitions of one struct in one translation unit is a hard
18 * redefinition error, and DCL's version is the one that is authoritative:
19 * it is the layout the native AuVirtIO layer fills in and the layout the
20 * module loader indexes, and `features_array` is not a member of it and
21 * cannot be added without deciding where in `_linux_prefix[0x328]` the
22 * feature bitmap mainline keeps there is supposed to live.
23 *
24 * So the header includes <linux/virtio.h> and stops. Everything mainline
25 * defines below the type declarations that DCL does not need is not needed
26 * here either: virtio_cread8/16/32/64, virtio_get_status/set_status,
27 * virtio_device_ready, virtio_find_vqs, the endian wrappers -- all of those
28 * are already in virtio.h, which is the file this one routes to.
29 *
30 * ### The four helpers that are actually missing
31 *
32 * __virtio_test_bit / virtio_has_feature / virtio_cread_feature are what a
33 * mainline driver tests a feature bit with; virtio_device_for_each_vq is the
34 * list walk. They cannot come from virtio.h because mainline does not put
35 * them there, and a ported source is written against mainline's *layout of
36 * the headers*, not against where DCL happens to prefer things.
37 */
38

43/*
44 * __virtio_test_bit(vdev, fbit) -- is feature `fbit` agreed with this device?
45 *
46 * mainline reads `vdev->features_array[fbit]`: a bitmap of the features that
47 * survived negotiation, filled once by virtio_finalise_features(). DCL has no
48 * negotiation to have survived -- see shim_finalize_features() in
49 * DCL/virtio_shim.c, which writes the device's whole feature set back
50 * unfiltered and sets FEATURES_OK (it reports failure only if the device
51 * itself rejects). The guest accepts everything the device offers, so
52 * "offered" and "agreed" name the same bits, and the bitmap mainline caches
53 * is here a query instead:
54 *
55 * vdev->config->get_features(vdev)
56 *
57 * which is shim_get_features() -- two reads of the device's DevFeature
58 * registers, select 0 then select 1, reassembled lo||hi. Two MMIO-ish reads
59 * per feature test, where mainline does one load from memory, on paths that
60 * run a handful of times at probe and once per config change: not worth a
61 * cache, and a cache would have to be invalidated by something that does not
62 * exist yet.
63 *
64 * The NULL guard is not defensive padding. use_multiport() calls this at
65 * virtio_console.c:343 during probe, and probe runs for a device whose config
66 * ops were installed by virtio_device_register() -- but a ported source that
67 * builds its own virtio_device would not have gone through that path, and a
68 * NULL call there faults with no register to point at. FALSE is the answer
69 * that keeps the caller on the single-port path, which is the safe one.
70 */
71static inline bool __virtio_test_bit(const struct virtio_device* vdev,
72 unsigned int fbit)
73{
74 u64 features;
75
76 if (!vdev || !vdev->config || !vdev->config->get_features)
77 return false;
78
79 features = vdev->config->get_features((struct virtio_device*)vdev);
80 return (features >> fbit) & 1ULL;
81}

83/*
84 * virtio_has_feature() -- the spelling drivers reach for. mainline wraps
85 * __virtio_test_bit() with a debug check that the feature is one the driver
86 * actually offered (typecheck() against the driver's feature_table, via
87 * virtio_check_driver_offered_feature()); that check is a CONFIG that DCL
88 * does not set, and reproducing its *expression* would mean reproducing
89 * typecheck() for a branch that is compiled out. The answer is the same.
90 */
91static inline bool virtio_has_feature(const struct virtio_device* vdev,
92 unsigned int fbit)
93{
94 return __virtio_test_bit(vdev, fbit);
95}

97/*
98 * virtio_cread_feature(vdev, fbit, structname, member, ptr) -- read a config
99 * field, but only if feature `fbit` is agreed; otherwise report -ENOENT.
100 *
101 * mainline's is a statement expression (linux/virtio_config.h:678) with the
102 * same three-part shape, and this is that expression verbatim. The caller at
103 * virtio_console.c:1999 compares the result against 0, so the two arms have
104 * to be 0 and a *negative* errno -- 0 on the read, -ENOENT when the feature
105 * is not there. virtio_cread() comes from <linux/virtio.h>.
106 *
107 * `-ENOENT` needs <linux/err.h>'s errno, which reaches <linux/kernel.h>,
108 * which includes <linux/virtio.h> before this file: the chain is already
109 * closed by the time this macro is expanded, and a macro expands where it is
110 * used, not where it is written.
111 */
112#define virtio_cread_feature(vdev, fbit, structname, member, ptr) \
113 ({ \
114 int _r = 0; \
115 if (!virtio_has_feature(vdev, fbit)) \
116 _r = -ENOENT; \
117 else \
118 virtio_cread((vdev), structname, member, ptr); \
119 _r; \
120 })

122/*
123 * virtio_device_for_each_vq(vdev, vq) -- walk every queue on a device.
124 *
125 * mainline's is one line (linux/virtio.h:221):
126 *
127 * list_for_each_entry(vq, &(vdev)->vqs, list)
128 *
129 * and this is the same line. It needs `vdev->vqs` (a list head, appended to
130 * DCL's struct virtio_device) and `vq->list` (appended to struct virtqueue);
131 * both were added for this macro, because the only caller -- remove_vqs(),
132 * virtio_console.c:1895 -- is the loop that flushes and frees every in-flight
133 * receive buffer, and a macro that quietly walked nothing would leave that
134 * loop silently empty rather than obviously wrong.
135 *
136 * The head is initialised by virtio_device_register() and the queues are
137 * linked in by shim_find_vqs(); _virtio_devices[]'s snapshot copy gets its
138 * own empty head so that walking *it* terminates instead of treating the
139 * original's head as if it were a struct virtqueue.
140 */
141#define virtio_device_for_each_vq(vdev, vq) \
142 list_for_each_entry(vq, &(vdev)->vqs, list)

144/*
145 * virtio_is_little_endian(vdev) -- byte order of this device's config fields
146 * and vring entries.
147 *
148 * mainline (linux/virtio_config.h:399) forwards to
149 * virtio_legacy_is_little_endian(), which is in <linux/virtio_byteorder.h>
150 * -- the header included above, and DCL's copy answers it with
151 * `#ifdef __LITTLE_ENDIAN` (line 9), which is true because types.h defines
152 * __LITTLE_ENDIAN as 1234 and clang predefines no such macro for
153 * aarch64-unknown-windows. The `vdev` parameter is accepted for mainline's
154 * signature and ignored: the legacy/modern split it exists to query has no
155 * second arm in a tree with one byte order.
156 */
157static inline bool virtio_is_little_endian(struct virtio_device* vdev)
158{
159 (void)vdev;
160 return virtio_legacy_is_little_endian();
161}

163/*
164 * The four converters a ported source names, in mainline's order and with
165 * mainline's parameter types -- `struct virtio_device *`, non-const, because
166 * mainline's is and a const mismatch would be a diagnostic on a call site
167 * that has nothing to do with endianness.
168 *
169 * virtio_console.c:553/:554 use cpu_to_virtio32/16 to write a control
170 * message into the out-queue; :1534/:1536/:1543 read the device's own
171 * control word back. They route through virtio_is_little_endian() above, so
172 * all four are the identity on this target -- which is the correct answer,
173 * not a shortcut: QEMU's virtio-serial and the aarch64 ports are both
174 * little-endian, and the bswap that would not happen here is the one that
175 * must not.
176 */
177static inline u16 virtio16_to_cpu(struct virtio_device* vdev, __virtio16 val)
178{
179 return __virtio16_to_cpu(virtio_is_little_endian(vdev), val);
180}
