92/* do y <<= n, clipping to range [-2^30, 2^30 - 1] (i.e. output has one guard bit) */
93/*
94#define CLIP_2N_SHIFT(y, n) { \
95 int sign = (y) >> 31; \
96 if (sign != (y) >> (30 - (n))) { \
97 (y) = sign ^ (0x3fffffff); \
98 } else { \
99 (y) = (y) << (n); \
100 } \
101 }
102*/
103/* clip to [-2^n, 2^n-1], valid range of n = [1, 30] */
104/*
105#define CLIP_2N(val, n) { \
106 if ((val) >> 31 != (val) >> (n)) \
107 (val) = ((val) >> 31) ^ ((1 << (n)) - 1); \
108 }
109*/

203typedef struct _ADTSHeader {
204 /* fixed */
205 unsigned char id; /* MPEG bit - should be 1 */
206 unsigned char layer; /* MPEG layer - should be 0 */
207 unsigned char protectBit; /* 0 = CRC word follows, 1 = no CRC word */
208 unsigned char profile; /* 0 = main, 1 = LC, 2 = SSR, 3 = reserved */
209 unsigned char sampRateIdx; /* sample rate index range = [0, 11] */
210 unsigned char privateBit; /* ignore */
211 unsigned char channelConfig; /* 0 = implicit, >0 = use default table */
212 unsigned char origCopy; /* 0 = copy, 1 = original */
213 unsigned char home; /* ignore */
214
215 /* variable */
216 unsigned char copyBit; /* 1 bit of the 72-bit copyright ID (transmitted as 1 bit per frame) */
217 unsigned char copyStart; /* 1 = this bit starts the 72-bit ID, 0 = it does not */
218 int frameLength; /* length of frame */
219 int bufferFull; /* number of 32-bit words left in enc buffer, 0x7FF = VBR */
220 unsigned char numRawDataBlocks; /* number of raw data blocks in frame */
221
222 /* CRC */
223 int crcCheckWord; /* 16-bit CRC check word (present if protectBit == 0) */
224} ADTSHeader;

246 unsigned char numADE; /* number of associated data elements (max = 7) */
247 unsigned char numCCE; /* number of valid channel coupling elements (max = 15) */
248 unsigned char monoMixdown; /* mono mixdown: bit 4 = present flag, bits 3-0 = element number */
249 unsigned char stereoMixdown; /* stereo mixdown: bit 4 = present flag, bits 3-0 = element number */
250 unsigned char matrixMixdown; /* matrix mixdown: bit 4 = present flag, bit 3 = unused,
251 bits 2-1 = index, bit 0 = pseudo-surround enable */
252 unsigned char fce[MAX_NUM_FCE]; /* front element channel pair: bit 4 = SCE/CPE flag, bits 3-0 = inst tag */
253 unsigned char sce[MAX_NUM_SCE]; /* side element channel pair: bit 4 = SCE/CPE flag, bits 3-0 = inst tag */
254 unsigned char bce[MAX_NUM_BCE]; /* back element channel pair: bit 4 = SCE/CPE flag, bits 3-0 = inst tag */
255 unsigned char lce[MAX_NUM_LCE]; /* instance tag for LFE elements */
256 unsigned char ade[MAX_NUM_ADE]; /* instance tag for ADE elements */
257 unsigned char cce[MAX_NUM_BCE]; /* channel coupling elements: bit 4 = switching flag, bits 3-0 = inst tag */
