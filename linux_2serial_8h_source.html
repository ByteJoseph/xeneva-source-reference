264/*
265 * Serial ioctl request numbers.
266 *
267 * mainline keeps these in include/uapi/asm-generic/ioctls.h, mixed in with
268 * every other request number on the architecture; DCL's tree has no
269 * asm-generic layer and no other home for them, so they sit here next to the
270 * rest of the serial vocabulary they operate on (struct serial_struct,
271 * struct serial_iso7816 -- both defined above, which is what makes the two
272 * _IOWR/_IOR entries below computable at all).
273 *
274 * Every value is an ABI constant: userspace passes these straight through
275 * tty_io.c to uart_ioctl(), and 8250_port.c's ioctl() dispatch switches on
276 * them. Copied verbatim from asm-generic/ioctls.h (mainline v7.2) -- the
277 * numbers are the interesting part, not the spelling.
278 *
279 * Note the encoding split these reveal: all but two are bare integers, while
280 * TIOCGISO7816/TIOCSISO7816 are built with _IOR/_IOWR from <linux/ioctl.h>.
281 * See that header for why two encodings coexist in the tree right now.
282 */
283#define TIOCGSERIAL 0x541E
284#define TIOCSSERIAL 0x541F
285#define TIOCGRS485 0x542E
286#define TIOCSRS485 0x542F
287#define TIOCSERCONFIG 0x5453
288#define TIOCSERGETLSR 0x5459 /* Get line status register */
289#define TIOCSERGETMULTI 0x545A /* Get multiport config */
290#define TIOCSERSETMULTI 0x545B /* Set multiport config */
291#define TIOCMIWAIT 0x545C /* wait for a change on serial input line(s) */
292#define TIOCGICOUNT 0x545D /* read serial port inline interrupt counts */
293#define TIOCSER_TEMT 0x01 /* Transmitter physically empty */
294#define TIOCGISO7816 _IOR('T', 0x42, struct serial_iso7816)
295#define TIOCSISO7816 _IOWR('T', 0x43, struct serial_iso7816)
