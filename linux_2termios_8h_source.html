4/*
5 * DCL <linux/termios.h> -- the termios structures and the flag vocabulary
6 * every tty file speaks.
7 *
8 * The structures are mainline's (uapi/asm-generic/termbits.h, which is what
9 * arm64 includes): struct termios/termios2/ktermios and struct winsize, with
10 * the full c_cc index list and all four flag sets copied verbatim. There is no
11 * room for interpretation in a flag table -- `CS8` is 0x30 because the UART
12 * divides c_cflag by 16 to program the line control register, and 8250_port.c
13 * does exactly that at serial8250_do_set_termios(). Inventing a "close enough"
14 * value here would silently produce a port that transmits but never receives.
15 *
16 * The kernel-side helpers (tty_termios_baud_rate and friends) are the reason
17 * this header is load-bearing rather than just a table: serial_core.c:504
18 * reads the line rate through tty_termios_baud_rate() to decide whether the
19 * port can be started, and 8250_port.c:2817 re-encodes a rate back into
20 * c_cflag so the divisor can be recomputed. Both directions have to agree, so
21 * they are written as one table read one way and one table scanned the other:
22 *
23 * tty_termios_baud_rate() c_cflag & CBAUD -> bits per second
24 * tty_termios_encode_baud_rate() bits per second -> c_cflag & CBAUD
25 *
26 * BOTHER (0x1000) means "not a standard rate": the real number lives in
27 * c_ospeed/c_ispeed, which is how a caller requests 4 MHz from a UART that has
28 * no named constant for it.
29 *
30 * Mainline keeps these as non-inline functions in drivers/tty/tty_termios.c.
31 * They are static inline here because stage 2 has no tty_termios.c to link
32 * against and a declaration nothing defines is a link error waiting for its
33 * first caller. The bodies are self-contained (a switch and a table), so the
34 * cost of inlining is nothing next to the port I/O it guards.
35 */

298/*
299 * --- kernel-side rate helpers ------------------------------------------
300 *
301 * One table, read in both directions. The order matters: the index *is* the
302 * CBAUD encoding for 1..15 and 0x1001..0x100f, so baud_table[1] is 50 and
303 * baud_table[0x1001 - 1] would be wrong -- hence the two explicit ranges
304 * below rather than one array with holes in it.
305 */
306static inline speed_t __dcl_cbaud_to_rate(unsigned int cbaud)
307{
308 static const speed_t std_rate[16] = {
309 0, 50, 75, 110, 134, 150, 200, 300,
310 600, 1200, 1800, 2400, 4800, 9600, 19200, 38400
311 };
312 static const speed_t high_rate[16] = {
313 /* index 0 == BOTHER: the rate is not encoded in c_cflag */
314 0, 57600, 115200, 230400, 460800, 500000, 576000, 921600,
315 1000000, 1152000, 1500000, 2000000, 2500000, 3000000,
316 3500000, 4000000
317 };
318
319 if (cbaud & CBAUDEX)
320 return high_rate[cbaud & 0x0f];
321 return std_rate[cbaud & 0x0f];
322}

324static inline speed_t tty_termios_baud_rate(const struct ktermios* termios)
325{
326 unsigned int cbaud = termios->c_cflag & CBAUD;
327
328 if (cbaud == BOTHER)
329 return termios->c_ospeed;
330 return __dcl_cbaud_to_rate(cbaud);
331}

333/*
334 * Input rate: CIBAUD is the output encoding shifted left by IBSHIFT, so the
335 * same table answers for it. A c_cflag that never mentions CIBAUD (every
336 * port before someone has set an input rate) falls back to the output rate --
337 * the common case, since tcsetattr() writes both together.
338 */
339static inline speed_t tty_termios_input_baud_rate(const struct ktermios* termios)
340{
341 unsigned int ibaud = (termios->c_cflag & CIBAUD) >> IBSHIFT;
342
343 if (ibaud == 0)
344 return tty_termios_baud_rate(termios);
345 if (ibaud == BOTHER)
346 return termios->c_ispeed;
347 return __dcl_cbaud_to_rate(ibaud);
348}

388/*
389 * Encode a pair of rates into c_cflag. Note the shift: CIBAUD is CBAUD in the
390 * high half, so the input encoding is written one IBSHIFT to the left. Getting
391 * this backwards yields a port whose *output* runs at the *input* rate, which
392 * is invisible until someone opens a modem at 9600 and sees garbage.
393 */
394static inline void tty_termios_encode_baud_rate(struct ktermios* termios,
395 speed_t ibaud, speed_t obaud)
396{
397 termios->c_cflag &= ~(CBAUD | CIBAUD);
398 termios->c_cflag |= __dcl_rate_to_cbaud(obaud) |
399 ((unsigned int)__dcl_rate_to_cbaud(ibaud) << IBSHIFT);
400 termios->c_ispeed = ibaud;
401 termios->c_ospeed = obaud;
402}
