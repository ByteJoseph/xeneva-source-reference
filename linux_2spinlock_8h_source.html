4/*
5 * DCL <linux/spinlock.h> -- the lock words tty/serial code wraps in guards.
6 *
7 * The primitives themselves are already in <linux/kernel.h> and they are
8 * no-ops:
9 *
10 * #define spin_lock_init(l) do{} while(0)
11 * #define spin_lock_irqsave(l,f) do{(f) = 0;}while(0)
12 * #define spin_unlock_irqrestore(l, f) do{}while(0)
13 *
14 * That is DCL's pre-existing contract -- driver work runs in one kernel
15 * context and Xeneva serialises at the process level -- and it is not a
16 * shortcut taken here. What this header adds is the rest of the *shape*
17 * mainline's code is written in: plain spin_lock/spin_unlock, the _irq and
18 * _bh variants (kernel.h carries only init and the irqsave pair), and the
19 * three guard classes <linux/cleanup.h> needs in order for
20 * `scoped_guard(spinlock_irq, &lock) { ... }` to mean anything.
21 *
22 * If real locking ever lands, the two places to change are kernel.h's macros
23 * and this header's guards -- nothing in between depends on the details.
24 *
25 * The irqsave guard deserves one honest note: because spin_lock_irqsave()
26 * sets flags to 0 and never touches the interrupt state, the guard has no
27 * interrupt state to carry across its scope, so it restores from a zero. If
28 * spin_lock_irqsave() ever becomes real, this guard must grow a flags field
29 * (DCL_LOCK_GUARD's struct is otherwise unchanged) -- otherwise it would
30 * silently re-enable interrupts that the acquire disabled.
31 */
32

47/*
48 * The raw-spinlock spelling -- DEFINE_RAW_SPINLOCK(), raw_spin_lock_irqsave(),
49 * raw_spin_unlock_irqrestore(). Only tty_ldisc.c uses it (:45, :112, :114),
50 * to guard the tty_ldiscs[] table, and the reason mainline distinguishes raw
51 * locks has no referent here: DCL's spinlock_t *is* an int with every
52 * operation a no-op (kernel.h:43), so there is no raw lock underneath the
53 * non-raw one for the distinction to point at. tty_ldisc.h's own note says
54 * the same thing about ld_semaphore's wait_lock.
55 *
56 * The type is `spinlock_t`, which the file-scope definition at :45 needs to
57 * be a complete type -- `int` -- and the flags assignment mirrors
58 * spin_lock_irqsave() in kernel.h: this never reads interrupt state, so the
59 * matching restore is a no-op rather than a re-enable of something that was
60 * never disabled.
61 */
62/*
63 * raw_spin_lock_init() -- tty_ldsem.c:69, inside __init_ldsem. Deliberately
64 * the same no-op as kernel.h's spin_lock_init() rather than a store to 0:
65 * the two spellings differ only in mainline's raw/non-raw split, which this
66 * header has already established has no referent here (see the block above),
67 * and two macros that initialised the same word differently would be a
68 * difference nobody could explain from the source that calls them. The word
69 * is never read -- spin_lock() is `do {} while (0)` -- so a value it does not
70 * have cannot matter.
71 */
72/*
73 * raw_spin_lock_irq() / raw_spin_unlock_irq() -- tty_ldsem.c's contended
74 * paths (:162, :175, :184, :190, :211 and the write-side twins), which drop
75 * the flags argument the _irqsave pair takes because they never saved any.
76 * Same body as spin_lock_irq()/spin_unlock_irq() three blocks up, and for the
77 * same reason: interrupts are never disabled by a lock here, so there is
78 * nothing for the unlock to re-enable. mainline distinguishes them by
79 * whether flags travel with the call; that is the whole difference.

116/*
117 * spin_trylock_irqsave(lock, flags) -- try the lock, and set flags either way.
118 *
119 * It has to be a macro, not a function: the call site is
120 * `spin_trylock_irqsave(&up->lock, *flags)` (serial_core.h:1005), where
121 * `flags` arrives as a *value* that the macro is expected to assign into the
122 * caller's lvalue and whose success boolean is the macro's value. A function
123 * could take a pointer and return the boolean, but it could not reproduce
124 * that spelling -- and the spelling is mainline's, not DCL's to change.
125 *
126 * Statement expression, so it can both assign and yield. mainline's
127 * implementation saves the interrupt flags first and restores them on
128 * failure; DCL's spin_lock_irqsave() sets flags to 0 and does not touch
129 * interrupt state at all (the header's own note), so saving means recording
130 * 0, and the matching restore in uart_port_trylock_irqsave()'s failure arm is
131 * a no-op too. Returning spin_trylock(l) -- which is `(1)` -- keeps the
132 * "lock acquired" answer true, which is the only thing the caller branches on.
133 */
134#define spin_trylock_irqsave(l, f)\
135 ({ (f) = 0; spin_trylock(l); })
