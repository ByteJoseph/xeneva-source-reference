4/*
5 * DCL <linux/spinlock.h> -- the lock words tty/serial code wraps in guards.
6 *
7 * The primitives themselves are already in <linux/kernel.h> and they are
8 * no-ops:
9 *
10 * #define spin_lock_init(l) do{} while(0)
11 * #define spin_lock_irqsave(l,f) do{(f) = 0;}while(0)
12 * #define spin_unlock_irqrestore(l, f) do{}while(0)
13 *
14 * That is DCL's pre-existing contract -- driver work runs in one kernel
15 * context and Xeneva serialises at the process level -- and it is not a
16 * shortcut taken here. What this header adds is the rest of the *shape*
17 * mainline's code is written in: plain spin_lock/spin_unlock, the _irq and
18 * _bh variants (kernel.h carries only init and the irqsave pair), and the
19 * three guard classes <linux/cleanup.h> needs in order for
20 * `scoped_guard(spinlock_irq, &lock) { ... }` to mean anything.
21 *
22 * If real locking ever lands, the two places to change are kernel.h's macros
23 * and this header's guards -- nothing in between depends on the details.
24 *
25 * The irqsave guard deserves one honest note: because spin_lock_irqsave()
26 * sets flags to 0 and never touches the interrupt state, the guard has no
27 * interrupt state to carry across its scope, so it restores from a zero. If
28 * spin_lock_irqsave() ever becomes real, this guard must grow a flags field
29 * (DCL_LOCK_GUARD's struct is otherwise unchanged) -- otherwise it would
30 * silently re-enable interrupts that the acquire disabled.
31 */

47/*
48 * The raw-spinlock spelling -- DEFINE_RAW_SPINLOCK(), raw_spin_lock_irqsave(),
49 * raw_spin_unlock_irqrestore(). Only tty_ldisc.c uses it (:45, :112, :114),
50 * to guard the tty_ldiscs[] table, and the reason mainline distinguishes raw
51 * locks has no referent here: DCL's spinlock_t *is* an int with every
52 * operation a no-op (kernel.h:43), so there is no raw lock underneath the
53 * non-raw one for the distinction to point at. tty_ldisc.h's own note says
54 * the same thing about ld_semaphore's wait_lock.
55 *
56 * The type is `spinlock_t`, which the file-scope definition at :45 needs to
57 * be a complete type -- `int` -- and the flags assignment mirrors
58 * spin_lock_irqsave() in kernel.h: this never reads interrupt state, so the
59 * matching restore is a no-op rather than a re-enable of something that was
60 * never disabled.

101/*
102 * __SPIN_LOCK_UNLOCKED(lockname) -- the static initialiser, the spelling a
103 * file-scope struct field needs where DEFINE_SPINLOCK's `name = 0` cannot be
104 * used. Two consumers, both struct members rather than objects:
105 *
106 * xarray.h:308 .xa_lock = __SPIN_LOCK_UNLOCKED(name.xa_lock)
107 * seqlock.h:809 .lock = __SPIN_LOCK_UNLOCKED(lockname)
108 *
109 * Neither existed here until drm_drv.c:65 expanded DEFINE_XARRAY_ALLOC()
110 * (drm_minors_xa, the minor-number allocator). Undeclared, the token reads as
111 * an implicit int function call inside a static initialiser, and clang reports
112 * "initializer element is not a compile-time constant" at the *declaration* --
113 * a message that never mentions __SPIN_LOCK_UNLOCKED, so the missing name is
114 * only findable by expanding the macro by hand.
115 *
116 * mainline's is a typeof() cast over a brace initialiser carrying the arch
117 * unlock value, lockdep maps and debug fields. None of that has a referent:
118 * spinlock_t is an int (kernel.h:43) and every lock operation is a no-op, so
119 * the lock starts unlocked by having no other state to start in. `{ 0 }`
120 * rather than `0` because the context is an initialiser for a struct member
121 * and braced is what the other two initialisers in XARRAY_INIT spell.
122 *
123 * The argument is dropped, not expanded -- callers pass an lvalue (an
124 * existing lock, or a field of the object being defined) and evaluating it
125 * during its own initialisation would be a read of an object not yet
126 * complete.
127 */
128#ifndef __SPIN_LOCK_UNLOCKED
129#define __SPIN_LOCK_UNLOCKED(lockname) { 0 }

147/*
148 * spin_trylock_irqsave(lock, flags) -- try the lock, and set flags either way.
149 *
150 * It has to be a macro, not a function: the call site is
151 * `spin_trylock_irqsave(&up->lock, *flags)` (serial_core.h:1005), where
152 * `flags` arrives as a *value* that the macro is expected to assign into the
153 * caller's lvalue and whose success boolean is the macro's value. A function
154 * could take a pointer and return the boolean, but it could not reproduce
155 * that spelling -- and the spelling is mainline's, not DCL's to change.
156 *
157 * Statement expression, so it can both assign and yield. mainline's
158 * implementation saves the interrupt flags first and restores them on
159 * failure; DCL's spin_lock_irqsave() sets flags to 0 and does not touch
160 * interrupt state at all (the header's own note), so saving means recording
161 * 0, and the matching restore in uart_port_trylock_irqsave()'s failure arm is
162 * a no-op too. Returning spin_trylock(l) -- which is `(1)` -- keeps the
163 * "lock acquired" answer true, which is the only thing the caller branches on.
164 */
165#define spin_trylock_irqsave(l, f)\
166 ({ (f) = 0; spin_trylock(l); })
