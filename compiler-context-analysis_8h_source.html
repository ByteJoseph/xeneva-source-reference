301/*
302 * Clang's analysis does not care precisely about the value, only that it is
303 * either zero or non-zero. So the __cond_acquires() interface might be
304 * misleading if we say that @ret is the value returned if acquired. Instead,
305 * provide symbolic variants which we translate.
306 */
307#define __cond_acquires_impl_true(x, ...) __try_acquires##__VA_ARGS__##_ctx_lock(1, x)

310#define __cond_acquires_impl_0(x, ...) __try_acquires##__VA_ARGS__##_ctx_lock(0, x)
311#define __cond_acquires_impl_nonnull(x, ...) __try_acquires##__VA_ARGS__##_ctx_lock(1, x)
