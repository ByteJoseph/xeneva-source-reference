4/*
5 * DCL <linux/bitops.h> -- bit arithmetic.
6 *
7 * tty_buffer.c includes this for one thing: __ALIGN_MASK(), which rounds a
8 * requested buffer length up to the next 256-byte boundary
9 * (`size = __ALIGN_MASK(size, TTYB_ALIGN_MASK)` at tty_buffer.c:172). That
10 * rounding is what lets the free list work at all -- a recycled buffer is only
11 * reused when the new request is <= MIN_TTYB_SIZE, and both sides have to
12 * agree on the granularity or a 257-byte request silently gets a 256-byte
13 * buffer.
14 *
15 * <linux/kernel.h> already has the same operation spelled __ALIGN_KERNEL_MASK
16 * with an extra typeof() cast; __ALIGN_MASK is mainline's name for it, kept
17 * here rather than in kernel.h because nothing outside a ported file has ever
18 * asked for that spelling.
19 *
20 * The bit-search helpers (find_next_bit and friends) are deliberately absent:
21 * no file in the staged tty/serial set calls them, and a bitmap search with
22 * no caller is a function nobody has ever compiled.
23 */

74/*
75 * assign_bit -- set or clear according to a value, in one expression.
76 * tty_port.h's flow-control setters are written against it (set_cts_flow,
77 * set_check_carrier, set_initialized, ...), all of which take a bool and are
78 * called with a literal. Written as a statement-macro rather than a function
79 * so `addr` keeps whatever type it arrived with -- tty_port_active() reads
80 * through a const pointer and tty_port_set_active() through a mutable one,
81 * and a function taking unsigned long* would force a cast on the first.
82 */
83#ifndef assign_bit
84#define assign_bit(nr, addr, val) \
