4/*
5 * DCL <linux/bitops.h> -- bit arithmetic.
6 *
7 * tty_buffer.c includes this for one thing: __ALIGN_MASK(), which rounds a
8 * requested buffer length up to the next 256-byte boundary
9 * (`size = __ALIGN_MASK(size, TTYB_ALIGN_MASK)` at tty_buffer.c:172). That
10 * rounding is what lets the free list work at all -- a recycled buffer is only
11 * reused when the new request is <= MIN_TTYB_SIZE, and both sides have to
12 * agree on the granularity or a 257-byte request silently gets a 256-byte
13 * buffer.
14 *
15 * <linux/kernel.h> already has the same operation spelled __ALIGN_KERNEL_MASK
16 * with an extra typeof() cast; __ALIGN_MASK is mainline's name for it, kept
17 * here rather than in kernel.h because nothing outside a ported file has ever
18 * asked for that spelling.
19 *
20 * The bit-search helpers used to be deliberately absent, on the grounds that
21 * no file in the staged tty/serial set called them. drm_drv.c:574 now does --
22 * for_each_set_bit() over the wedge-recovery bitmask in drm_dev_wedged_event()
23 * -- so they are here, and the reasoning is restated rather than deleted: a
24 * bitmap search still has no caller that would notice how it is implemented,
25 * only one that needs the right bit number back. The loops below are a linear
26 * walk; mainline's find_next_bit() reads a word at a time and skips whole
27 * zero words. That difference is invisible at the one call site, whose
28 * bitmap is a single unsigned long, and would stop being invisible the day a
29 * driver scans a large mask in a hot path -- at which point the word-wise
30 * version belongs here, not a faster loop over the same array.
31 */

33/*
34 * BITS_PER_LONG and DECLARE_BITMAP sit above the kernel.h include, which is
35 * not where either would naturally go. kernel.h pulls in a wide subtree
36 * (virtio.h, completion.h, wait.h, sched.h) and nodemask_types.h sits at the
37 * far end of it, already using DECLARE_BITMAP(bits, MAX_NUMNODES). By the time
38 * that chain reaches it this header's own include guard is set, so
39 * nodemask_types.h's #include <linux/bitops.h> is a no-op and a definition
40 * placed below would not have been seen. Defining them first is what makes
41 * that chain parse.
42 *
43 * For DECLARE_BITMAP that is also the only order that works, not merely one
44 * that does: <linux/bitmap.h> includes this header at its top, so a file
45 * starting there arrives with DECLARE_BITMAP undefined and bitmap.h's own copy
46 * -- mainline's home for it -- not yet reached. It has to live here.
47 *
48 * BITS_PER_LONG is a fallback only -- <linux/kernel.h> also defines it. Kept
49 * in step with kernel.h (sizeof(long)*8, not 64): this target is LLP64, see
50 * the note there for what a hardcoded 64 does to the word index and the shift
51 * below.

141/*
142 * BIT_WORD, __ffs, __set_bit, __test_and_set_bit, __test_and_clear_bit --
143 * the __-prefixed base operations, which are a family beside the macros above
144 * rather than another spelling of them.
145 *
146 * mainline's split between the two is atomicity: set_bit() may be atomic on
147 * some architectures and __set_bit() explicitly is not, and a caller picks the
148 * __ form when it already holds the lock or does not need one. DCL's
149 * set_bit() above is already a plain |= with no barrier, so the two are the
150 * same operation here -- but they stay two names, because the callers are
151 * mainline's (xarray.c:95, :102, :135 and drm_mm.c:478) and a later port that
152 * does need the distinction should not find half the family missing.
153 *
154 * __ffs() is index-of-first-set-bit, and mainline documents the answer as
155 * undefined for a word of 0: drm_mm.c:1887 calls it on `last + 1`, which the
156 * surrounding loop has already established is non-zero. __builtin_ctzl is
157 * what mainline's generic version reaches for as well; `long` is 32 bits on
158 * this target and the builtin follows the type, so the width is right.
159 *
160 * __test_and_*_bit() return the bit's previous state, which is what makes the
161 * xarray mark code read as it does -- node_set_mark() answers "was it already
162 * set" from the return, not from a second read.
163 */
164#ifndef BIT_WORD
165#define BIT_WORD(nr) ((nr) / BITS_PER_LONG)
166#endif

168static inline unsigned long __ffs(unsigned long word)

170 return (unsigned long)__builtin_ctzl(word);

178static inline int __test_and_set_bit(int nr, volatile unsigned long* addr)
179{
180 unsigned long* p = (unsigned long*)addr + BIT_WORD(nr);
181 unsigned long old = *p;
182 unsigned long mask = BIT_MASK(nr);
183
184 *p = old | mask;
185 return (old & mask) != 0;
186}

188static inline int __test_and_clear_bit(int nr, volatile unsigned long* addr)
189{
190 unsigned long* p = (unsigned long*)addr + BIT_WORD(nr);
191 unsigned long old = *p;
192 unsigned long mask = BIT_MASK(nr);
193
194 *p = old & ~mask;
195 return (old & mask) != 0;
196}

198/*
199 * __clear_bit and clear_bit_unlock -- the two clears the DRM set calls by
200 * name, beside the set/test family above.
201 *
202 * __clear_bit is the non-atomic member: drm_mm.c:844 drops
203 * DRM_MM_NODE_SCANNED_BIT after a scan block is removed, with the scan list
204 * already private to the caller. Its body is __set_bit's mirror; written as
205 * a typed inline rather than reusing the clear_bit macro so the address
206 * keeps the same information __set_bit already gets.
207 *
208 * clear_bit_unlock is mainline's name for "clear it and let it be seen"
209 * when releasing a lock -- drm_mm.c:648 clears DRM_MM_NODE_ALLOCATED_BIT as
210 * the node is freed, and any concurrent scanner must not miss that. On this
211 * uniprocessor target every bit operation is already a plain
212 * read-modify-write with no barrier to order it against, so the two are the
213 * same instruction sequence; keeping the name means the vendored call site
214 * does not have to be rewritten to say so, and the day this port gains an
215 * SMP build the barrier belongs in exactly this one place.
216 */
217static inline void __clear_bit(int nr, volatile unsigned long* addr)
218{
219 addr[BIT_WORD(nr)] &= ~BIT_MASK(nr);
220}
221
222static inline void clear_bit_unlock(int nr, volatile unsigned long* addr)
223{
224 __clear_bit(nr, addr);
225}

227/*
228 * assign_bit -- set or clear according to a value, in one expression.
229 * tty_port.h's flow-control setters are written against it (set_cts_flow,
230 * set_check_carrier, set_initialized, ...), all of which take a bool and are
231 * called with a literal. Written as a statement-macro rather than a function
232 * so `addr` keeps whatever type it arrived with -- tty_port_active() reads
233 * through a const pointer and tty_port_set_active() through a mutable one,
234 * and a function taking unsigned long* would force a cast on the first.
235 */
236#ifndef assign_bit
237#define assign_bit(nr, addr, val) \
238 do { \
239 if (val) \
240 set_bit((nr), (addr)); \
241 else \
242 clear_bit((nr), (addr)); \
243 } while (0)
244#endif

299/*
300 * ffs, fls, fls_long, hweight32/hweight64 -- word scans and population
301 * counts, the second family of bit arithmetic the DRM set reached for.
302 *
303 * mainline gets ffs and fls from <asm-generic/bitops/>; both reduce to the
304 * two builtins used here. __builtin_ffs() answers 0 for 0, which is ffs()'s
305 * contract; fls() must answer 0 too and __builtin_clz() is undefined for 0,
306 * hence the ternary in front of it. drm_blend.c:352 turns the lowest set
307 * rotation bit into an array index (ffs of a mask), and
308 * drm_client_modeset.c:191 asks for the highest set rotation bit (fls).
309 *
310 * fls_long is fls() for this target's four-byte long, and the 64-bit arm is
311 * removed by the preprocessor rather than left behind a runtime
312 * `if (BITS_PER_LONG == 64)`: both arms of such a condition would have to
313 * compile, and DCL carries no fls64(). dma-resv.c:335 (the fence-array
314 * growth estimate) is the only caller.
315 *
316 * hweight32/hweight64 are the popcounts. drm_crtc.c:991 and
317 * drm_crtc_helper.c:507 ask whether a crtc can be driven by more than one
318 * encoder, drm_connector.c:2444/:3351 count encoders and scaling modes, and
319 * drm_property.c:218 sizes a bitmask property by how many bits its
320 * validation mask has. The builtins follow the argument type, so the 32/64
321 * split is exactly what each name promises.
322 */
323#ifndef ffs
324#define ffs(x) __builtin_ffs(x)
325#endif
