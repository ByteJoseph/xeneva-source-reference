30/*
31 * calevt.h -- calendar event store, shared by the Calendar app and
32 * XELnch's LuitFlow ribbon. Header only, no allocation.
33 *
34 * On disk (FAT32 without long file names, so every name is 8.3):
35 *
36 * /CALEVT/ one folder for all events
37 * /CALEVT/202611.EVT one file per month: YYYYMM.EVT
38 *
39 * file = CalEvtHeader + count * CalEvent, events sorted by
40 * (day, hour, minute). A month with no events has no file.
41 *
42 * Costs: a month is one read / one write of at most 4060 bytes.
43 * Lookups are binary searches over the sorted month, O(log n); an insert
44 * is O(log n) to find the slot + one memmove; the "which days have
45 * events" mask is rebuilt in O(n) only when the month changes.
46 */

108 static const uint8_t d[13] = { 0, 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31 };

114/* 0 = Sunday */
115static inline int CalEvtWeekday(int y, int m, int d) {
116 static const int t[12] = { 0, 3, 2, 5, 0, 3, 5, 1, 4, 6, 2, 4 };
117 if (m < 3)
118 y -= 1;
119 return (y + y / 4 - y / 100 + y / 400 + t[m - 1] + d) % 7;
120}

126/* wall clock in the calendar's time zone (days -> civil date, Hinnant) */
127static inline void CalEvtNow(CalNow* n) {
128 int64_t sec = 0, nsec = 0;
129 _KeGetWalltime(&sec, &nsec);
130 sec += CALEVT_TZ;
131 int64_t z = sec / 86400;
132 int64_t sod = sec % 86400;
133 if (sod < 0) {
134 sod += 86400;
135 z -= 1;
136 }
137 z += 719468;
138 int64_t era = (z >= 0 ? z : z - 146096) / 146097;
139 unsigned doe = (unsigned)(z - era * 146097);
140 unsigned yoe = (doe - doe / 1460 + doe / 36524 - doe / 146096) / 365;
141 int64_t y = (int64_t)yoe + era * 400;
142 unsigned doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
143 unsigned mp = (5 * doy + 2) / 153;
144 unsigned d = doy - (153 * mp + 2) / 5 + 1;
145 unsigned m = mp + (mp < 10 ? 3 : -9);
146 n->year = (int)(y + (m <= 2));
147 n->month = (int)m;
148 n->day = (int)d;
149 n->hour = (int)(sod / 3600);
150 n->minute = (int)((sod / 60) % 60);
151 n->weekday = CalEvtWeekday(n->year, n->month, n->day);
152}

165static inline void CalEvtPath(char* out, int year, int month) {
166 const char* dir = CALEVT_DIR "/";
167 int i = 0;
168 while (dir[i]) {
169 out[i] = dir[i];
170 i++;
171 }
172 out[i++] = (char)('0' + (year / 1000) % 10);
173 out[i++] = (char)('0' + (year / 100) % 10);
174 out[i++] = (char)('0' + (year / 10) % 10);
175 out[i++] = (char)('0' + year % 10);
176 out[i++] = (char)('0' + (month / 10) % 10);
177 out[i++] = (char)('0' + month % 10);
178 out[i++] = '.';
179 out[i++] = 'E';
180 out[i++] = 'V';
181 out[i++] = 'T';
182 out[i] = '\0';
183}

185/* first index whose key >= key, O(log n) */
186static inline int CalEvtLowerBound(const CalMonth* cm, uint32_t key) {
187 int lo = 0, hi = cm->count;
188 while (lo < hi) {
189 int mid = (lo + hi) >> 1;
190 if (CalEvtKeyOf(&cm->ev[mid]) < key)
191 lo = mid + 1;
192 else
193 hi = mid;
194 }
195 return lo;
196}

198/* events of one day: returns how many, *first = index of the first */
199static inline int CalEvtDayRange(const CalMonth* cm, int day, int* first) {
200 int a = CalEvtLowerBound(cm, CalEvtKey(day, 0, 0));
201 int b = CalEvtLowerBound(cm, CalEvtKey(day + 1, 0, 0));
202 *first = a;
203 return b - a;
204}

206static inline void CalEvtRebuildMask(CalMonth* cm) {
207 cm->dayMask = 0;

213/* insertion sort: the stored file is already sorted, so this is O(n)
214 * on good files and only repairs a hand-edited one */
215static inline void CalEvtSortFix(CalMonth* cm) {
216 for (int i = 1; i < cm->count; i++) {
217 CalEvent t = cm->ev[i];
218 uint32_t k = CalEvtKeyOf(&t);
219 int j = i - 1;
220 while (j >= 0 && CalEvtKeyOf(&cm->ev[j]) > k) {
221 cm->ev[j + 1] = cm->ev[j];
222 j--;
223 }
224 cm->ev[j + 1] = t;
225 }
226}

228/* load a month; a missing or foreign file gives an empty month. 0 = ok */
229static inline int CalEvtLoad(CalMonth* cm, int year, int month) {
230 cm->year = (uint16_t)year;
231 cm->month = (uint8_t)month;
232 cm->count = 0;
233 cm->dirty = 0;
234 cm->dayMask = 0;
235 char path[20];
236 CalEvtPath(path, year, month);
237 int fd = _KeOpenFile(path, FILE_OPEN_READ_ONLY);
238 if (fd == -1)
239 return 0;
240 XEFileStatus st;
241 memset(&st, 0, sizeof(st));
242 _KeFileStat(fd, &st);
243 size_t want = st.size < CALEVT_IMAGE_MAX ? st.size : CALEVT_IMAGE_MAX;
244 size_t got = want >= sizeof(CalEvtHeader) ? _KeReadFile(fd, &cm->hdr, want) : 0;
245 _KeCloseFile(fd);
246 if (got < sizeof(CalEvtHeader) || got > CALEVT_IMAGE_MAX || cm->hdr.magic != CALEVT_MAGIC)
247 return -1;
248 int n = cm->hdr.count;
249 int avail = (int)((got - sizeof(CalEvtHeader)) / sizeof(CalEvent));
250 if (n > avail)
251 n = avail;
252 if (n > CALEVT_MAX)
253 n = CALEVT_MAX;
254 int ok = 0;
255 for (int i = 0; i < n; i++) {
256 CalEvent* e = &cm->ev[i];
257 if (e->day < 1 || e->day > 31 || e->hour > 23 || e->minute > 59)
258 continue;
259 e->title[CALEVT_TITLE_MAX - 1] = '\0';
260 if (ok != i)
261 cm->ev[ok] = *e;
262 ok++;
263 }
264 cm->count = (uint16_t)ok;
265 CalEvtSortFix(cm);
266 CalEvtRebuildMask(cm);
267 return 0;
268}

270/*
271 * write the month back: the folder is created on first use, the old file
272 * removed first (the FAT layer does not truncate on rewrite) and the new
273 * one written in one call. An empty month just removes its file.
274 */
275static inline int CalEvtSave(CalMonth* cm) {
276 char path[20];
277 CalEvtPath(path, cm->year, cm->month);
278 _KeCreateDir((char*)CALEVT_DIR);
279 _KeRemoveFile(path);
280 cm->dirty = 0;
281 if (cm->count == 0)
282 return 0;
283 int fd = _KeOpenFile(path, FILE_OPEN_WRITE | FILE_OPEN_CREAT);
284 if (fd == -1)
285 return -1;
286 memset(&cm->hdr, 0, sizeof(cm->hdr));
287 cm->hdr.magic = CALEVT_MAGIC;
288 cm->hdr.count = cm->count;
289 cm->hdr.year = cm->year;
290 cm->hdr.month = cm->month;
291 size_t len = sizeof(CalEvtHeader) + (size_t)cm->count * sizeof(CalEvent);
292 size_t put = _KeWriteFile(fd, &cm->hdr, len);
293 _KeCloseFile(fd);
294 return put == len ? 0 : -1;
295}

297/* insert keeping the order; returns the index or -1 when full */
298static inline int CalEvtInsert(CalMonth* cm, const CalEvent* e) {
299 if (cm->count >= CALEVT_MAX)
300 return -1;
301 /* after equal keys, so same-time events keep their entry order */
302 int at = CalEvtLowerBound(cm, CalEvtKeyOf(e) + 1);
303 memmove(&cm->ev[at + 1], &cm->ev[at], (size_t)(cm->count - at) * sizeof(CalEvent));
304 cm->ev[at] = *e;
305 cm->ev[at].title[CALEVT_TITLE_MAX - 1] = '\0';
306 cm->count++;
307 if (e->day < 32)
308 cm->dayMask |= 1u << e->day;
309 cm->dirty = 1;
310 return at;
311}

313static inline void CalEvtRemove(CalMonth* cm, int idx) {
314 if (idx < 0 || idx >= cm->count)
315 return;
316 int day = cm->ev[idx].day;
317 memmove(&cm->ev[idx], &cm->ev[idx + 1], (size_t)(cm->count - idx - 1) * sizeof(CalEvent));
318 cm->count--;
319 int first;
320 if (CalEvtDayRange(cm, day, &first) == 0)
321 cm->dayMask &= ~(1u << day);
322 cm->dirty = 1;
323}

353/*
354 * CalEvtFindNext -- the first event at or after now: the rest of this
355 * month, else the start of the next one (two reads at most, O(log n)
356 * each). scratch is clobbered. Returns 1 and fills out / daysAhead.
357 */
358static inline int CalEvtFindNext(CalMonth* scratch, const CalNow* now, CalEvent* out, int* daysAhead) {
359 if (CalEvtLoad(scratch, now->year, now->month) == 0) {
360 int i = CalEvtLowerBound(scratch, CalEvtKey(now->day, now->hour, now->minute));
361 if (i < scratch->count) {
362 *out = scratch->ev[i];
363 *daysAhead = out->day - now->day;
364 return 1;
365 }
366 }
367 int y = now->year, m = now->month + 1;
368 if (m > 12) {
369 m = 1;
370 y++;
371 }
372 if (CalEvtLoad(scratch, y, m) == 0 && scratch->count) {
373 *out = scratch->ev[0];
374 *daysAhead = CalEvtDaysIn(now->year, now->month) - now->day + out->day;
375 return 1;
376 }
377 return 0;
378}
