4/*
5 * DCL <linux/hashtable.h> -- the small open-chained table 8250_core.c keeps
6 * its per-irq port lists in:
7 *
8 * #define IRQ_HASH_BITS 5
9 * static DEFINE_HASHTABLE(irq_lists, IRQ_HASH_BITS);
10 * hash_add(irq_lists, &i->node, up->port.irq);
11 * hash_for_each_possible(irq_lists, obj, node, irq) ...
12 * hlist_del(&i->node);
13 *
14 * It needs three things to be right rather than to merely compile: both the
15 * add and the lookup must fold the key the same way (otherwise a port would
16 * be filed in one bucket and searched for in another and simply never be
17 * found), the bucket count must be a compile-time power of two, and the node
18 * unlink must tolerate a node that was never linked (8250_core.c:123 deletes
19 * a port it may not have inserted).
20 *
21 * hash_min() is therefore one macro used by both sides. Mainline folds with
22 * Fibonacci hashing; this is the plain mask. The two agree on which bucket a
23 * key belongs to -- only the spread differs, and 32 buckets over a handful of
24 * UARTs does not care about spread. HASH_BITS() gets its log2 from __builtin_ctz,
25 * which is exact for the power-of-two sizes DECLARE/DEFINE_HASHTABLE create.
26 */
