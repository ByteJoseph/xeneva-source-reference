71/*
72 * mainline: kmalloc(size, gfp). Xeneva's allocator takes the size only and
73 * DCL drops the flags -- GFP_KERNEL is 0 here -- with one exception:
74 * __GFP_ZERO selects the zeroing path. It cannot be dropped, because
75 * vendored code reaches the zeroing through the flag and nowhere else:
76 * drmm_kzalloc() calls drmm_kmalloc(dev, size, gfp | __GFP_ZERO), and
77 * drm_managed.c's alloc_dr() builds the whole drmres block -- header plus
78 * payload -- with one kmalloc_node_track_caller(tot_size, gfp, nid). Ignore
79 * the flag and drmm_kzalloc() hands back whatever TLSF held last: vkms's
80 * plane came back with a garbage properties.count, and the first
81 * drm_object_attach_property() turned it into a wild store.
82 * dcl_kmalloc_flags() below is the dispatch that reads it.
83 *
84 * Zero-size requests are the one disagreement vendored code trips over:
85 * mainline's kmalloc(0) -- and kzalloc(0), kcalloc(0, ...) -- answer with
86 * ZERO_SIZE_PTR, which is non-NULL, and callers are written for that.
87 * drm_atomic_commit_init() is the example that found this:
88 *
89 * state->colorops = kzalloc_objs(*state->colorops, num_colorop);
90 * if (!state->colorops)
91 * goto fail;
92 *
93 * and a driver with no color pipelines reports num_colorop == 0 -- vkms
94 * does -- so a NULL here becomes -ENOMEM on every atomic commit before the
95 * ioctl has looked at anything the caller sent. The native allocators
96 * refuse zero (Mm/kmalloc.c: both return NULL for an empty request, kmalloc
97 * on `!size`, kcalloc on either factor), so the request is rounded up to
98 * one byte rather than special-cased: the pointer stays real, zeroed where
99 * the name promises it, and freeable -- none of which ZERO_SIZE_PTR would
100 * be here, since kfree() expects a TLSF block.
101 *
102 * The helper below is expanded at a point where the macro does not exist
103 * yet, so its body resolves to the native one-argument kmalloc() from
104 * Mm/kmalloc.h; no recursion, and native DCL files that include
105 * Mm/kmalloc.h directly keep calling the plain function.
106 */

108 return kmalloc(size ? size : 1);

696/*
697 * memdup_user_nul(src, len) -- copy `len` bytes out of a user pointer and
698 * append a terminator, so a driver gets a kernel string of exactly the
699 * length the user asked for. Callers are lib/string_helpers.c:179 and
700 * drm_ioctl.c:590, and both test with IS_ERR() rather than NULL -- which is
701 * the part worth getting right, because mainline distinguishes "no memory"
702 * (-ENOMEM) from "the user pointer was bad" (-EFAULT), and a NULL here would
703 * read as neither.
704 *
705 * The copy is memcpy rather than copy_from_user, and that is not a shortcut:
706 * <linux/mm_shim.c>'s opening comment records that Xeneva runs a syscall with
707 * the calling process's page tables live and the devfs bridge already hands
708 * raw user pointers to ->read/->write, so copy_from_user *is* a memcpy there
709 * too. The two ERR_PTR arms are kept because they are the contract, not
710 * because of a boundary.
711 *
712 * The len >= INT_MAX arm is mainline's guard against `len + 1` overflowing
713 * the allocation size; it is unreachable for a drm ioctl name and cheap to
714 * keep.
715 */
716static inline void* memdup_user_nul(const void* src, size_t len)
717{
718 char* p;
719
720 if (len >= (size_t)0x7fffffff)
721 return ERR_PTR(-EINVAL);
722
723 p = (char*)dcl_kmalloc((unsigned int)(len + 1));
724 if (!p)
725 return ERR_PTR(-ENOMEM);
726
727 memcpy(p, src, len);
728 p[len] = '\0';
729 return p;
730}
