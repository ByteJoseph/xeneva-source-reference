15/*
16 * mainline: kmalloc(size, gfp). Xeneva's allocator takes the size only --
17 * GFP flags are advisory in DCL (GFP_KERNEL is 0) -- so the 2-argument form
18 * mainline sources write drops the flags.
19 *
20 * The helper below is expanded at a point where the macro does not exist
21 * yet, so its body resolves to the native one-argument kmalloc() from
22 * Mm/kmalloc.h; no recursion, and native DCL files that include
23 * Mm/kmalloc.h directly keep calling the plain function.
24 */
25static inline void* dcl_kmalloc(unsigned int size) {
26 return kmalloc(size);
27}

31/*
32 * kzalloc(size, flags) -- allocate and zero.
33 *
34 * DCL needs this as its own thing because the native allocator does not zero.
35 * kmalloc() hands back whatever the TLSF free list last held; only kcalloc()
36 * (KernelAA64/Mm/kmalloc.c) memsets, and it does so because *it* is the one
37 * promising zeroed memory. Going through kcalloc(1, size) rather than
38 * kmalloc()+memset() means the zeroing is one native call with one size
39 * computation instead of an allocate-then-sweep, and it is the same code
40 * path that already carries the "this must be zero" contract.
41 *
42 * This matters more than it looks: virtio-rng.c does
43 *
44 * vi = kzalloc_obj(struct virtrng_info);
45 *
46 * and then reads vi->data_avail, vi->data_idx, vi->hwrng_register_done and
47 * vi->hwrng_removed before ever writing them. With a non-zeroing kmalloc()
48 * that is a garbage entropy-buffer length, a random memcpy bound, and a
49 * driver that may skip or repeat its own cleanup -- all invisible until the
50 * numbers happen to be large.
51 *
52 * The flags argument is accepted and dropped, exactly as kmalloc() drops it:
53 * GFP_KERNEL is 0 in DCL and __GFP_ZERO is advisory here, because zeroing is
54 * unconditional rather than opt-in.
55 */
56static inline void* dcl_kzalloc(unsigned int size) {
57 return kcalloc(1, size);
58}

60#define kzalloc(size, flags) dcl_kzalloc(size)

62/*
63 * kzalloc_obj(TYPE) -- mainline's "one object of this type, zeroed":
64 *
65 * vi = kzalloc_obj(struct virtrng_info);
66 *
67 * mainline's slab.h spells it `__alloc_objs(kzalloc, default_gfp(__VA_ARGS__),
68 * typeof(TYPE), 1)`, i.e. sizeof(typeof(TYPE)) bytes with room for exactly one.
69 * Single parameter here on purpose: that is the shape every caller in this
70 * tree uses, and GFP flags are advisory (see above), so there is nothing for a
71 * second argument to carry. A driver that does write kzalloc_obj(T, gfp) gets
72 * a syntax error at the call site rather than a silently ignored flag -- which
73 * is the useful outcome, since the fix is to call kzalloc(sizeof(T), gfp).
74 */
75#define kzalloc_obj(type) dcl_kzalloc((unsigned int)sizeof(type))

77/*
78 * kmalloc_obj(type, ...) -- the uninitialised twin of kzalloc_obj, and the
79 * two shapes the drivers actually write:
80 *
81 * p->em485 = kmalloc_obj(struct uart_8250_em485, GFP_ATOMIC);
82 * (8250_port.c:560)
83 * port = kmalloc_obj(struct port);
84 * (virtio_console.c:1328, :1974)
85 *
86 * kzalloc_obj() takes the type alone; kmalloc_obj() historically took flags
87 * as a second parameter, which is why the trailing parameter is `...` and not
88 * a named one: mainline's spelling varies by call site, and a macro that
89 * insists on exactly two rejects the one-argument caller with "too few
90 * arguments provided to function-like macro invocation" -- an error about the
91 * macro's arity that says nothing about which side is wrong.
92 *
93 * The flags are swallowed rather than honoured for the reason the block below
94 * gives at length: GFP_ATOMIC means "must not sleep", DCL's allocator does
95 * not block on anything -- it is a bump over the TLSF heap -- so the guarantee
96 * GFP_ATOMIC exists to make is satisfied whether or not the flag is looked at.
97 * Nothing else in gfp_t is honoured either (see the flags comment above
98 * kmalloc_flex: an unrecognised gfp bit should be a compile error there
99 * rather than a silently different allocation -- and with `...` it is now
100 * silently accepted, which is the one thing this change trades away).
101 */
102#define kmalloc_obj(type, ...) dcl_kmalloc((unsigned int)sizeof(type))

104/*
105 * kmalloc_flex(p, member, count, flags) -- allocate "struct + count of the
106 * flexible array".
107 *
108 * The call site is tty_buffer.c:187:
109 *
110 * p = kmalloc_flex(*p, data, 2 * size, GFP_ATOMIC | __GFP_NOWARN);
111 *
112 * -- the first argument is a *dereference*, because mainline's macro (slab.h
113 * v7.2) is `__alloc_flex(kmalloc, gfp, typeof(VAR_OR_TYPE), FAM, COUNT)`,
114 * i.e. `typeof(*p)` -- the struct, and the size follows from that type. The
115 * spelling below has to reproduce that, and the trap is where the parameter
116 * sits. Writing `sizeof(*(p))` substitutes `*p` into `*(p)`, giving
117 * `sizeof(*(*p))` -- a dereference of a struct -- which is precisely the
118 * "indirection requires pointer operand" error clang reported on the first
119 * build. So the parameter appears bare: `typeof(p)` with the argument tokens
120 * `*p` becomes `typeof(*p)`, the struct itself -- and a struct is not a
121 * pointer, so `(typeof(p))0` fails again ("used type 'typeof (*p)' (aka
122 * 'struct tty_buffer') where arithmetic or pointer type is required"). The
123 * null pointer comes from `typeof(&p)`: `&*p` is `struct tty_buffer *`, and
124 * `((typeof(&p))0)->data` is then an ordinary member access through a null
125 * pointer, unevaluated, giving sizeof(u8) == 1.
126 *
127 * The count is in *elements of the flexible member*, and the element size
128 * comes from the member rather than a hardcoded 1, so a change to `data[]`'s
129 * type cannot silently mis-size every allocation.
130 *
131/*
132 * The flags argument is accepted and ignored, and it is now `...` rather than
133 * a named fourth parameter: virtio_console.c:411 writes the three-argument
134 * form -- `kmalloc_flex(*buf, sg, pages)` -- because mainline's macro dropped
135 * its gfp argument, while tty_buffer.c:187 still writes the four-argument
136 * `kmalloc_flex(*p, data, 2 * size, GFP_ATOMIC | __GFP_NOWARN)`. A macro that
137 * insists on one count of parameters rejects the other caller with "too few
138 * arguments provided to function-like macro invocation", an error about
139 * arity rather than about anything either caller did wrong. Both forms
140 * substitute the same expansion; the swalled arguments are the GFP flags,
141 * which are documentation in DCL and not policy (see the note above).
142 *
143 * This is where GFP_ATOMIC | __GFP_NOWARN from tty_buffer.c lands, and both
144 * are advisory here. A failed allocation still returns NULL -- tty_buffer_alloc()
145 * checks for it -- so the "don't warn on failure" half of __GFP_NOWARN is moot:
146 * DCL's allocator does not warn.
147 *
148 * __GFP_NOWARN exists as a name only so that expression parses. GFP flags are
149 * all 0 (see kernel.h): they are documentation in DCL, not policy.
150 */
151#ifndef __GFP_NOWARN
152#define __GFP_NOWARN 0
153#endif
154#ifndef __GFP_ZERO
155#define __GFP_ZERO 0
156#endif
157#ifndef __GFP_RETRY_MAYFAIL
158#define __GFP_RETRY_MAYFAIL 0
159#endif

162 dcl_kmalloc((unsigned int)(sizeof(typeof(p)) + \
163 sizeof(*((typeof(&p))0)->member) * (size_t)(count)))

166/*
167 * kzalloc_objs(type, count) -- an array of `count` objects, zeroed.
168 *
169 * serial_core.c wants it twice: kzalloc_objs(struct uart_state, drv->nr) at
170 * :2737 to hold a driver's port states, and
171 * kzalloc_objs(*uport->tty_groups, num_groups) at :3109 for the attribute
172 * groups the tty device exposes. Both are `struct`-or-expression first, count
173 * second, which is why this is not kzalloc_obj() with an extra argument
174 * bolted on: kmalloc_obj takes flags there, kzalloc_obj takes nothing, and a
175 * single name with a different second parameter would be a trap. Separate
176 * name, and the plural is mainline's own distinction (kcalloc_array ->
177 * kcalloc).
178 *
179 * The cast to unsigned int matches what the rest of this header hands
180 * dcl_kzalloc; a count that overflows 32 bits of byte length would wrap here
181 * rather than fail, and the only counts in reach are the number of UART ports
182 * (small) and their attribute groups (smaller).
183 */
184#define kzalloc_objs(type, count) \
185 dcl_kzalloc((unsigned int)(sizeof(type) * (unsigned int)(count)))

187/*
188 * kmalloc_objs(type, count) -- the uninitialised twin, same two-argument
189 * shape as kzalloc_objs() above.
190 *
191 * virtio_console.c is the first caller: :1820 `vqs = kmalloc_objs(struct
192 * virtqueue *, nr_queues);` and its three siblings at :1821-1823, each
193 * allocating an array of pointers. The first argument is a *type* here --
194 * `struct virtqueue *` -- and sizeof() of a pointer type is exactly what an
195 * array of those pointers needs, so the expansion is the same
196 * sizeof * count as kzalloc_objs without the zeroing.
197 *
198 * Named separately rather than as kzalloc_objs with a flag bolted on, for the
199 * reason that block gives: two callers of two names cannot disagree about
200 * whether the memory comes back zeroed, whereas one name with a third
201 * parameter could be read either way at the call site.
202 */
203#define kmalloc_objs(type, count) \
204 dcl_kmalloc((unsigned int)(sizeof(type) * (unsigned int)(count)))

206/*
207 * kmemdup(src, len, gfp) -- copy `len` bytes into a fresh allocation.
208 *
209 * mainline declares it in <linux/slab.h> too (kernel: mm/util.c), so this is
210 * its home and not an import. One caller, virtio_console.c:1118:
211 *
212 * data = kmemdup(buf, count, GFP_ATOMIC);
213 * if (!data)
214 * return -ENOMEM;
215 *
216 * which needs exactly the contract mainline gives: NULL on failure, and the
217 * failure *checked for*, which is why the body must not silently fall back to
218 * anything. The gfp flag is swallowed like every other one in this header --
219 * see the note above kmalloc_obj -- and memcpy comes from <linux/string.h>,
220 * which this header now includes for it (kernel.h names memcpy but only
221 * commentates on where mainline gets it from).
222 *
223 * size_t length, unsigned int size: the cast matches what the rest of this
224 * header hands the allocator, and a length beyond 4 GB would wrap here rather
225 * than fail. The only caller's count is a virtio-console control message.
226 */
227static inline void* kmemdup(const void* src, size_t len, gfp_t gfp)
228{
229 void* p = dcl_kmalloc((unsigned int)len);
230
231 (void)gfp;
232 if (p && len)
233 memcpy(p, src, len);
234 return p;
235}

237/*
238 * get_zeroed_page() / free_page() -- the tty xmit buffer's allocator
239 * (serial_core.c:255 and :296, one page per port).
240 *
241 * TWO THINGS ABOUT THE SIGNATURE, both because this is aarch64-unknown-windows:
242 *
243 * 1. The parameter and return are `uintptr_t`, not mainline's `unsigned
244 * long`. This target is LLP64: unsigned long is 4 bytes, a pointer is 8,
245 * and get_zeroed_page hands back a heap address the TLSF allocator carved
246 * out of `0xFFFFC00000000000`. Declaring it `unsigned long` would narrow
247 * on the way out, so every page handed to a port would be an address in
248 * the low 4 GB that nothing maps -- and free_page() would then be handed
249 * back a number that could not identify it. This is the partner half of
250 * a delta recorded in Vendored/MANIFEST.md: the two `unsigned long` lines
251 * in serial_core.c that carry the address have to be uintptr_t too, and
252 * doing only one half of that pair would achieve nothing.
253 *
254 * 2. It is one page from the ordinary allocator, not from a page allocator:
255 * Xeneva's pmmngr manages physical pages for the initial identity map and
256 * has no per-page free list for the kernel heap, while kcalloc() already
257 * zeroes and kfree() already understands the block. The page needs to be
258 * zeroed because kfifo_init() is handed it as a fresh ring and reads
259 * nothing that was not written. PAGE_SIZE comes from <Mm/vmmngr.h>,
260 * included at the top of this header.
261 */
262static inline uintptr_t get_zeroed_page(gfp_t gfp)
263{
264 (void)gfp;
265 return (uintptr_t)dcl_kzalloc(PAGE_SIZE);
266}

274/*
275 * kasprintf(gfp, fmt, ...) -- allocate and format, mainline's
276 * include/linux/kernel.h spelling; it is here because this is the header
277 * that owns allocation and serial_core.c includes it directly for kzalloc_objs.
278 *
279 * The window is fixed at DCL_KASPRINTF_MAX rather than mainline's
280 * measure-then-allocate pair. There is no measuring call to hand the format
281 * to twice: _vsnprintf() would report a length, but the sizing pass upstream
282 * does needs vsnprintf(NULL, 0, ...) to be legal, and DCL's formatter lives
283 * in Xeneva's _print.c with no NULL-buffer contract to rely on -- guessing
284 * that it has one would be a heap overrun waiting on an assumption. So the
285 * buffer is allocated first at a size no caller can exceed and the copy is
286 * bounded, which fails as a truncated name instead of as a write past the
287 * end.
288 *
289 * The single caller is serial_core.c:3087, kasprintf(GFP_KERNEL, "%s%u",
290 * drv->dev_name, name_base + line) -- a driver name and a port number, an
291 * order of magnitude inside 128. If a ported file ever formats a longer
292 * thing here, the number to raise is DCL_KASPRINTF_MAX and the check is
293 * that kasprintf's caller still tests for NULL.
294 */
295#define DCL_KASPRINTF_MAX 128

297static inline char* kasprintf(gfp_t gfp, const char* fmt, ...)
298{
299 char* buf;
300 va_list ap;
301 int n;
302
303 (void)gfp; /* GFP is advisory in DCL, as above */
304
305 buf = (char*)dcl_kmalloc(DCL_KASPRINTF_MAX);
306 if (!buf)
307 return NULL; /* serial_core.c:3089 tests this */
308
309 va_start(ap, fmt);
310 n = _vsnprintf(buf, DCL_KASPRINTF_MAX, fmt, ap);
311 va_end(ap);
312
313 if (n < 0)
314 {
315 kfree(buf);
316 return NULL;
317 }
318 buf[n < DCL_KASPRINTF_MAX ? n : DCL_KASPRINTF_MAX - 1] = '\0';
319 return buf;
320}
