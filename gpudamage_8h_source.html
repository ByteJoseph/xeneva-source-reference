18 w > d->width - x || h > d->height - y)

20 uint32_t r = (uint32_t)(x + w), b = (uint32_t)(y + h);

33static inline int GpuDamageTake(GpuDamage* d, GpuSrvFrame* f) {
34 if (d->busy) return GPUSRV_E_BUSY;
35 if (!d->dirty) return GPUSRV_E_EMPTY;
36 if (++d->next == 0) ++d->next;
37 d->pending = d->next; d->busy = 1; d->dirty = 0;
38 f->sequence = d->pending;
39 f->x = d->left; f->y = d->top;
40 f->width = d->right - d->left; f->height = d->bottom - d->top;
41 return GPUSRV_OK;
42}

44static inline int GpuDamageAck(GpuDamage* d, uint64_t sequence) {
