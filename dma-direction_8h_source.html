5/*
6 * DCL <linux/dma-direction.h> -- the direction enum, and nothing else.
7 *
8 * mainline keeps it here so dma-mapping.h, dma-buf.h and the sg_table users
9 * can share one name. DCL's copy existed nowhere: every prototype that took
10 * `enum dma_data_direction` had the tag first mentioned inside its own
11 * parameter list, which C scopes to that declaration -- so dma-buf.h:579's
12 * declaration and the definition in DCL/drm_stubs.c were two different types
13 * with the same spelling, and clang said "conflicting types" rather than
14 * "you forgot a header". Pulling the enum into file scope fixes the pair and
15 * every later declaration at once.
16 *
17 * The values are mainline's: the direction is also the argument DCL's
18 * stubs ignore, but a caller that passes DMA_FROM_DEVICE into an sg_table
19 * walk should not have it become 4.
20 */
21enum dma_data_direction {
22 DMA_BIDIRECTIONAL = 0,
23 DMA_TO_DEVICE = 1,
24 DMA_FROM_DEVICE = 2,
25 DMA_NONE = 3,
26};
