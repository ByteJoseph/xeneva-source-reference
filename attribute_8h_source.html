20/* Provide public ATTRIBUTE_* names for the private _GL_ATTRIBUTE_*
21 macros used within Gnulib. */
22
23/* The placement of these attributes depends on the kind of declaration
24 and, in some cases, also on the programming language (C vs. C++).
25
26 In function declarations and function definitions:
27
28 * ATTRIBUTE_NOTHROW must come after the parameter list.
29
30 * The macros
31 ATTRIBUTE_CONST
32 ATTRIBUTE_PURE
33 DEPRECATED
34 MAYBE_UNUSED
35 NODISCARD
36 REPRODUCIBLE
37 UNSEQUENCED
38 must come before the return type, and more precisely:
39 - In a function declaration/definition without a storage-class
40 specifier: at the beginning of the declaration/definition.
41 - In a function declaration/definition with a storage-class
42 specifier:
43 - In C: before the storage-class specifier.
44 - In C++: between the storage-class specifier and the return type.
45
46 * The other macros can be placed
47 - Either
48 - In a function declaration/definition without a storage-class
49 specifier: at the beginning of the declaration/definition.
50 - In a function declaration/definition with a storage-class
51 specifier: between the storage-class specifier and the return
52 type.
53 - Or after the parameter list,
54 ∙ but after ATTRIBUTE_NOTHROW if present.
55
56 In other declarations, such as variable declarations:
57
58 * Either
59 - In C: before the storage-class specifier.
60 - In C++: between the storage-class specifier and the return type.
61 Then they apply to all entities that are declared by the declaration.
62
63 * Or immediately after the name of an entity being declared by the
64 declaration. Then they apply to that entity only.
65 */
66

72 * C23 standard attributes. These have macro names that do not begin with
73 'ATTRIBUTE_'.

119/* Attributes for memory-allocating functions. */
120
121/* The function returns a pointer to freshly allocated memory. */
122/* Applies to: functions. */
123#define ATTRIBUTE_MALLOC _GL_ATTRIBUTE_MALLOC
124
125/* ATTRIBUTE_ALLOC_SIZE ((N)) - The Nth argument of the function
126 is the size of the returned memory block.
127 ATTRIBUTE_ALLOC_SIZE ((M, N)) - Multiply the Mth and Nth arguments
128 to determine the size of the returned memory block. */
129/* Applies to: functions, pointer to functions, function types. */
130#define ATTRIBUTE_ALLOC_SIZE(args) _GL_ATTRIBUTE_ALLOC_SIZE (args)
131
132/* ATTRIBUTE_DEALLOC (F, I) declares that the function returns pointers
133 that can be freed by passing them as the Ith argument to the
134 function F.
135 ATTRIBUTE_DEALLOC_FREE declares that the function returns pointers that
136 can be freed via 'free'; it can be used only after declaring 'free'. */
137/* Applies to: functions. Cannot be used on inline functions. */
138#define ATTRIBUTE_DEALLOC(f, i) _GL_ATTRIBUTE_DEALLOC(f, i)
139#define ATTRIBUTE_DEALLOC_FREE _GL_ATTRIBUTE_DEALLOC_FREE

146/* Applies to: functions. */

150/* ================== Attributes for compiler diagnostics ================== */

152/* Attributes that help the compiler diagnose programmer mistakes.
153 Some of them may also help for some compiler optimizations. */
154
155/* ATTRIBUTE_FORMAT ((ARCHETYPE, STRING-INDEX, FIRST-TO-CHECK)) -
156 The STRING-INDEXth function argument is a format string of style
157 ARCHETYPE, which is one of:
158 printf, gnu_printf
159 scanf, gnu_scanf,
160 strftime, gnu_strftime,
161 strfmon,
162 or the same thing prefixed and suffixed with '__'.
163 If FIRST-TO-CHECK is not 0, arguments starting at FIRST-TO_CHECK
164 are suitable for the format string. */
165/* Applies to: functions. */
166#define ATTRIBUTE_FORMAT(spec) _GL_ATTRIBUTE_FORMAT (spec)
167
168/* ATTRIBUTE_NONNULL ((N1, N2,...)) - Arguments N1, N2,... must not be NULL.
169 ATTRIBUTE_NONNULL () - All pointer arguments must not be null. */
170/* Applies to: functions. */
171#define ATTRIBUTE_NONNULL(args) _GL_ATTRIBUTE_NONNULL (args)
172
173/* The function's return value is a non-NULL pointer. */
174/* Applies to: functions. */
175#define ATTRIBUTE_RETURNS_NONNULL _GL_ATTRIBUTE_RETURNS_NONNULL

177/* Warn if the caller does not use the return value,
178 unless the caller uses something like ignore_value. */
179/* Applies to: function, enumeration, class. */

236/* It is OK for a compiler to move calls to the function and to omit
237 calls to the function if another call has the same arguments or the
238 result is not used.
239 This attribute is safe for a function that neither depends on
240 nor affects state, and always returns exactly once -
241 e.g., does not raise an exception, call longjmp, or loop forever.
242 (This attribute is stricter than ATTRIBUTE_PURE because the
243 function cannot observe state. It is stricter than UNSEQUENCED
244 because the function must return exactly once and cannot depend on
245 state addressed by its arguments.) */
246/* Applies to: functions. */
247#define ATTRIBUTE_CONST _GL_ATTRIBUTE_CONST
248
249/* It is OK for a compiler to move calls to the function and to omit duplicate
250 calls to the function with the same arguments, so long as the state
251 addressed by its arguments is the same.
252 This attribute is safe for a function that is effectless, idempotent,
253 stateless, and independent; see ISO C 23 § 6.7.12.7 for a definition of
254 these terms.
255 (This attribute is stricter than REPRODUCIBLE because the function
256 must be stateless and independent. It is looser than ATTRIBUTE_CONST
257 because the function need not return exactly once and can depend
258 on state addressed by its arguments.)
259 See also <https://www.open-std.org/jtc1/sc22/wg14/www/docs/n2956.htm> and
260 <https://stackoverflow.com/questions/76847905/>.
261 ATTENTION! Efforts are underway to change the meaning of this attribute.
262 See <https://www.open-std.org/jtc1/sc22/wg14/www/docs/n3424.htm>. */
263/* Applies to: functions, pointer to functions, function type. */
264#define UNSEQUENCED _GL_ATTRIBUTE_UNSEQUENCED
265
266/* It is OK for a compiler to move calls to the function and to omit
267 calls to the function if another call has the same arguments or the
268 result is not used, and if observable state is the same.
269 This attribute is safe for a function that does not affect observable state
270 and always returns exactly once.
271 (This attribute is looser than ATTRIBUTE_CONST because the function
272 can depend on observable state. It is stricter than REPRODUCIBLE
273 because the function must return exactly once and cannot affect
274 state addressed by its arguments.) */
275/* Applies to: functions. */
276#define ATTRIBUTE_PURE _GL_ATTRIBUTE_PURE
277
278/* It is OK for a compiler to move calls to the function and to omit duplicate
279 calls to the function with the same arguments, so long as the state
280 addressed by its arguments is the same and is updated in time for
281 the rest of the program.
282 This attribute is safe for a function that is effectless and idempotent; see
283 ISO C 23 § 6.7.12.7 for a definition of these terms.
284 (This attribute is looser than UNSEQUENCED because the function need
285 not be stateless and idempotent. It is looser than ATTRIBUTE_PURE
286 because the function need not return exactly once and can affect
287 state addressed by its arguments.)
288 See also <https://www.open-std.org/jtc1/sc22/wg14/www/docs/n2956.htm> and
289 <https://stackoverflow.com/questions/76847905/>.
290 ATTENTION! Efforts are underway to change the meaning of this attribute.
291 See <https://www.open-std.org/jtc1/sc22/wg14/www/docs/n3424.htm>. */
292/* Applies to: functions, pointer to functions, function type. */
293#define REPRODUCIBLE _GL_ATTRIBUTE_REPRODUCIBLE
294
295/* The function is rarely executed. */
296/* Applies to: functions. */
297#define ATTRIBUTE_COLD _GL_ATTRIBUTE_COLD
298
299/* If called from some other compilation unit, the function executes
300 code from that unit only by return or by exception handling,
301 letting the compiler optimize that unit more aggressively. */
302/* Applies to: functions. */
303#define ATTRIBUTE_LEAF _GL_ATTRIBUTE_LEAF
304
305/* For struct members: The member has the smallest possible alignment.
306 For struct, union, class: All members have the smallest possible alignment,
307 minimizing the memory required. */
308/* Applies to: struct members, struct, union,
309 in C++ also: class. */
310#define ATTRIBUTE_PACKED _GL_ATTRIBUTE_PACKED
