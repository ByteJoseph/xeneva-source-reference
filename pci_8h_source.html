1/*
2 * DCL pci.h -- the PCI surface the drm set names, and now a bus behind it.
3 *
4 * This header was compile-only for its first act: there was no PCI on the
5 * boards this kernel targets, drm_file.c:987 and drm_ioctl.c:152 asked
6 * dev_is_pci(dev->dev) before every bus-id setup, the answer was always
7 * false, and the struct below was a three-member sketch sized for the BDF
8 * string those reads print. The scanout stage changed what it has to carry:
9 * QEMU's bochs-display is a real PCI device, and the vendored driver that
10 * owns it (Vendored/drivers/gpu/drm/tiny/bochs.c) compiles and links whole
11 * against this surface -- probe/remove/shutdown records, the id_table match
12 * quadruple, BAR resources with start/length/flags, and the
13 * pci_set/get_drvdata pair.
14 *
15 * The bus behind the surface is DCL/pci_shim.c: it walks the ECAM window the
16 * native pcie code already maps, fills one struct pci_dev per function, and
17 * runs id_table matching plus probe when a vendored driver registers. This
18 * header stays a pure data contract except for one participant:
19 * dev_is_pci() now has to answer for real devices (drm_file.c prints the BDF
20 * string when it is true), and the only honest registry of "which devices
21 * this bus made" lives in the fabric, so the inline forwards there. That
22 * query is also what keeps the vendored drm core's own PCI paths (drm_edid's
23 * switcheroo wrappers) answering correctly for a device that is now actually
24 * behind a PCI bus, and keeps answering false for every device that is not.
25 *
26 * struct device lives in the pinned <linux/fs.h> (fs.h:184), which is why
27 * this header includes fs.h rather than declaring the member opaque: an
28 * incomplete struct device cannot be a member. The pinned struct is 800
29 * bytes whether it describes a PCI device or not -- the fabric embeds it
30 * inside its own pci_dev and never grows it, per the note in device.h about
31 * writes at offset 800 being a heap overrun.
32 */

79/*
80 * struct pci_dev -- identity plus BARs, which is the whole of what a driver
81 * ported into this tree reads.
82 *
83 * mainline's pci_dev is a kernel object: device lists, PM state, MSI
84 * vectors, config-space accessors, the driver bound to it. None of that has
85 * code here -- the fabric matches once at initcall time and nothing ever
86 * unbinds -- so this carries the DCL delta on purpose, after the pinned
87 * struct device:
88 *
89 * - vendor/device/subsystem_vendor/subsystem_device and class, so
90 * pci_match_one_device()'s quadruple test and class filter can run.
91 * class is class<<8 | subclass, mainline's spelling.
92 * - revision, because bochs_hw_init() selects its register window from
93 * `pdev->revision >= 2` (the qext block exists from revision 2 on).
94 * - resource[6], filled for a matched device only (the fabric probes sizes
95 * by writing ~0 to a BAR -- a write effect no unmatched device should
96 * see) with mainline's start/end/flags fields.
97 * - driver_data, which is where pci_set_drvdata()/pci_get_drvdata()
98 * keep the drm_device pointer. A field here rather than mainline's
99 * dev->driver_data because struct device is the pinned 800-byte blob
100 * and device.h's note explains why nothing may grow it.
101 */
102struct pci_dev {
103 struct pci_bus* bus;
104 unsigned int devfn;
105 struct device dev;
106
107 unsigned short vendor;
108 unsigned short device;
109 unsigned short subsystem_vendor;
110 unsigned short subsystem_device;
111 unsigned int class; /* class<<8 | subclass */
112 unsigned char revision;
113 struct resource resource[6]; /* BARs 0..5, mainline's indices */
114 void* driver_data;
115};

117/*
118 * PCI_SLOT / PCI_FUNC -- the two bitfields packed into devfn. mainline spells

120 * prints both, and the fabric packs devices the same way.
121 */
122#define PCI_SLOT(devfn) (((devfn) >> 3) & 0xff)
123#define PCI_FUNC(devfn) ((devfn) & 0x07)

167/*
168 * struct pci_driver -- the registration record, reduced to what the fabric
169 * runs and what a driver initialises. mainline's err_handler, dynids,
170 * driver_managed_dma and friends have no code here to call them.
171 *
172 * The nested struct device_driver is the shared tag from
173 * <linux/device/driver.h> (also used by virtio_driver); only its `name` and
174 * `pm` members exist, and bochs.c writes `.driver.pm` through this nesting.

183 void (*shutdown)(struct pci_dev* dev);
184 struct device_driver driver;
185};

187/*
188 * Registration is match-and-probe, run immediately: the fabric scans once,
189 * then for every device whose id_table entry matches and which no earlier
190 * driver took, calls ->probe. There is no deferred bus callback to simulate
191 * because there is no other time for a device to appear.
192 *
193 * The return value is 0 even when probes fail -- mainline's
194 * pci_register_driver() reports whether the *driver* registered, not whether
195 * its probes succeeded, and a vendored driver keeps its initcall result
196 * unread either way. A driver whose probe failed on every device is visible
197 * in the boot log, not as an error code.
198 */
199int pci_register_driver(struct pci_driver* drv);
200void pci_unregister_driver(struct pci_driver* drv);

241 * pci_get_device() / pci_dev_put(), as virtio-gpu's init names them: mainline
242 * walks the device list and returns a match with a reference held, the caller
243 * dropping it with pci_dev_put() when done. The DCL fabric's slot table is
244 * the list (DCL/pci_shim.c walks ECAM into it; nothing is ever removed), and
245 * the from-argument keeps mainline's iteration contract -- resuming *after*
246 * the device handed to the previous call -- because the one caller passes
247 * NULL and a second caller, if it ever comes, will expect the same walk.
248 *
249 * The reference itself is the no-op of a bus whose devices outlive every
250 * driver: there is no removal path to fence against, so get is "the pointer
251 * is still valid" and put has nothing to drop. Both bodies are in
252 * DCL/pci_shim.c beside the scan.

258/*
259 * pci_is_vga() -- mainline's legacy-VGA class test, body verbatim
260 * (include/linux/pci.h:784 in linux-7.2.6): the device either announces the
261 * display-VGA class or the old not-defined-VGA one. Only the class matters
262 * here; mainline's ioport/parity extras are x86 machinery this port has no
263 * registers for.
264 *
265 * What reads it: virtio_gpu_driver_init()'s VGA-quirk branch. The device
266 * this port runs against (virtio-gpu-pci) reports 0x0380, so the answer is
267 * false and the branch -- vga_get_interruptible()/vga_put() -- is skipped;
268 * virtio_gpu_probe()'s aperture_remove_conflicting_pci_devices() is skipped
269 * for the same reason.
270 */
271static inline bool pci_is_vga(struct pci_dev* pdev)
272{
273 if ((pdev->class >> 8) == PCI_CLASS_DISPLAY_VGA)
274 return true;
275
276 if ((pdev->class >> 8) == PCI_CLASS_NOT_DEFINED_VGA)
277 return true;
278
279 return false;
280}
