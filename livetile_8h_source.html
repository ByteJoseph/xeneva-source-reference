30/*
31 * livetile.h -- public, header-only client API for the XELnch launcher.
32 *
33 * Any app can include this (Libs/Chitralekha is already on every app's
34 * include path) to push live-tile content to its own launcher tile, and
35 * Namdapha uses it to ask the launcher to show/hide itself with the
36 * Metro animation instead of toggling the window behind its back.
37 *
38 * Transport: one PostEvent per update, posted to the launcher's thread.
39 * The launcher registers itself under process token XELNCH_PROCESS_TOKEN,
40 * so lookup is a single syscall -- no name service, no polling.
41 *
42 * IMPORTANT: never put pointers in the event (pValue / charValue ...).
43 * They point into the sender's address space and are garbage inside the
44 * launcher. All text travels inline in charValue3.
45 *
46 * PostEvent layout for XELNCH_MSG_LIVE_UPDATE:
47 * dword = FNV-1a hash of the app path exactly as written in
48 * /lncher.cnf [App] (e.g. "/audplr.exe") -> XelnchHash()
49 * dword2 = layout (XELNCH_TILE_*)
50 * dword3 = badge count (0 = no badge)
51 * dword4 = progress 0..1000 (XELNCH_NO_PROGRESS = hidden)
52 * charValue3 = "line1\0line2\0" (<= 99 bytes total, truncated)
53 */

83/* FNV-1a 32 bit, O(len). Launcher and clients must hash the same bytes. */
84static inline uint32_t XelnchHash(const char* s) {
85 uint32_t h = 2166136261u;
86 while (*s) {
87 h ^= (uint8_t)*s++;
88 h *= 16777619u;
89 }
90 return h;
91}

93/* returns the launcher thread id or -1 when the launcher is not running */
94static inline int XelnchGetThread() {
95 int tid = _KeProcessTokenGetThreadID(XELNCH_PROCESS_TOKEN);
96 return tid > 0 ? tid : -1;
97}

99static inline int XelnchPost(int postboxfd, PostEvent* e) {
100 int tid = XelnchGetThread();
101 if (tid < 0)
102 return -1;
103 e->to_id = (uint16_t)tid;
104 _KeFileIoControl(postboxfd, POSTBOX_PUT_EVENT, e);
105 return 0;
106}

108/*
109 * @brief XelnchLiveTileUpdate -- push live content to this app's tile
110 * @param postboxfd -- ChitralekhaApp::postboxfd of the caller
111 * @param appPath -- path exactly as in /lncher.cnf, e.g. "/audplr.exe"
112 * @param layout -- XELNCH_TILE_*
113 * @param line1, line2 -- may be NULL
114 * @param badge -- 0 hides the badge
115 * @param progress -- 0..1000 or XELNCH_NO_PROGRESS
116 * Rate limit yourself: one update per second is plenty for a tile.
117 */
118static inline int XelnchLiveTileUpdate(int postboxfd,
119 const char* appPath,
120 uint8_t layout,
121 const char* line1,
122 const char* line2,
123 uint16_t badge,
124 uint16_t progress) {
125 PostEvent e;
126 memset(&e, 0, sizeof(PostEvent));
127 e.type = XELNCH_MSG_LIVE_UPDATE;
128 e.dword = XelnchHash(appPath);
129 e.dword2 = layout;
130 e.dword3 = badge;
131 e.dword4 = progress;
132 size_t off = 0;
133 const size_t cap = sizeof(e.charValue3) - 1;
134 if (line1) {
135 size_t n = strlen(line1);
136 if (n > cap - 1)
137 n = cap - 1;
138 memcpy(e.charValue3, (void*)line1, n);
139 off = n;
140 }
141 e.charValue3[off++] = '\0';
142 if (line2 && off < cap) {
143 size_t n = strlen(line2);
144 if (n > cap - off)
145 n = cap - off;
146 memcpy(e.charValue3 + off, (void*)line2, n);
147 }
148 return XelnchPost(postboxfd, &e);
149}
