4/*

6 * this scaffolding.
7 *
8 * mkh.sh created the original so clang would stop reporting a missing include
9 * and start reporting what was *used* from it, which is exactly what
10 * happened: virtio_console.c's port_fops_open() calls nonseekable_open(), and
11 * the include that resolves it upstream is <linux/fs.h> -- the one header DCL
12 * does not edit. So what this header owes is spelled here, in mainline's own
13 * home for it: mainline splits fs/open.c's file helpers out to a header
14 * rather than making every caller reach through fs.h, and DCL has one more
15 * reason than mainline does to want that split.
16 *
17 * <linux/cdev.h> includes this file, which is how a character-device driver
18 * reaches it: port_fops_open() is a .cdev open, and a driver should not have
19 * to name a second header for a helper its first one already owes it.
20 * It is safe to include anywhere -- it pulls <linux/fs.h>, <linux/fcntl.h>,
21 * <linux/capability.h>, <linux/pid.h> and nothing else, and specifically not
22 * <linux/kobject.h>, which is why the bodies live in
23 * DCL/linux_cdev_shim.c rather than somewhere that already includes cdev.h.
24 * None of the four can reach <linux/sched.h>, which is the property that
25 * matters here: see the note at the bottom of fcntl.h.

118/*
119 * struct fd and the CLASS(fd, ...) plumbing -- mainline's fdget() result
120 * type, carried here because drm_syncobj.c:710 uses it, and this header is
121 * where mainline puts it (include/linux/file.h).
122 *
123 * There is no file-descriptor table in this kernel: no __fdget(), no fget(),
124 * and nothing that maps an unsigned int to a struct file. So the constructor
125 * answers an empty fd for every input, which is not a stub pretending to
126 * work -- it is the correct answer, and drm_syncobj.c is written for it:
127 * :710 `if (fd_empty(f)) return -EINVAL;` is the first thing that function
128 * does after the declaration, so an fd-based syncobj import returns -EINVAL
129 * and touches nothing behind it.
130 *
131 * fd_destructor is empty for the same reason fd_constructor returns empty:
132 * there is no reference to drop, because there was never one to take. It is
133 * spelled by name rather than built with DEFINE_FREE(), because CLASS()'s
134 * expansion looks for `name##_destructor` specifically -- see the note on
135 * CLASS() in <linux/cleanup.h>.
136 *
137 * The mainline struct also carries `flags` (FMODE_PATH and friends), kept
138 * here so a later fdget() could be dropped in without the type changing.
139 */
140struct fd {
141 struct file* file;
142 unsigned int flags;
143};
