141 return (uint16_t)(p[0] | (p[1] << 8));

143static inline uint32_t AcLE32(const uint8_t* p) {
144 return (uint32_t)p[0] | ((uint32_t)p[1] << 8) | ((uint32_t)p[2] << 16) | ((uint32_t)p[3] << 24);

147 return ((uint32_t)(p[0] & 0x7F) << 21) | ((uint32_t)(p[1] & 0x7F) << 14) |
148 ((uint32_t)(p[2] & 0x7F) << 7) | (uint32_t)(p[3] & 0x7F);

188static bool AcReaderOpen(AcReader* r, const char* path) {
189 memset(r, 0, sizeof(*r));
190 r->buf = g_readBuf;
191 r->fd = _KeOpenFile((char*)path, FILE_OPEN_READ_ONLY);
192 if (r->fd < 0)
193 return false;
194 XEFileStatus st;
195 memset(&st, 0, sizeof(st));
196 _KeFileStat(r->fd, &st);
197 r->fileSize = (uint32_t)st.size;
198 if (r->fileSize == 0) {
199 _KeCloseFile(r->fd);
200 r->fd = -1;
201 return false;
202 }
203 return true;

207 if (r->fd >= 0)

214static inline int AcReaderAvail(const AcReader* r) {
215 return r->len - r->pos;
216}

218/* make at least `want` unread bytes available, unless the file ends first */
219static void AcReaderFill(AcReader* r, int want) {
220 if (want > AC_READER_CAP - AC_FILE_BLOCK)
221 want = AC_READER_CAP - AC_FILE_BLOCK;
222 if (AcReaderAvail(r) >= want)
223 return;
224 if (r->pos > 0) {
225 memmove(r->buf, r->buf + r->pos, (size_t)(r->len - r->pos));
226 r->bufBase += (uint32_t)r->pos;
227 r->len -= r->pos;
228 r->pos = 0;
229 }
230 while (!r->eof && AcReaderAvail(r) < want && r->len + AC_FILE_BLOCK <= AC_READER_CAP) {
231 if (r->nextBlock >= r->fileSize) {
232 r->eof = true;
233 break;
234 }
235 size_t n = _KeReadFile(r->fd, r->buf + r->len, AC_FILE_BLOCK);
236 if (n == 0 || n == (size_t)-1) {
237 r->eof = true;
238 break;
239 }
240 /* the last cluster comes back whole: trust the file size, not n */
241 uint32_t valid = r->fileSize - r->nextBlock;
242 if (valid > AC_FILE_BLOCK)
243 valid = AC_FILE_BLOCK;
244 if (n < valid)
245 valid = (uint32_t)n;
246 r->len += (int)valid;
247 r->nextBlock += AC_FILE_BLOCK;
248 if (r->nextBlock >= r->fileSize)
249 r->eof = true;
250 }
251}
252
253/* position the reader at an arbitrary byte offset */
254static bool AcReaderSeek(AcReader* r, uint32_t off) {
255 if (off > r->fileSize)
256 off = r->fileSize;
257 /* already buffered (e.g. walking WAV chunk headers): no I/O */
258 if (off >= r->bufBase && off < r->bufBase + (uint32_t)r->len) {
259 r->pos = (int)(off - r->bufBase);
260 return true;
261 }
262 uint32_t aligned = off & ~(uint32_t)(AC_FILE_BLOCK - 1);
263 if (_KeFileSetOffset(r->fd, aligned) != 0)
264 return false;
265 r->nextBlock = aligned;
266 r->bufBase = aligned;
267 r->len = r->pos = 0;
268 r->eof = (aligned >= r->fileSize);
269 AcReaderFill(r, AC_FILE_BLOCK);
270 int skip = (int)(off - aligned);
271 r->pos = skip < r->len ? skip : r->len;
272 return true;
273}

348/* copy what was just accepted by the device into the visualizer ring */
349static void AcPublishVis(const int16_t* stereo, int frames) {
350 uint32_t w = g_sh.visWrite; /* only this thread writes it */
351 for (int i = 0; i < frames; i++) {

353 g_sh.vis[(w + (uint32_t)i) & (AC_VIS_RING - 1)] = (int16_t)m;

358/*
359 * AcDeviceWrite -- hand one full block to the device. Returns false when the
360 * UI asked for something else (new track, seek, quit): the caller drops the
361 * rest of its decoded chunk.
362 */
363static bool AcDeviceWrite(AcTrack* t) {
364 size_t r = _KeWriteFile(g_snd, g_out, AC_OUT_BYTES);
365 if (r == 0 || r == (size_t)-1) {
366 /* no pacing device behind us: keep real time ourselves */
367 _KeProcessSleep((AC_OUT_FRAMES * 1000) / AC_OUT_HZ);
368 }
369 AcPublishVis(g_out, AC_OUT_FRAMES);
370 t->framesOut += AC_OUT_FRAMES;
371 AcPublishPosition(t);
372 g_outFill = 0;
373
374 _KeProcessSleep(AC_YIELD_MS);
375
376 while (AcLoad(&g_sh.paused) && !AcInterrupted())
377 _KeProcessSleep(10);
378 return !AcInterrupted();
379}

381static bool AcOutPush(AcTrack* t, const int16_t* stereo, int frames) {
382 while (frames > 0) {
383 int k = AC_OUT_FRAMES - g_outFill;
384 if (k > frames)
385 k = frames;
386 memmove(g_out + g_outFill * 2, stereo, (size_t)k * 4);
387 g_outFill += k;
388 stereo += k * 2;
389 frames -= k;
390 if (g_outFill == AC_OUT_FRAMES && !AcDeviceWrite(t))
391 return false;
392 }
393 return true;
394}

396/* source-rate stereo -> 48 kHz stereo -> device */
397static bool AcEmit(AcTrack* t, const int16_t* stereo, int frames, uint32_t hz) {
398 if (hz == AC_OUT_HZ || hz == 0)
399 return AcOutPush(t, stereo, frames);
400 if (hz != t->rsHz) {
401 AccentResamplerInitialize(&t->rs, hz, AC_OUT_HZ);
402 t->rsHz = hz;
403 }
404 while (frames > 0) {
405 int n = frames < AC_RS_CHUNK ? frames : AC_RS_CHUNK;
406 size_t out = AccentResamplerProcess(&t->rs, stereo, (size_t)n, g_rsOut);
407 if (out > AC_RS_OUT_MAX)
408 out = AC_RS_OUT_MAX;
409 if (!AcOutPush(t, g_rsOut, (int)out))
410 return false;
411 stereo += n * 2;
412 frames -= n;
413 }
414 return true;
415}

421static bool AcWavOpen(AcTrack* t) {
422 AcReader* r = &t->rd;
423 AcReaderFill(r, 12);
424 if (AcReaderAvail(r) < 12 || memcmp(r->buf, "RIFF", 4) != 0 ||
425 memcmp(r->buf + 8, "WAVE", 4) != 0)
426 return false;
427
428 bool haveFmt = false;
429 uint32_t p = 12;
430 for (int guard = 0; guard < 64 && p + 8 <= r->fileSize; guard++) {
431 if (!AcReaderSeek(r, p))
432 return false;
433 AcReaderFill(r, 8 + 40);
434 if (AcReaderAvail(r) < 8)
435 return false;
436 const uint8_t* h = r->buf + r->pos;
437 uint32_t size = AcLE32(h + 4);
438 if (memcmp(h, "fmt ", 4) == 0 && size >= 16 && AcReaderAvail(r) >= 24) {
439 int fmt = AcLE16(h + 8);
440 if (fmt == 0xFFFE && size >= 40 && AcReaderAvail(r) >= 8 + 26)
441 fmt = AcLE16(h + 8 + 24); /* WAVE_FORMAT_EXTENSIBLE sub-format */
442 t->wavFormat = fmt;
443 t->wavChannels = AcLE16(h + 10);
444 t->wavRate = AcLE32(h + 12);
445 t->wavBlockAlign = AcLE16(h + 20);
446 t->wavBits = AcLE16(h + 22);
447 haveFmt = true;
448 } else if (memcmp(h, "data", 4) == 0) {
449 if (!haveFmt)
450 return false;
451 t->dataStart = p + 8;
452 t->dataSize = size;
453 if (t->dataStart + t->dataSize > r->fileSize || size == 0xFFFFFFFF)
454 t->dataSize = r->fileSize - t->dataStart;
455 break;
456 }
457 p += 8 + size + (size & 1);
458 }
459 if (!haveFmt || !t->dataStart)
460 return false;
461
462 bool pcm = (t->wavFormat == 1 &&
463 (t->wavBits == 8 || t->wavBits == 16 || t->wavBits == 24 || t->wavBits == 32));
464 bool flt = (t->wavFormat == 3 && t->wavBits == 32);
465 if ((!pcm && !flt) || t->wavChannels < 1 || t->wavRate < 4000 || t->wavRate > 192000)
466 return false;
467 if (t->wavBlockAlign != t->wavChannels * (t->wavBits / 8))
468 t->wavBlockAlign = t->wavChannels * (t->wavBits / 8);
469
470 t->durMs =
471 (uint32_t)(((uint64_t)(t->dataSize / (uint32_t)t->wavBlockAlign) * 1000) / t->wavRate);
472 t->wavConsumed = 0;
473 return AcReaderSeek(r, t->dataStart);
474}

476static inline int16_t AcWavSample(const AcTrack* t, const uint8_t* s) {
477 switch (t->wavBits) {
478 case 8:
479 return (int16_t)(((int)s[0] - 128) * 256); /* not <<: negative shift is UB */
480 case 16:
481 return (int16_t)AcLE16(s);
482 case 24:
483 return (int16_t)AcLE16(s + 1);
484 default:
485 if (t->wavFormat == 3) {
486 float f;
487 memmove(&f, s, 4);
488 if (f > 1.0f)
489 f = 1.0f;
490 if (f < -1.0f)
491 f = -1.0f;
492 return (int16_t)(f * 32767.0f);
493 }
494 return (int16_t)AcLE16(s + 2);
495 }
496}

498static int AcWavStep(AcTrack* t) {
499 AcReader* r = &t->rd;
500 uint32_t left = t->dataSize - t->wavConsumed;
501 if (left < (uint32_t)t->wavBlockAlign)
502 return AC_STEP_END;
503 int ba = t->wavBlockAlign;
504 int maxFrames = 1024;
505 uint32_t want = (uint32_t)(maxFrames * ba);
506 if (want > left)
507 want = left;
508 AcReaderFill(r, (int)want);
509 uint32_t avail = (uint32_t)AcReaderAvail(r);
510 if (avail > left)
511 avail = left;
512 int frames = (int)(avail / (uint32_t)ba);
513 if (frames > maxFrames)
514 frames = maxFrames;
515 if (frames == 0)
516 return AC_STEP_END;
517
518 const uint8_t* src = r->buf + r->pos;
519 int bps = t->wavBits / 8;
520 bool stereo = t->wavChannels > 1;
521 for (int i = 0; i < frames; i++) {
522 const uint8_t* f = src + i * ba;
523 int16_t l = AcWavSample(t, f);
524 g_stereo[i * 2] = l;
525 g_stereo[i * 2 + 1] = stereo ? AcWavSample(t, f + bps) : l;
526 }
527 r->pos += frames * ba;
528 t->wavConsumed += (uint32_t)(frames * ba);
529 return AcEmit(t, g_stereo, frames, t->wavRate) ? AC_STEP_OK : AC_STEP_ABORT;
530}

536static bool AcMp3Open(AcTrack* t) {
537 AcReader* r = &t->rd;
538 AcReaderFill(r, 10);
539 t->dataStart = 0;
540 if (AcReaderAvail(r) >= 10 && memcmp(r->buf, "ID3", 3) == 0) {
541 t->dataStart = 10 + AcSynchsafe(r->buf + 6);
542 if (r->buf[5] & 0x10)
543 t->dataStart += 10; /* footer */
544 if (t->dataStart >= r->fileSize)
545 return false;
546 if (!AcReaderSeek(r, t->dataStart))
547 return false;
548 }
549 t->dataSize = r->fileSize - t->dataStart;
550 t->mp3Bytes = t->mp3Samples = 0;
551 t->mp3Hz = 0;
552 t->durMs = 0;
553 mp3dec_init(&g_mp3);
554 return true;
555}

557static int AcMp3Step(AcTrack* t) {
558 AcReader* r = &t->rd;
559 AcReaderFill(r, AC_MP3_WINDOW);
560 int avail = AcReaderAvail(r);
561 if (avail <= 0)
562 return AC_STEP_END;
563
564 mp3dec_frame_info_t info;
565 memset(&info, 0, sizeof(info));
566 int samples = mp3dec_decode_frame(&g_mp3, r->buf + r->pos, avail, g_pcm, &info);
567 if (info.frame_bytes == 0) {
568 if (r->eof)
569 return AC_STEP_END;
570 r->pos = r->len; /* no frame anywhere in a full window: skip it */
571 return AC_STEP_OK;
572 }
573 r->pos += info.frame_bytes;
574 if (samples <= 0)
575 return AC_STEP_OK; /* ID3/garbage frame: skipped */
576
577 if ((uint32_t)info.hz != t->mp3Hz) {
578 t->mp3Hz = (uint32_t)info.hz;
579 AcStore(&g_sh.srcHz, t->mp3Hz);
580 }
581 if (info.bitrate_kbps > 0)
582 AcStore(&g_sh.kbps, (uint32_t)info.bitrate_kbps);
583
584 /* duration = data bytes * (decoded time / decoded bytes) */
585 t->mp3Bytes += (uint64_t)info.frame_bytes;
586 t->mp3Samples += (uint64_t)samples;
587 if (t->mp3Bytes >= 16 * 1024 || r->eof) {
588 uint64_t decodedMs = (t->mp3Samples * 1000) / (t->mp3Hz ? t->mp3Hz : AC_OUT_HZ);
589 uint32_t dur = (uint32_t)(((uint64_t)t->dataSize * decodedMs) / t->mp3Bytes);
590 if (dur != t->durMs) {
591 t->durMs = dur;
592 AcStore(&g_sh.durMs, dur);
593 }
594 }
595
596 const int16_t* stereo = g_pcm;
597 if (info.channels == 1) {
598 for (int i = 0; i < samples; i++)
599 g_stereo[i * 2] = g_stereo[i * 2 + 1] = g_pcm[i];
600 stereo = g_stereo;
601 }
602 return AcEmit(t, stereo, samples, (uint32_t)info.hz) ? AC_STEP_OK : AC_STEP_ABORT;
603}

615static bool AcTrackOpen(AcTrack* t, const char* path) {
616 memset(t, 0, sizeof(*t));
617 t->rd.fd = -1;
618 if (!AcReaderOpen(&t->rd, path))
619 return false;
620
621 bool ok;
622 if (AcHasExt(path, "wav")) {
623 t->codec = AC_CODEC_WAV;

630 AcReaderClose(&t->rd);
631 return false;
632 }
633 AcResetOutput(t);

636 AcStore(&g_sh.durMs, t->durMs);
637 AcStore(&g_sh.posMs, 0);
638 AcStore(&g_sh.srcHz, t->codec == AC_CODEC_WAV ? t->wavRate : 0);
639 AcStore(&g_sh.kbps,
640 t->codec == AC_CODEC_WAV
641 ? (uint32_t)(((uint64_t)t->wavRate * (uint32_t)t->wavBlockAlign * 8) / 1000)
642 : 0);
643 return true;
644}

646static void AcTrackSeek(AcTrack* t, int permille) {
647 if (permille < 0)
648 permille = 0;
649 if (permille > 1000)
650 permille = 1000;
651 if (t->codec == AC_CODEC_WAV) {
652 uint32_t frames = t->dataSize / (uint32_t)t->wavBlockAlign;
653 uint32_t target = (uint32_t)(((uint64_t)frames * (uint32_t)permille) / 1000);
654 t->wavConsumed = target * (uint32_t)t->wavBlockAlign;
655 AcReaderSeek(&t->rd, t->dataStart + t->wavConsumed);
656 t->baseMs = (uint32_t)(((uint64_t)target * 1000) / t->wavRate);
657 } else {
658 if (!t->durMs)
659 return; /* not enough decoded yet to know where to go */
660 uint32_t off =
661 t->dataStart + (uint32_t)(((uint64_t)t->dataSize * (uint32_t)permille) / 1000);
662 AcReaderSeek(&t->rd, off);
663 mp3dec_init(&g_mp3); /* resyncs on the next frame header */
664 t->baseMs = (uint32_t)(((uint64_t)t->durMs * (uint32_t)permille) / 1000);
665 }

688/*
689 * AcAudioThread -- engine entry point (created with _KeCreateThread)
690 */
691static void AcAudioThread() {
692 AcEngineRegister();
693 AcTrack* t = &g_track;
694 bool open = false;
695
696 for (;;) {
697 if (AcLoad(&g_sh.quit))
698 break;
699
700 uint32_t seq = AcLoad(&g_sh.cmdSeq);
701 if (seq != AcLoad(&g_sh.ackSeq)) {
702 char path[AC_PATH_MAX];
703 int cmd = AcLoadI(&g_sh.cmd);
704 memmove(path, g_sh.path, sizeof(path));
705 path[AC_PATH_MAX - 1] = '\0';
706 AcStore(&g_sh.ackSeq, seq); /* the UI may now reuse g_sh.path */
707 AcStoreI(&g_sh.seekPermille, -1);
708
709 if (open) {
710 AcReaderClose(&t->rd);
711 open = false;
712 }
713 if (cmd == AC_CMD_PLAY) {
714 AcStore(&g_sh.trackSeq, seq);
715 if (AcTrackOpen(t, path)) {
716 open = true;
717 AcStore(&g_sh.state, AC_STATE_PLAYING);
718 } else {
719 AcFinishTrack(t, AC_STATE_ERROR);
720 }
721 } else {
722 AcStore(&g_sh.state, AC_STATE_IDLE);
723 }
724 continue;
725 }
726
727 if (!open) {
728 _KeProcessSleep(15);
729 continue;
730 }
731
732 int seek = __atomic_exchange_n(&g_sh.seekPermille, -1, __ATOMIC_ACQ_REL);
733 if (seek >= 0)
734 AcTrackSeek(t, seek);
735
736 if (AcLoad(&g_sh.paused)) {
737 _KeProcessSleep(10);
738 continue;
739 }
740
741 int step = (t->codec == AC_CODEC_WAV) ? AcWavStep(t) : AcMp3Step(t);
742 if (step == AC_STEP_END) {
743 AcFlushTail(t);
744 AcFinishTrack(t, AC_STATE_ENDED);
745 open = false;
746 }
747 }
748
749 if (open)
750 AcReaderClose(&t->rd);
751 AcEngineUnregister();
752 AcStore(&g_sh.done, 1);
753 for (;;)
754 _KeProcessSleep(1000); /* parked until the process exits */
755}
