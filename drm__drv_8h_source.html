33/*
34 * <linux/capability.h> for capable(), which the drm sources reach through
35 * this header without naming it: drm_framebuffer.c:557 and :635 write
36 * `capable(CAP_SYS_ADMIN)` and include neither capability.h nor pci.h -- and
37 * pci.h is the carrier for the other two consumers, drm_file.c:149 and
38 * drm_ioctl.c:152.
39 *
40 * In mainline the chain is longer and automatic: this header includes
41 * <linux/fs.h>, and mainline's fs.h includes <linux/capability.h>. DCL's
42 * fs.h is a pinned ABI blob that includes nothing, so the chain has no last
43 * link and this header names the source itself. mainline's drm_drv.h does
44 * include <linux/fs.h> (for struct file's role in drm_driver), which is why
45 * this is the faithful place for it rather than a bolt-on.
46 *
47 * Leaf: capability.h includes only <stdbool.h>.
48 */
49#include <linux/capability.h>

547static inline bool drm_core_check_all_features(const struct drm_device *dev,
548 u32 features)
549{
550 u32 supported = dev->driver->driver_features & dev->driver_features;
551
552 return features && (supported & features) == features;
553}

565static inline bool drm_core_check_feature(const struct drm_device *dev,
566 enum drm_driver_feature feature)
567{
568 return drm_core_check_all_features(dev, feature);

579static inline bool drm_drv_uses_atomic_modeset(struct drm_device *dev)
580{
581 return drm_core_check_feature(dev, DRIVER_ATOMIC) ||
582 (dev->mode_config.funcs && dev->mode_config.funcs->atomic_commit != NULL);
