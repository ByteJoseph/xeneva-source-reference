110static inline uint32_t ns_fpclassify_pattern(double x) {
111 union ns_qjs_double_bits b;
112 b.d = x;
113 uint32_t hi = (uint32_t)(b.i >> 32);
114 uint32_t mant = hi & 0x000fffffu;
115 uint32_t expn = hi >> 20;
116 if (expn == 0)
117 return (mant == 0) ? 0 /* zero */ : 3 /* sub */;
118 if (expn == 0x7ff)
119 return (mant == 0) ? 1 /* inf */ : 4 /* nan */;
120 return 2 /* normal */;
121}
