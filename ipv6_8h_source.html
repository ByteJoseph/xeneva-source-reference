121static inline int ip6_prefix_equal(const ip6_addr* a, const ip6_addr* b, uint8_t prefixLen) {
122 int full;
123 int rem;
124 uint8_t mask;
125 int i;
126
127 if (prefixLen > 128)
128 prefixLen = 128;
129 full = prefixLen / 8;
130 rem = prefixLen % 8;
131 for (i = 0; i < full; i++) {
132 if (a->s6_addr[i] != b->s6_addr[i])
133 return 0;
134 }
135 if (rem) {
