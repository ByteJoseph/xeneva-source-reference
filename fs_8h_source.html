161/*
162 * Two DCL-only members are carved out of the padding this struct declares
163 * above, rather than appended after it, and the three things that make this
164 * struct an ABI pin all still hold afterwards: kobj_name at 0, devt at 708,
165 * sizeof 800. Appending would have kept the first two and silently broken
166 * the third -- the module embeds `struct device` inside its own objects and
167 * was compiled with a 800-byte one, so an 816-byte DCL would have DCL's
168 * writes landing past the end of the module's allocation.
169 *
170 * The offsets are mainline's, not chosen: `struct device` opens with `struct
171 * kobject kobj`, which the struct cdev above measures at 64 bytes, so
172 * `parent` is the next member at 64 -- the same slot the module would read
173 * if it ever read one. `of_node` sits at 696, the last 8-byte-aligned hole
174 * before the devt pin, with 704..707 left as `_pad` so devt does not move.
175 * Both are read-only at their two call sites (serial_core.c:3093 and :3236),
176 * and DCL's own devices come out of device_create()'s memset, so they read
177 * zero rather than whatever the module left in those bytes.
178 *
179 * The three _Static_asserts below exist so that none of this has to be
180 * re-argued: if a future edit moves devt or grows the struct, the build says
181 * so at this line instead of at boot, when the failure would be a corrupt
182 * device table and no message to explain it.
183 */
184struct device {
185 const char* kobj_name; /* 0 (kobject.name) */
186 unsigned char _head0[56]; /* 8..63 */
187 struct device* parent; /* 64 (after kobj, as in mainline) */
188 unsigned char _mid[624]; /* 72..695 */
189 struct device_node* of_node; /* 696 */
190 unsigned char _pad[4]; /* 700..707 */
191 unsigned int devt; /* 708 */
192 unsigned char _tail[88]; /* 712..799 */
193};
