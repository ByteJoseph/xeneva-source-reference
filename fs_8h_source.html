159/*
160 * linux-7.2.6 struct inode: i_rdev 76, i_cdev 528, size 560.
161 * The module reads only i_cdev (port_fops_open resolves its port from
162 * inode->i_cdev->dev); i_rdev is kept for the bridge's own bookkeeping.
163 *
164 * i_mapping is the second deviation from the "pinned, do not edit" rule for
165 * this header, and it is the same shape as the first: a member the drm core
166 * cannot work without, taken out of padding rather than added at the end.
167 *
168 * Four sites read it -- drm_file.c:386 and :1074, drm_gem.c:747 and
169 * drm_prime.c:251 -- all of them `...->i_mapping` on a struct inode *, which
170 * is the one form a macro cannot fix: the token is to the right of `->`, and
171 * a member that does not exist cannot be named, aliased or discarded from the
172 * outside. Every consumer is an address-and-store (`filp->f_mapping =
173 * dev->anon_inode->i_mapping`), so the object it points at is never touched
174 * through this header -- an opaque pointer is enough.
175 *
176 * It is placed at offset 80, the first byte of what used to be _mid[448],
177 * and _mid is shortened to 440 to keep every other offset where it was:
178 * i_rdev still at 76, i_cdev still at 528, sizeof still 560. Nothing is
179 * persisted, but the ABI these offsets describe belongs to code this header
180 * does not own, and a blob whose padding moved would be a blob whose one
181 * readable member landed somewhere else.
182 */
183struct inode {
184 unsigned char _head[76]; /* 0..75 */
185 unsigned int i_rdev; /* 76 */
186 struct address_space* i_mapping; /* 80 */
187 unsigned char _mid[440]; /* 88..527 */
188 struct cdev* i_cdev; /* 528 */
189 unsigned char _tail[24]; /* 536..559 */
190};

232/*
233 * Two DCL-only members are carved out of the padding this struct declares
234 * above, rather than appended after it, and the three things that make this
235 * struct an ABI pin all still hold afterwards: kobj_name at 0, devt at 708,
236 * sizeof 800. Appending would have kept the first two and silently broken
237 * the third -- the module embeds `struct device` inside its own objects and
238 * was compiled with a 800-byte one, so an 816-byte DCL would have DCL's
239 * writes landing past the end of the module's allocation.
240 *
241 * The offsets are mainline's, not chosen: `struct device` opens with `struct
242 * kobject kobj`, which the struct cdev above measures at 64 bytes, so
243 * `parent` is the next member at 64 -- the same slot the module would read
244 * if it ever read one. `of_node` sits at 696, the last 8-byte-aligned hole
245 * before the devt pin, with 704..707 left as `_pad` so devt does not move.
246 * Both are read-only at their two call sites (serial_core.c:3093 and :3236),
247 * and DCL's own devices come out of device_create()'s memset, so they read
248 * zero rather than whatever the module left in those bytes.
249 *
250 * The three _Static_asserts below exist so that none of this has to be
251 * re-argued: if a future edit moves devt or grows the struct, the build says
252 * so at this line instead of at boot, when the failure would be a corrupt
253 * device table and no message to explain it.
254 */
255struct device {
256 union {
257 const char* kobj_name; /* 0 (kobject.name) */
258 unsigned char kobj[64]; /* 0..63, same as struct kobject */
259 };
260 struct device* parent; /* 64 (after kobj, as in mainline) */
261 const struct class* class; /* 72 */
262 const struct device_type* type; /* 80 */
263 void* driver_data; /* 88 */
264 void (*release)(struct device*); /* 96 */
265 const struct attribute_group** groups; /* 104 */
266 unsigned char _mid[584]; /* 112..695 */
267 struct device_node* of_node; /* 696 */
268 unsigned char _pad[4]; /* 700..707 */
269 unsigned int devt; /* 708 */
270 unsigned char _tail[88]; /* 712..799 */
271};
