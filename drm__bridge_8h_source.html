1363static inline struct drm_bridge_state *
1364drm_bridge_get_current_state(struct drm_bridge *bridge)
1365{
1366 if (!bridge)
1367 return NULL;
1368
1369 /*
1370 * Only atomic bridges will have bridge->base initialized by
1371 * drm_atomic_private_obj_init(), so we need to make sure we're
1372 * working with one before we try to use the lock.
1373 */
1374 if (!bridge->funcs || !bridge->funcs->atomic_reset)
1375 return NULL;
1376
1377 drm_modeset_lock_assert_held(&bridge->base.lock);
1378
1379 if (!bridge->base.state)
1380 return NULL;
1381
1382 return drm_priv_to_bridge_state(bridge->base.state);
1383}
