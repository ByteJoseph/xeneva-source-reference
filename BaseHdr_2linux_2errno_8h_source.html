4/*
5 * DCL <linux/errno.h> -- the numbers a ported file returns.
6 *
7 * <linux/kernel.h> owns the bulk of the set (EPERM through the socket
8 * errnos), and it is kept as the owner rather than split across two headers
9 * that could drift apart. Everything below is here because a file in the
10 * staged tty/serial set asks for it and kernel.h did not have it:
11 *
12 * EBADF / EINTR / ENOTTY -- tty_io.c's read/write/ioctl error paths
13 * ENOIOCTLCMD -- the tty layer's "this ldisc does not do
14 * ioctls" answer (mainline: 515)
15 * EINPROGRESS -- n_tty's poll path
16 * EFAULT / EACCES / EPIPE ...-- the usual open/write failures
17 *
18 * All guarded, so <linux/mm.h> (which also defines EFAULT and EFBIG) and
19 * kernel.h can be included in any order without a redefinition.
20 *
21 * ERESTARTSYS lives in kernel.h's errno block -- 512, matching mm.h -- because
22 * serial_core.c:1144's scoped_cond_guard() needs it from <linux/mutex.h>
23 * alone.
24 *
25 * The IS_ERR/ERR_PTR family is *not* here; it is in <linux/err.h>, because
26 * mainline splits them that way and a ported file includes <linux/err.h> by
27 * name.
28 */
