4/*
5 * DCL <linux/idr.h> -- struct ida and the small allocation API.
6 *
7 * virtio-rng.c uses exactly two things from it:
8 *
9 * static DEFINE_IDA(rng_index_ida);
10 * vi->index = ida_alloc(&rng_index_ida, GFP_KERNEL);
11 * ...
12 * ida_free(&rng_index_ida, index);
13 *
14 * the index being used to name the device ("virtio-<n>"). mainline gets to
15 * this header by accident of its include graph -- there is no idr.h among the
16 * driver's own includes. The chain, confirmed against the header package:
17 *
18 * linux/virtio.h -> linux/device.h -> linux/kobject.h
19 * -> linux/sysfs.h -> linux/kernfs.h -> linux/idr.h
20 *
21 * DCL has no device.h (that chain is why <linux/virtio.h> here also pulls in
22 * completion.h and pm.h), so virtio.h includes this file directly instead of
23 * walking five headers to reach the same place. The driver's own #include
24 * list is therefore untouched, which is what has to stay true.
25 *
26 * ### struct ida keeps its own counter
27 *
28 * mainline's is `struct ida { struct xarray xa; }` -- a radix tree of ID
29 * bitmaps, because a kernel hands out millions of them. DCL's
30 * ida_alloc_range() (linux_kmod_shim.c) is a monotone counter, and the
31 * counter lives in this struct so that every ida numbers on its own. Three
32 * exist now and their numbering must not mix: virtio-rng's device index,
33 * virtio-console's vtermno (min 1), and drm's connector ids -- the
34 * per-config connector_ida, whose upper bound is 31, and the per-type idas
35 * in drm_connector_enum_list. One static counter in the shim served them all
36 * and would eventually have handed connector_ida an id past its own bound;
37 * the counter moved in here when the second user showed up.
38 *
39 * Ids are never reused -- mainline's reuses freed ones -- which costs
40 * nothing at DCL's scale: an ida here numbers devices or connectors, a
41 * handful per boot.
42 */

50/*
51 * struct idr -- DCL's is a fixed slot table; mainline's is a radix tree.
52 *
53 * The drm core hangs three of these off a device (mode_config object/tile
54 * ids, GEM object names) and one off every open file (GEM handles), and all
55 * it asks of them is "give me an int in [base, base+N) and give it back as a
56 * pointer". A 64-slot array answers that directly; what it does not answer
57 * is the two things mainline's tree buys -- ids beyond 64 at once, and
58 * lock-free lookup. Both are noted where they would bite:
59 * idr_alloc() (DCL/linux_idr_shim.c) returns -ENOSPC rather than growing, and
60 * rcu_read_lock() is a no-op, so the "lockless idr_find" mainline documents
61 * is simply single-threaded here like every other DCL lock.
62 *
63 * A slot answers two questions -- is the id taken, and what does it hold --
64 * and mainline keeps them apart: idr_alloc() may reserve an id and store
65 * NULL in it, which drm leans on. A connector's id is reserved as a NULL
66 * entry by __drm_mode_object_add() (register_obj false) and filled in later
67 * by drm_mode_object_register() through idr_replace() at connector
68 * registration; a GEM handle is reserved the same way in
69 * drm_gem_handle_create_tail(). A table of pointers alone cannot tell a
70 * reserved NULL entry from a free slot -- it would hand the id out a second
71 * time, and idr_replace() would refuse to fill the reservation -- so `used`
72 * carries the state and `slots` the value.
73 *
74 * idr_init_base()/idr_init()/idr_is_empty() are static inline over these
75 * fields, mirroring mainline (idr.h:152-181), so a `struct idr` member in
76 * drm_file.h or drm_mode_config.h is complete without a shim call.
77 */
78#define DCL_IDR_SLOTS 64
79
80struct idr {
81 void *slots[DCL_IDR_SLOTS]; /* slot i holds id base + i */
82 unsigned char used[DCL_IDR_SLOTS]; /* slot i's id has been reserved */
83 unsigned int base; /* first id this idr may hand out */
84};
