87static inline css_code_t buildOPV(opcode_t opcode, uint8_t flags, uint16_t value)
88{
89 return (opcode & 0x3ff) | (flags << 10) | ((value & 0x3fff) << 18);
90}

107static inline bool isImportant(css_code_t OPV)
