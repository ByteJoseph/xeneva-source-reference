4/*
5 * DCL <linux/types.h> -- the fixed-width typedefs a ported file starts with.
6 *
7 * Everything mainline puts here already exists in <linux/kernel.h> (u8 through
8 * u64, s8 through s64, __u8 through __u64, phys_addr_t, gfp_t), so this header
9 * is a re-export: it exists because a ported file opens with
10 * `#include <linux/types.h>` and the include has to resolve.
11 *
12 * It also pulls in <stdbool.h> for `bool`/`true`/`false`, which the tty_buffer
13 * struct fields (`bool flags`) and every `if (likely(...))` in the set need.
14 * Xeneva's stdbool.h defines `bool` as _Bool (stdbool.h:35).
15 *
16 * Deliberately does *not* re-typedef anything: two typedefs of the same name
17 * are only legal in C11 when they are identical, and "identical" here would
18 * mean byte-for-byte matching an expression from a header we do not own.
19 */
20
21#include <linux/kernel.h>
22#include <stdbool.h>

24/*
25 * DCL delta -- byte order.
26 *
27 * mainline reaches le16_to_cpu() and friends through its byteorder layer
28 * (<linux/byteorder/generic.h> via <asm/byteorder.h>), which comes in with the
29 * rest of the driver's own include set. DCL has no byteorder layer -- nothing
30 * needed one before this port -- so the twelve conversions are defined here,
31 * which is the first header <linux/virtio_byteorder.h> includes (line 6).
32 *
33 * __LITTLE_ENDIAN is the one that is not merely convenience: clang does *not*
34 * predefined it for aarch64-unknown-windows (only __BYTE_ORDER__ comes from
35 * the compiler, verified), and virtio_byteorder.h:9 tests __LITTLE_ENDIAN to
36 * decide whether the legacy virtio device layout is little-endian. Taken at
37 * face value that reads "big-endian", which would run every virtio16/virtio32
38 * field through bswap on a little-endian target and hand the device garbage.
39 * The 1234/4321 values match mainline's.
40 *
41 * aarch64 is little-endian, so the le_* forms are identity and the be_* forms
42 * bswap -- which is exactly mainline's little_endian.h, written out rather
43 * than fetched, because pulling in its arch indirection would reintroduce the
44 * <asm/byteorder.h> hop this tree has no file for.
45 */
46#define __LITTLE_ENDIAN 1234
47#define __BIG_ENDIAN 4321
48#define __PDP_ENDIAN 3412
