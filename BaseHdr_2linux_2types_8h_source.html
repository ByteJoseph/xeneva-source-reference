4/*
5 * DCL <linux/types.h> -- the fixed-width typedefs a ported file starts with.
6 *
7 * Everything mainline puts here already exists in <linux/kernel.h> (u8 through
8 * u64, s8 through s64, __u8 through __u64, phys_addr_t, gfp_t), so this header
9 * is a re-export: it exists because a ported file opens with
10 * `#include <linux/types.h>` and the include has to resolve.
11 *
12 * It also pulls in <stdbool.h> for `bool`/`true`/`false`, which the tty_buffer
13 * struct fields (`bool flags`) and every `if (likely(...))` in the set need.
14 * Xeneva's stdbool.h defines `bool` as _Bool (stdbool.h:35).
15 *
16 * Deliberately does *not* re-typedef anything: two typedefs of the same name
17 * are only legal in C11 when they are identical, and "identical" here would
18 * mean byte-for-byte matching an expression from a header we do not own.
19 */

21/*
22 * __kernel_long_t / __kernel_ulong_t -- <uapi/linux/sysinfo.h>'s struct
23 * sysinfo hangs every field on them, and that header arrives through
24 * uapi/linux/kernel.h:5, which linux/math.h:7 includes, which kernel.h:90
25 * includes.
26 *
27 * They sit *above* the include of kernel.h, and that placement is the whole
28 * point: when the chain above runs, sysinfo.h includes <linux/types.h> from
29 * inside kernel.h -- by then this header's guard is already set, so the
30 * re-entry is skipped and anything defined *below* the kernel.h include does
31 * not exist yet. The first build with the kernel.h -> math.h edge failed at
32 * sysinfo.h:9 with "unknown type name '__kernel_long_t'" for exactly that
33 * reason. These two are self-contained -- no size_t, nothing from kernel.h --
34 * so they can go first; the rest of the file cannot.
35 *
36 * The fixed-width spelling rather than mainline's `long`/`unsigned long` is
37 * about the target: aarch64-unknown-windows is LLP64, so `long` is 32 bits
38 * here where Linux/arm64's is 64 (verified: sizeof(long) == 4). Following
39 * mainline's spelling would silently shrink struct sysinfo from 104 bytes to
40 * 64 and turn the `_f[20 - 2*sizeof(__kernel_ulong_t) - sizeof(__u32)]`
41 * trailer into 8 bytes of padding where Linux has none. Nothing in DCL
42 * implements sysinfo() or ships the struct over an ABI boundary today, but a
43 * uapi struct whose size nobody has checked is exactly the thing that gets
44 * found out by the first syscall that does.
45 */
46typedef long long __kernel_long_t;
47typedef unsigned long long __kernel_ulong_t;

94/*
95 * DCL delta -- byte order.
96 *
97 * mainline reaches le16_to_cpu() and friends through its byteorder layer
98 * (<linux/byteorder/generic.h> via <asm/byteorder.h>), which comes in with the
99 * rest of the driver's own include set. DCL has no byteorder layer -- nothing
100 * needed one before this port -- so the twelve conversions are defined here,
101 * which is the first header <linux/virtio_byteorder.h> includes (line 6).
102 *
103 * __LITTLE_ENDIAN is the one that is not merely convenience: clang does *not*
104 * predefined it for aarch64-unknown-windows (only __BYTE_ORDER__ comes from
105 * the compiler, verified), and virtio_byteorder.h:9 tests __LITTLE_ENDIAN to
106 * decide whether the legacy virtio device layout is little-endian. Taken at
107 * face value that reads "big-endian", which would run every virtio16/virtio32
108 * field through bswap on a little-endian target and hand the device garbage.
109 * The 1234 matches mainline's.
110 *
111 * aarch64 is little-endian, so the le_* forms are identity and the be_* forms
112 * bswap -- which is exactly mainline's little_endian.h, written out rather
113 * than fetched, because pulling in its arch indirection would reintroduce the
114 * <asm/byteorder.h> hop this tree has no file for.
115 *
116 * __BIG_ENDIAN is deliberately absent, and that is not a sibling omission of
117 * the paragraph above. mainline defines exactly one of the pair -- <linux/
118 * kconfig.h> chooses from CONFIG_CPU_BIG_ENDIAN -- and four vendored files
119 * read *presence* as the endianness test: drm_fourcc.h:42 (the DRM_FORMAT_HOST_*
120 * aliases), drm_fourcc.c:221, drm_framebuffer.c:357 and seq_buf.c:278. Both
121 * macros published means all four take the big-endian arm here, so every
122 * scanout format gets stamped with DRM_FORMAT_BIG_ENDIAN and the seq_buf hex
123 * dumps come out byte-reversed -- precisely the failure the Stage 3
124 * writeback pixel-compare is there to catch, but a month earlier.
125 */
126#define __LITTLE_ENDIAN 1234
127#define __PDP_ENDIAN 3412
