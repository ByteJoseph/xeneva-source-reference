27/*
28 * struct seq_file -- complete, where this header used to carry only the
29 * forward declaration.
30 *
31 * virtio_console.c:1264 is the first ported source that reads a member of
32 * one (`struct port *port = s->private;`), and a forward declaration makes
33 * that "incomplete definition of type 'struct seq_file'". mainline's struct
34 * is the whole /proc iterator state -- buf, size, from, count, version,
35 * index, op, lock, private -- and none of the rest has a referent here: DCL
36 * has no procfs (the header's own note above), seq_printf() takes `const
37 * void*` so no member of this type is read from any header, and the only
38 * member a ported source names is `private`, which is what single_open()
39 * writes the show callback's data into.
40 *
41 * So the members below are the ones a *reader* of this header could
42 * plausibly name, in mainline's spelling for the ones that are kept, and the
43 * layout is deliberately not mainline's: nothing in this tree passes a
44 * seq_file across a boundary that already has an opinion about its size.

54/*
55 * struct seq_operations -- the four callbacks a /proc reader drives. Added
56 * because tty_ldisc.c:215 defines `const struct seq_operations

58 *
59 * Field order and types are mainline's; the one thing worth naming is that

61 * iterator hands a `loff_t*` back as an opaque token -- tty_ldiscs_seq_show()

80/*
81 * single_open() / single_release() -- the "one screen, no iteration" pair.
82 *
83 * They are here rather than in <linux/fs.h> for the same reason every other
84 * declaration in this port is somewhere other than <linux/fs.h>: that header
85 * is pinned by static asserts and does not move.
86 *
87 * mainline's single_open() is `m->private = data; m->op = &single_operations;`
88 * and returns single_start() so the first read calls show() once. DCL's body
89 * (DCL/linux_cdev_shim.c) records data and reports "one record", which is the
90 * same contract as far as a stub that never serves a read needs to keep.
91 *
92 * The show callback type is mainline's: `int (*)(struct seq_file *, void *)`,
93 * which is what DEFINE_SHOW_ATTRIBUTE() below passes it.

99/*
100 * DEFINE_SHOW_ATTRIBUTE(__name) -- mainline's, minus the inode.
101 *
102 * mainline expands to `<name>_open()` followed by `<name>_fops`, and the open
103 * arm is
104 *
105 * return single_open(file, __name##_show, inode->i_private);
106 *
107 * where i_private is what debugfs_create_file()'s `data` argument was stored
108 * in. DCL's struct inode (fs.h:128) is a pinned blob with no i_private to
109 * read -- the field is inside `_head[76]` -- so the data is passed as NULL
110 * instead, and that is not a live difference: DCL's debugfs_create_file() is
111 * a stub (DCL/linux_cdev_shim.c) that installs no file and opens no inode, so
112 * there is no path by which this fops table is ever entered and no
113 * show callback that would dereference a NULL port.
114 *
115 * The five members of the fops table are the ones mainline sets. `owner`,
116 * `read` and `llseek` are assigned across types that differ from mainline's
117 * (fs.h types llseek as `void*`, and THIS_MODULE is `(void*)0`) -- accepted
118 * pointer warnings, not errors, and the same shape mem.c's fops already
119 * produce.
120 */
121#define DEFINE_SHOW_ATTRIBUTE(__name) \
122 static int __name##_open(struct inode* inode, struct file* file) \
123 { \
124 (void)inode; \
125 return single_open(file, __name##_show, NULL); \
126 } \
127 static const struct file_operations __name##_fops = { \
128 .owner = THIS_MODULE, \
129 .open = __name##_open, \
130 .read = seq_read, \
131 .llseek = seq_lseek, \
132 .release = single_release, \
133 }
