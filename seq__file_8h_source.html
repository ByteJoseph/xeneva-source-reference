100/*
101 * DEFINE_SHOW_ATTRIBUTE(__name) -- mainline's, minus the inode.
102 *
103 * mainline expands to `<name>_open()` followed by `<name>_fops`, and the open
104 * arm is
105 *
106 * return single_open(file, __name##_show, inode->i_private);
107 *
108 * where i_private is what debugfs_create_file()'s `data` argument was stored
109 * in. DCL's struct inode (fs.h:128) is a pinned blob with no i_private to
110 * read -- the field is inside `_head[76]` -- so the data is passed as NULL
111 * instead, and that is not a live difference: DCL's debugfs_create_file() is
112 * a stub (DCL/linux_cdev_shim.c) that installs no file and opens no inode, so
113 * there is no path by which this fops table is ever entered and no
114 * show callback that would dereference a NULL port.
115 *
116 * The five members of the fops table are the ones mainline sets. `owner`,
117 * `read` and `llseek` are assigned across types that differ from mainline's

119 * pointer warnings, not errors, and the same shape mem.c's fops already
120 * produce.

136/*
137 * seq_write(seq, data, len) -- append a block to a seq_file's buffer.
138 *
139 * mainline's body from fs/seq_file.c:824, verbatim except for one line: the
140 * overflow arm spells `seq_set_overflow(seq)`, which is mainline's own inline
141 * for `m->count = m->size` (include/linux/seq_file.h). Writing the member
142 * directly is the same store with no second name to carry for it, and DCL's
143 * struct seq_file above has exactly the two members that line touches.
144 *
145 * Callers are lib/seq_buf.c:54 and drm_debugfs.c's edid dump at :2272, and
146 * both test the return -- 0 for written, non-zero for "did not fit" -- so the
147 * two answers have to stay distinct rather than both being 0.
148 */
149static inline int seq_write(struct seq_file* seq, const void* data, size_t len)
150{
151 if (seq->count + len < seq->size) {
152 memcpy(seq->buf + seq->count, data, len);
153 seq->count += len;
154 return 0;
155 }
156 seq->count = seq->size; /* mainline's seq_set_overflow() */
157 return -1;
158}

160/*
161 * mangle_path(s, p, esc) -- write the pathname `p` into buffer `s`, escaping
162 * every character found in the set `esc` as a backslash and three octal
163 * digits, and return where the output ended (or NULL if it would not fit).
164 *
165 * mainline's body from fs/seq_file.c:441, and the part worth reading twice is
166 * that this is an *in-place* transform: seq_buf.c:324 calls
167 * `mangle_path(buf, p, esc)` where `p` is d_path()'s return, which points
168 * into that same buffer. So `s` trails `p` as it writes, and the two bounds
169 * are what stop it running off the end: `s <= p` says the write head has not
170 * passed the read head, and `s + 4 > p` says there is room for a full
171 * four-character escape before it would. Neither is a redundant check -- the
172 * first is only meaningful because the two pointers share a buffer, and the
173 * second is what makes an escape that would overwrite an unescaped byte
174 * behind it come back NULL instead of corrupting the path.
175 *
176 * Escaping in octal rather than doubling the backslash is mainline's choice,
177 * kept because the output is read by the same /proc consumers upstream has.
178 */
179static inline char* mangle_path(char* s, const char* p, const char* esc)
180{
181 while (s <= p) {
182 char c = *p++;
183
184 if (!c) {
185 return s;
186 } else if (!strchr(esc, c)) {
187 *s++ = c;
188 } else if (s + 4 > p) {
189 break;
190 } else {
191 *s++ = '\\';
192 *s++ = (char)('0' + ((c & 0300) >> 6));
193 *s++ = (char)('0' + ((c & 070) >> 3));
194 *s++ = (char)('0' + (c & 07));
195 }
196 }
197 return NULL;
198}
