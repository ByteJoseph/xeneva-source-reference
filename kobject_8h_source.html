211/*
212 * kobject_uevent() -- a macro, deliberately.
213 *
214 * mainline has a function here, and DCL's DCL/linux_cdev_shim.c:151 still
215 * defines one (the no-op body a udev notification would need). It cannot stay
216 * a *declaration*, because virtio_console.c:1654 calls it as
217 *
218 * kobject_uevent(&port->dev->kobj, KOBJ_CHANGE);
219 *
220 * and `port->dev->kobj` is a member that does not exist: <linux/fs.h>'s
221 * struct device -- the one file this tree must not edit -- carries `kobj_name`
222 * at offset 0 (the slot mainline's kobj.name occupies) and no `kobj`. A macro
223 * whose parameters never appear in its replacement list never parses them, so
224 * the whole invocation is replaced before the parser sees `->kobj`. Same
225 * idiom as spin_lock(l) in <linux/spinlock.h>:36.
226 *
227 * The two definitions have to agree about not existing: this replaces the
228 * declaration rather than being added beside it, because a declaration that
229 * survives would still type-check `&port->dev->kobj` in any translation unit
230 * that happened to include this header before <linux/sysfs.h>. And the
231 * behaviour is unchanged -- the shim's body is empty. DCL has no udev, so the
232 * event had nowhere to go either way.
233 *
234 * kobject_uevent_env() joined it when drm_drv.c:599
235 *
236 * return kobject_uevent_env(&dev->primary->kdev->kobj, KOBJ_CHANGE, envp);
237 *
238 * became the first caller: same absent member, and this one uses the
239 * return value, so the macro answers (0) -- "event delivered", the answer
240 * that keeps drm_dev_wedged_event() on its success path.
241 */
242#define kobject_uevent(kobj, action) ((void)0)
243#define kobject_uevent_env(kobj, action, envp) (0)
244int kobject_synth_uevent(struct kobject *kobj, const char *buf, size_t count);
