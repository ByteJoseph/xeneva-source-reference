18/*
19 * Address-space / provenance annotations mainline hangs on declarations.
20 *
21 * They are empty here, and that is the entire job: on a mainline build they
22 * carry a pointer's provenance (user, kernel, MMIO, sparse's bitwise) through
23 * a checker DCL does not run -- there is no sparse in this toolchain, only
24 * -Wall -Wextra over a freestanding aarch64 target.
25 *
26 * What is not optional is that the *names* exist. `typedef u64 __bitwise
27 * upf_t;` with no __bitwise defined turns the annotation into an ordinary
28 * identifier, and clang parses it as the typedef's NAME: __bitwise comes out
29 * as the declared type, upf_t is left stray, and the next line's
30 * `typedef unsigned int __bitwise upstat_t;` becomes a redefinition of
31 * __bitwise. The diagnostic then names upstat_t and points at the second
32 * line, so the fault reads as "upstat_t conflicts with upf_t" when the real
33 * problem is a missing macro six tokens earlier. That is a very cheap way to
34 * lose an hour, hence the note rather than a bare #define.
35 *
36 * Only the annotations the tty/serial headers actually reference are defined,
37 * not mainline's full set -- <linux/cleanup.h> records the same preference.
38 */
39#ifndef __bitwise
40#define __bitwise
41#endif
42#ifndef __force
43#define __force
44#endif
45#ifndef __user
46#define __user
47#endif
48#ifndef __kernel
49#define __kernel
50#endif
51#ifndef __iomem
52#define __iomem
53#endif

56/*
57 * fallthrough -- mainline's spelling in linux/compiler_attributes.h, where it
58 * is a statement because that is the whole vocabulary: the source says
59 * `fallthrough;` as its own statement (8250_port.c:1779), and on a build
60 * with -Wimplicit-fallthrough the macro is what turns that into an explicit
61 * annotation instead of a warning. Expanding to an empty do-while is the
62 * correct body -- there is no attribute here to attach, because DCL does not
63 * enable the warning either -- and an expression-less *statement* is required,
64 * not an expression: `do {} while (0)` is the one that both parses where the
65 * source put it and disappears.
66 */
67#define fallthrough do {} while (0)

70/*
71 * __must_check -- and it is deliberately not the same kind of thing as the
72 * six annotations above it.
73 *
74 * __bitwise/__force/__user/__kernel/__iomem are sparse annotations: they
75 * carry meaning for a checker that is not running, and clang has no spelling
76 * for them, so empty is the truthful definition. __must_check is different
77 * -- clang does act on it (warn_unused_result), and defining it empty here
78 * would make a promise the source is making and the compiler is no longer
79 * keeping. The first user is Vendored/drivers/tty/tty.h:100:
80 *
81 * int __must_check tty_ldisc_init(struct tty_struct *tty);
82 *
83 * i.e. "this returns an error code and the caller must look at it". A port
84 * that dropped the annotation would compile exactly the same and let a failed
85 * ldisc attach read as a successful one, which is the specific bug the
86 * annotation exists to prevent.
87 *
88 * mainline: linux/compiler_attributes.h.
89 */
90#ifndef __must_check
91#define __must_check __attribute__((warn_unused_result))
92#endif

94/*
95 * __sched -- "put this in the scheduler section". mainline's is a section
96 * attribute (linux/compiler_attributes.h); DCL links one flat image with no
97 * such section, so it is empty -- but it has to *be* something, because it is
98 * spelled as a declaration-level annotation in front of a pointer:
99 *
100 * static struct ld_semaphore __sched *
101 * down_read_failed(...)
102 *
103 * (tty_ldsem.c:155 and :229). Left undefined, that line parses as a
104 * declaration of an object named __sched and the real function never gets
105 * named, which is why both showed up in the implicit-call list rather than as
106 * "unknown type". Empty is also correct: the annotation describes placement,
107 * and there is no placement to control.
108 */
109#define __sched

111/*
112 * __printf(fmt_idx, arg_idx) -- format-string checking for varargs
113 * declarations. mainline: linux/compiler_attributes.h. kobject.h decorates
114 * every varargs entry point with it (kobject_set_name, kobject_add,
115 * kobject_init_and_add, add_uevent_var), so it has to parse as a
116 * declaration-level annotation in front of a return type -- left undefined,
117 * "the line parses as a declaration of an object named __printf" and the
118 * function it belongs to is never declared, exactly like __sched above.
119 */
120#ifndef __printf
121#define __printf(fmt_idx, arg_idx) __attribute__((__format__(__printf__, fmt_idx, arg_idx)))
122#endif
