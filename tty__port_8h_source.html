4/*
5 * DCL <linux/tty_port.h> -- port-level state shared by every tty driver.
6 *
7 * Ported from mainline v7.2. A tty_port outlives the tty that opens it: the
8 * port is the device, the tty is one open of it, which is why the two have
9 * separate reference counts (port->kref for the port, tty->kref for the tty)
10 * and why port->tty can go NULL on a hangup while the port stays.
11 *
12 * The parts that matter most to the 8250 port:
13 *
14 * buf -- the flip buffer head. uart_handle_rx_char() writes into it
15 * from interrupt context; tty_flip_buffer_push() commits.
16 * xmit_buf / xmit_fifo
17 * -- the transmit ring. serial_core.c allocates xmit_buf on
18 * demand (serial_core.c:262) and binds it to xmit_fifo with
19 * kfifo_init(); uart_write() pushes, the driver's start_tx
20 * drains, and WAKEUP_CHARS is the threshold for telling the
21 * tty there is room again.
22 * iflags -- TTY_PORT_* bits, read with test_bit() throughout.
23 * client_ops -- how buffered input reaches the ldisc. This is the indirection
24 * tty_buffer.c calls through (port->client_ops->receive_buf),
25 * added upstream so that a tty port can be fed without a tty
26 * (serdev, and the console). tty_port.h:49 names the default.
27 *
28 * The tty_port_tty guard at the bottom of this file: mainline writes it with
29 * __DEFINE_UNLOCK_GUARD and reads the guarded pointer back through
30 * scoped_tty(). It was deferred on the grounds that inventing acquire/release
31 * semantics without a call site would mean guessing rather than reading them
32 * off mainline -- and then 8250_port.c arrived in stage 3 with the call site
33 * at serial8250_update_uartclk(), line 2608. So it is ported from that one
34 * use, which is how it should have been written the first time.
35 */

268/*
269 * Guard class `tty_port_tty` and scoped_tty().
270 *
271 * mainline spells the definitions
272 *
273 * __DEFINE_UNLOCK_GUARD(tty_port_tty, struct tty_struct, tty_kref_put(_T->lock));
274 * static inline class_tty_port_tty_t class_tty_port_tty_constructor(struct tty_port *tport)
275 * { return (class_tty_port_tty_t){ .lock = tty_port_tty_get(tport) }; }
276 * #define scoped_tty() ((struct tty_struct *)(__guard_ptr(tty_port_tty)(&scope)))
277 *
278 * DCL spells them on its own machinery because __DEFINE_UNLOCK_GUARD is not
279 * ported (see <linux/cleanup.h> for why the DEFINE_* family is only partly
280 * here), and because the acquire argument is a *tty_port* while the guarded
281 * object is a *tty* -- mainline's constructor function exists precisely to
282 * cross that type boundary, so it cannot be expressed as DEFINE_LOCK_GUARD_1,
283 * whose constructor takes the guarded type.
284 *
285 * The one real adaptation is scoped_tty(): mainline reads the variable the
286 * guard macro named `scope`, DCL's scoped_guard() names it `_dclg`, so the
287 * macro expands to `_dclg.lock`. Everything else -- who takes a reference
288 * (tty_port_tty_get) and who drops it (tty_kref_put), and that the release is
289 * skipped when the port holds no tty, so a NULL guard must not call kref_put
290 * on NULL -- is read straight off the two mainline lines above.
291 *
292 * Semantics worth stating, since it is easy to get wrong in a driver: the
293 * guard holds one *reference*, not a borrowed pointer, so the tty cannot
294 * disappear while scoped_guard()'s body runs, and leaving the scope drops that
295 * reference. That is exactly what 8250_port.c:2608 wants -- it goes on to
296 * take termios_rwsem on the tty it just looked up.
297 */
298typedef struct {
299 struct tty_struct* lock;
300} dclg_tty_port_tty;
301
302static inline void dclg_tty_port_tty_release(dclg_tty_port_tty* g)
303{
304 struct tty_struct* _T = g->lock;
305 if (_T) {
306 tty_kref_put(_T);
307 }
308}
309
310static inline dclg_tty_port_tty dclg_tty_port_tty_ctor(struct tty_port* tport)
311{
312 dclg_tty_port_tty g;
313 g.lock = tty_port_tty_get(tport);
314 return g;
315}
