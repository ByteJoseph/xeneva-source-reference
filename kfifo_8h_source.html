4/*
5 * DCL <linux/kfifo.h> -- the byte ring behind a tty port's transmit buffer.
6 *
7 * serial_core.c keeps everything waiting to go out the UART in one of these:
8 * tty_port_alloc_xmit_buf() hands it a PAGE_SIZE buffer (serial_core.c:262),
9 * uart_write() pushes bytes in, and the driver's start_tx pulls them out and
10 * writes them to the shift register. 8250_port.c:1722 decides when to raise a
11 * write wakeup by comparing kfifo_len() against WAKEUP_CHARS, so the length
12 * has to be exact or the tty never learns the port drained.
13 *
14 * Deliberate simplification: mainline's kfifo is a macro-generated type-safe
15 * family with a __kfifo core, a mask for power-of-two indexing, and record
16 * variants for framed data. None of that is needed for a byte ring -- and
17 * mainline's own `size` must be a power of two, which a macro cannot check.
18 * This implementation indexes with `% size` instead: one division per byte on
19 * a 115200-baud transmit path is nothing, and it cannot be wrong if a future
20 * caller passes a size that is not a power of two. The `in`/`out` counters
21 * free-run and wrap, so occupancy is `in - out` in plain unsigned arithmetic.
22 *
23 * The subset is exactly what the staged set calls: kfifo_init, kfifo_reset,
24 * kfifo_is_empty, kfifo_len, kfifo_avail, kfifo_put, kfifo_in. kfifo_get and
25 * kfifo_out were absent for two stages on the grounds that nothing read
26 * through this ring; stage 3 changed that, and serial_core.h now reads it
27 * three ways (uart_fifo_out, uart_fifo_get, uart_xmit_advance), so all three
28 * are at the bottom of this file.
29 */

94/*
95 * The three reads serial_core.h asks for and this header did not have.
96 *
97 * mainline spells all three as macros built on __kfifo_out with an element
98 * size; DCL's struct kfifo is byte-oriented (no esize field, `size` is a byte
99 * capacity), so the bodies are plain byte loops below -- same three
100 * behaviours, which is what the callers actually depend on:
101 *
102 * kfifo_out(f, buf, n) copy out and *consume* up to n bytes
103 * kfifo_get(f, val) copy out and consume exactly one byte; the count
104 * it returns is 0 or 1, which is why
105 * uart_fifo_get() can add it straight to
106 * icount.tx as "one more byte left the fifo"
107 * kfifo_skip_count(f, n) consume without copying, for
108 * uart_xmit_advance() -- the transmit path has
109 * already accounted for the bytes it handed to the
110 * UART and only needs the ring to move on
111 *
112 * The clamp to kfifo_len() matters in all three: mainline's variants return
113 * short rather than reading past the consumer mark, and a ring that could be
114 * asked for more than it holds would otherwise wrap and re-deliver bytes.
115 *
116 * out is a free-running counter here (kfifo_len is `in - out`, taken mod
117 * `size` on access), so advancing it needs no mask -- the same convention the
118 * two writers above already use for `in`.
119 */
120static inline unsigned int kfifo_out(struct kfifo* f, void* buf,
121 unsigned int n)
122{
123 unsigned int i;
124
125 if (!f->buf)
126 return 0;
127 if (n > kfifo_len(f))
128 n = kfifo_len(f);
129
130 for (i = 0; i < n; i++)
131 ((unsigned char*)buf)[i] = f->buf[(f->out + i) % f->size];
132 f->out += n;
133 return n;
134}
