4/*
5 * DCL <linux/hrtimer.h> -- the two RS485 turnaround timers 8250_port.c owns.
6 *
7 * This header exists because struct serial8250_em485 embeds two struct
8 * hrtimer BY VALUE (serial_8250.h:126-127), so the type has to be complete at
9 * the point of declaration -- and mainline reaches it there by accident, not
10 * by listing it: upstream's serial_8250.h includes errno/serial_core/
11 * serial_reg/platform_device and nothing that names hrtimer, so struct
12 * hrtimer is complete only because serial_core.h -> sched.h happens to drag it
13 * in. DCL's serial_core.h is assembled from mainline but its include list is
14 * the subset BaseHdr actually has, so the dependency is written down in
15 * serial_8250.h instead of arriving by side effect.
16 *
17 * WHAT THESE DO, AND WHAT THEY DO NOT
18 * -----------------------------------
19 * hrtimer_start() records the target time and sets a pending flag. It does
20 * NOT queue anything, and no callback ever fires from it.
21 *
22 * That is not a placeholder chosen because it was quick: Xeneva runs no timer
23 * interrupt at all, which is already documented at the top of
24 * <linux/timer.h> -- there is no tick to count jiffies, so `extern jiffies`
25 * would link to a word that never moves, and DCL reads the same counter
26 * udelay() burns against instead. With no tick there is no place to run a
27 * timer queue from, and <linux/timer.h>'s add_timer()/mod_timer() have the
28 * same contract today: set pending, do not schedule, with the queue left as a
29 * TODO for whoever brings up real deferred work. These functions are the same
30 * contract, written in ktime rather than jiffies, so the two halves of DCL's
31 * timer story do not disagree.
32 *
33 * The consequence, stated plainly because it is a real limitation and not a
34 * formality: serial8250_em485_handle_stop_tx()/..._handle_start_tx() never
35 * run, so the delay between "shift from TX to RX" and "shift back" is never
36 * applied. That delay exists to let the last stop bit leave the wire before
37 * the driver flips an RS485 transceiver's direction pin.
38 *
39 * Why that costs nothing today: these two timers are armed only by
40 * serial8250_em485_init(), which is called only from
41 * serial8250_em485_config() -- the generic ->rs485_config() callback, i.e.
42 * the handler for TIOCSRS485. No port DCL opens is put into RS485 mode, so
43 * p->em485 is NULL, the em485 code paths are not entered, and nothing is
44 * waiting for a callback that will not come. If a stage ahead does bring up
45 * RS485 (an iMX8MP or RPi transceiver is exactly the hardware this is for),
46 * this header is the thing to replace -- not the drivers, which are already
47 * calling the right functions with the right arguments.
48 *
49 * Struct layout is DCL's own: 8250_port.c never touches a field of struct
50 * hrtimer, it only calls hrtimer_setup/start/cancel/try_to_cancel and returns
51 * HRTIMER_NORESTART from the callbacks, so nothing here needs to agree with
52 * mainline's node/expires layout. See <linux/serial_core.h> for why the same
53 * freedom is NOT taken with struct uart_port -- that one is laid out by
54 * designated initialisers in the driver, field by field.
55 */

94struct hrtimer {

96 * decide absolute vs relative without the caller repeating it. */

106/*
107 * All four are static inline rather than defined in a shim .c: there is no
108 * shared state to put in a shim (no queue, no lock, no clock base), and
109 * inlining keeps a driver TU from gaining an unresolved symbol it did not
110 * need. If a queue ever appears, they move to a .c and these disappear.
111 */
112static inline void hrtimer_setup(
113 struct hrtimer* timer,
114 enum hrtimer_restart (*function)(struct hrtimer* timer),
115 clockid_t clock_id, enum hrtimer_mode mode)
116{
117 (void)clock_id; /* CLOCK_MONOTONIC only; see header comment */
118 timer->mode = mode;
119 timer->function = function;
120 timer->_softexpires = 0;
121 timer->state = HRTIMER_STATE_INACTIVE;
122 timer->pending = 0;
123}

125static inline void hrtimer_start(struct hrtimer* timer, ktime_t tim,
126 enum hrtimer_mode mode)
127{
128 timer->mode = mode;
129 timer->_softexpires = tim;
130 timer->state = HRTIMER_STATE_ENQUEUED;
131 timer->pending = 1;
132}

134/*
135 * -1 means "running, do not delete" on mainline and is what hrtimer_cancel()
136 * retries on. DCL never runs a callback, so nothing can be running and 0/1
137 * (was it armed) is always the whole answer -- the return shape is kept
138 * because hrtimer_cancel()'s callers test it.
139 */
140static inline int hrtimer_try_to_cancel(struct hrtimer* timer)
141{
142 int was_pending = timer->pending;
143 timer->pending = 0;
144 timer->state = HRTIMER_STATE_INACTIVE;
145 return was_pending;
146}
147
148static inline int hrtimer_cancel(struct hrtimer* timer)
149{
150 return hrtimer_try_to_cancel(timer);
151}
