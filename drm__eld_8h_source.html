87static inline const u8 *drm_eld_sad(const u8 *eld)
88{
89 unsigned int ver, mnl;
90
91 ver = (eld[DRM_ELD_VER] & DRM_ELD_VER_MASK) >> DRM_ELD_VER_SHIFT;
92 if (ver != 2 && ver != 31)
93 return NULL;
94
95 mnl = drm_eld_mnl(eld);
96 if (mnl > 16)
97 return NULL;
98
99 return eld + DRM_ELD_CEA_SAD(mnl, 0);
100}
