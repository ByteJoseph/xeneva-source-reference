194#define mipi_dbi_command(dbi, cmd, seq...) \
195({ \
196 const u8 d[] = { seq }; \
197 struct device *dev = &(dbi)->spi->dev; \
198 int ret; \
199 ret = mipi_dbi_command_stackbuf(dbi, cmd, d, ARRAY_SIZE(d)); \
200 if (ret) \
201 dev_err_ratelimited(dev, "error %d when sending command %#02x\n", ret, cmd); \
202 ret; \
203})
