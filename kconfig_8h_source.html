23/*
24 * The use of "&&" / "||" is limited in certain expressions.
25 * The following enable to calculate "and" / "or" with macro expansion only.
26 */
27#define __and(x, y) ___and(x, y)
28#define ___and(x, y) ____and(__ARG_PLACEHOLDER_##x, y)
29#define ____and(arg1_or_junk, y) __take_second_arg(arg1_or_junk y, 0)

31#define __or(x, y) ___or(x, y)
32#define ___or(x, y) ____or(__ARG_PLACEHOLDER_##x, y)
33#define ____or(arg1_or_junk, y) __take_second_arg(arg1_or_junk 1, y)

35/*
36 * Helper macros to use CONFIG_ options in C/CPP expressions. Note that
37 * these only work with boolean and tristate options.
38 */
39
40/*
41 * Getting something that works in C and CPP for an arg that may or may
42 * not be defined is tricky. Here, if we have "#define CONFIG_BOOGER 1"
43 * we match on the placeholder define, insert the "0," for arg1 and generate
44 * the triplet (0, 1, 0). Then the last step cherry picks the 2nd arg (a one).
45 * When CONFIG_BOOGER is not defined, we generate a (... 1, 0) pair, and when
46 * the last step cherry picks the 2nd arg, we get a zero.
47 */
48#define __is_defined(x) ___is_defined(x)
49#define ___is_defined(val) ____is_defined(__ARG_PLACEHOLDER_##val)
50#define ____is_defined(arg1_or_junk) __take_second_arg(arg1_or_junk 1, 0)
51
52/*
53 * IS_BUILTIN(CONFIG_FOO) evaluates to 1 if CONFIG_FOO is set to 'y', 0
54 * otherwise. For boolean options, this is equivalent to
55 * IS_ENABLED(CONFIG_FOO).
56 */
57#define IS_BUILTIN(option) __is_defined(option)
58
59/*
60 * IS_MODULE(CONFIG_FOO) evaluates to 1 if CONFIG_FOO is set to 'm', 0
61 * otherwise. CONFIG_FOO=m results in "#define CONFIG_FOO_MODULE 1" in
62 * autoconf.h.
63 */
64#define IS_MODULE(option) __is_defined(option##_MODULE)
65
66/*
67 * IS_REACHABLE(CONFIG_FOO) evaluates to 1 if the currently compiled
68 * code can call a function defined in code compiled based on CONFIG_FOO.
69 * This is similar to IS_ENABLED(), but returns false when invoked from
70 * built-in code when CONFIG_FOO is set to 'm'.
71 */
72#define IS_REACHABLE(option) __or(IS_BUILTIN(option), \
73 __and(IS_MODULE(option), __is_defined(MODULE)))
74
75/*
76 * IS_ENABLED(CONFIG_FOO) evaluates to 1 if CONFIG_FOO is set to 'y' or 'm',
77 * 0 otherwise. Note that CONFIG_FOO=y results in "#define CONFIG_FOO 1" in
78 * autoconf.h, while CONFIG_FOO=m results in "#define CONFIG_FOO_MODULE 1".
79 */
80#define IS_ENABLED(option) __or(IS_BUILTIN(option), IS_MODULE(option))
81
82#endif /* __LINUX_KCONFIG_H */
