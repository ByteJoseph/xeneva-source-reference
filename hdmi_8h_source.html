62#define HDMI_AVI_INFOFRAME_SIZE 13
63#define HDMI_SPD_INFOFRAME_SIZE 25
64#define HDMI_AUDIO_INFOFRAME_SIZE 10
65#define HDMI_DRM_INFOFRAME_SIZE 26
66#define HDMI_VENDOR_INFOFRAME_SIZE 4
67
68/*
69 * HDMI 1.3a table 5-14 states that the largest InfoFrame_length is 27,
70 * not including the packet header or checksum byte. We include the
71 * checksum byte in HDMI_INFOFRAME_HEADER_SIZE, so this should allow
72 * HDMI_INFOFRAME_SIZE(MAX) to be the largest buffer we could ever need
73 * for any HDMI infoframe.
74 */
75#define HDMI_MAX_INFOFRAME_SIZE 27
76
77#define HDMI_INFOFRAME_SIZE(type) \
78 (HDMI_INFOFRAME_HEADER_SIZE + HDMI_ ## type ## _INFOFRAME_SIZE)
