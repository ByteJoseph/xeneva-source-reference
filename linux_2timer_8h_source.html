13/*
14 * timer_container_of(ptr, timer_member, field) -- recover the struct owning
15 * a timer_list from the timer callback's argument. mainline spells it the
16 * same way, in include/linux/timer.h.
17 *
18 * The three arguments look like a plain container_of() with an extra one,
19 * and the difference matters: container_of()'s first argument is the pointer
20 * *to the member*, whereas here it is the pointer *to the container* -- and
21 * the container is the variable being declared. So `vblank` in
22 *
23 * struct drm_vblank_crtc *vblank = timer_container_of(vblank, t,
24 * disable_timer);
25 *
26 * is read before it exists, but only for its type. That is why the expansion
27 * takes __typeof__(*ptr) rather than deriving the type from the member, and
28 * why the second argument is what gets passed through to container_of():
29 * there the static_assert checks that the pointer points at a `field`, and
30 * `t` does.
31 *
32 * __typeof__ rather than typeof because this target is not in GNU mode by
33 * default.

38/*
39 * Jiffies -- derived from the ARM generic timer, not counted by a tick.
40 *
41 * Nothing in Xeneva increments a tick counter: there is no timer IRQ calling
42 * a scheduler tick, so an `extern volatile unsigned long jiffies` would link
43 * to a variable that never moves, and every `time_after(jiffies, timeout)` a
44 * driver writes would then be false forever -- a polling loop that never
45 * notices its deadline, which reads as a hang rather than a timeout. Reading
46 * the same counter that udelay() burns against keeps comparisons honest with
47 * no tick to service, at HZ (250) resolution.
48 *
49 * It is an object-like macro, so `jiffies++` and `&jiffies` do not compile.
50 * Nothing in the tree does either -- the uses are reads (8250_port.c,
51 * n_tty.c, serial_core.c) -- and a compile error is the right answer if that
52 * changes, rather than a silently wrong increment.
53 */
54unsigned long dcl_jiffies_now(void);
55#define jiffies (dcl_jiffies_now())

62/*
63 * nsecs_to_jiffies(n) -- the conversion uart_poll_timeout() makes
64 * (serial_core.h:1315) to turn a frame time into a polling deadline.
65 *
66 * Truncating division by the nanoseconds-per-tick, which is div_u64()'s
67 * behaviour upstream: mainline rounds only in nsecs_to_jiffies64(), and
68 * matching the 64-bit one here would change a deadline by a tick in a
69 * function whose caller already does `max(..., 1UL)` to keep it from hitting
70 * zero. The round trip is exact at HZ 250: NSEC_PER_SEC / HZ is 4,000,000,
71 * an exact divisor of the second, so nothing accumulates.
72 */
73/*
74 * A macro, and the reason is the include order in <linux/kernel.h>: it pulls
75 * this header in at line 100 and only defines NSEC_PER_SEC at line 281, so a
76 * function body written here would be checked against a NSEC_PER_SEC that
77 * does not exist yet -- including through this file's own include of
78 * kernel.h, which the guard has already emptied. Expanding at the use site
79 * (serial_core.h:1315) instead, where kernel.h has finished, is the whole
80 * difference; `n` appears exactly once below, so there is no double
81 * evaluation for the macro form to introduce.
82 */
83#define nsecs_to_jiffies(n) ((unsigned long)((n) / (NSEC_PER_SEC / HZ)))
84
85/*
86 * nsecs_to_jiffies64(n) -- the 64-bit form drm_syncobj.c waits with (:442, a
87 * submit timeout, and :1220, an absolute deadline) and mainline keeps in
88 * <linux/jiffies.h>. A macro, and for the same include-order reason
89 * nsecs_to_jiffies above is one: <linux/kernel.h> pulls this header in before
90 * it defines NSEC_PER_SEC, so the division has to be spelled at the call
91 * site, where kernel.h has finished. At HZ 250 the divisor is exact
92 * (NSEC_PER_SEC / HZ == 4000000), which is mainline's own common case
93 * (`div_u64(n, NSEC_PER_SEC / HZ)` in kernel/time/time.c) with an ordinary
94 * 64-bit division; `n` is read once.
95 */
96#define nsecs_to_jiffies64(n) ((u64)((n) / (NSEC_PER_SEC / HZ)))
