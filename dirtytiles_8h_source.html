108/* mark every tile the rectangle touches (clipped to the screen) */
109static inline void DirtyTilesMark(DirtyTiles* t, int x, int y, int w, int h) {
110 if (!t->bits)
111 return;
112 int x1 = x + w, y1 = y + h;
113 if (x < 0)
114 x = 0;
115 if (y < 0)
116 y = 0;
117 if (x1 > t->w)
118 x1 = t->w;
119 if (y1 > t->h)
120 y1 = t->h;
121 if (x1 <= x || y1 <= y)
122 return;
123 int c0 = x >> DIRTYTILES_SHIFT, c1 = (x1 - 1) >> DIRTYTILES_SHIFT;
124 int r0 = y >> DIRTYTILES_SHIFT, r1 = (y1 - 1) >> DIRTYTILES_SHIFT;
125 int w0 = c0 >> 6, w1 = c1 >> 6;
126 uint64_t m0 = ~0ULL << (c0 & 63);
127 uint64_t m1 = ~0ULL >> (63 - (c1 & 63));
128 for (int r = r0; r <= r1; r++) {
129 uint64_t* row = t->bits + (size_t)r * t->words;
130 if (w0 == w1) {
131 row[w0] |= m0 & m1;
132 } else {
133 row[w0] |= m0;
134 for (int k = w0 + 1; k < w1; k++)
135 row[k] = ~0ULL;
136 row[w1] |= m1;
137 }
138 }
139 t->any = true;
140}

153static inline int DirtyTilesEmit(DirtyTiles* t, Rect* out, int max) {
154 if (!t->bits || !t->any || max <= 0)
155 return 0;
156 int n = 0, open = 0;
157 const int exact = max - 1; /* out[max - 1] is kept for the overflow box */
158 bool overflow = false;
159 int ox0 = 1 << 30, oy0 = 1 << 30, ox1 = -1, oy1 = -1; /* overflow box, tiles */
160 for (int r = 0; r < t->rows; r++) {
161 uint64_t* row = t->bits + (size_t)r * t->words;
162 int cur = 0, o = 0;
163 /* runs of set tiles, left to right, by bit scanning */
164 int c = 0;
165 while (c < t->cols) {
166 int wi = c >> 6;
167 uint64_t word = row[wi] & (~0ULL << (c & 63));
168 while (!word && ++wi < t->words)
169 word = row[wi];
170 if (!word)
171 break;
172 int s = (wi << 6) + _dirtytiles_ctz(word);
173 if (s >= t->cols)
174 break;
175 /* end of the run: first clear bit at or after s */
176 int ei = s >> 6;
177 uint64_t inv = ~row[ei] & (~0ULL << (s & 63));
178 while (!inv && ++ei < t->words)
179 inv = ~row[ei];
180 int e = inv ? (ei << 6) + _dirtytiles_ctz(inv) : t->words << 6;
181 if (e > t->cols)
182 e = t->cols;
183 /* the same run as an open rectangle of the row above: extend it */
184 while (o < open && t->openX0[o] < s)
185 o++;
186 int rect = -1;
187 if (o < open && t->openX0[o] == s && t->openX1[o] == e) {
188 rect = t->openRect[o];
189 out[rect].h = ((r + 1) << DIRTYTILES_SHIFT) - out[rect].y;
190 } else if (n < exact || (n == exact && max == 1 && !overflow)) {
191 rect = n++;
192 out[rect].x = s << DIRTYTILES_SHIFT;
193 out[rect].y = r << DIRTYTILES_SHIFT;
194 out[rect].w = (e - s) << DIRTYTILES_SHIFT;
195 out[rect].h = DIRTYTILES_SIZE;
196 } else {
197 overflow = true;
198 ox0 = s < ox0 ? s : ox0;
199 ox1 = e > ox1 ? e : ox1;
200 oy0 = r < oy0 ? r : oy0;
201 oy1 = r + 1;
202 }
203 if (rect >= 0) {
204 t->curX0[cur] = s;
205 t->curX1[cur] = e;
206 t->curRect[cur] = rect;
207 cur++;
208 }
209 c = e;
210 }
211 /* this row's runs are the open rectangles for the next row */
212 for (int k = 0; k < cur; k++) {
213 t->openX0[k] = t->curX0[k];
214 t->openX1[k] = t->curX1[k];
215 t->openRect[k] = t->curRect[k];
216 }
217 open = cur;
218 memset(row, 0, (size_t)t->words * sizeof(uint64_t));
219 }
220 t->any = false;
221 if (overflow) {
222 if (n == max) { /* max == 1: fold the one rectangle into the box */
223 int x0 = out[0].x >> DIRTYTILES_SHIFT, y0 = out[0].y >> DIRTYTILES_SHIFT;
224 int x1 = x0 + (out[0].w >> DIRTYTILES_SHIFT), y1 = y0 + (out[0].h >> DIRTYTILES_SHIFT);
225 ox0 = x0 < ox0 ? x0 : ox0;
226 oy0 = y0 < oy0 ? y0 : oy0;
227 ox1 = x1 > ox1 ? x1 : ox1;
228 oy1 = y1 > oy1 ? y1 : oy1;
229 n = 0;
230 }
231 out[n].x = ox0 << DIRTYTILES_SHIFT;
232 out[n].y = oy0 << DIRTYTILES_SHIFT;
233 out[n].w = (ox1 - ox0) << DIRTYTILES_SHIFT;
234 out[n].h = (oy1 - oy0) << DIRTYTILES_SHIFT;
235 n++;
236 }
237 for (int i = 0; i < n; i++) { /* the last tile row / column is partial */
238 if (out[i].x + out[i].w > t->w)
239 out[i].w = t->w - out[i].x;
240 if (out[i].y + out[i].h > t->h)
241 out[i].h = t->h - out[i].y;
242 }
243 return n;
244}
