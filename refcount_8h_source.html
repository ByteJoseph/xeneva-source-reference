2/*
3 * DCL <linux/refcount.h> -- mainline's saturating refcount, over DCL's atomics.
4 *
5 * Why this exists at all: BaseHdr/drm/ is 101 byte-ports of mainline headers,
6 * and struct kref in mainline holds a refcount_t. drm_connector.c:1070 does
7 * not merely call a kref helper -- it reaches inside the struct and writes
8 *
9 * if (!refcount_dec_and_test(&conn->base.refcount.refcount))
10 * return;
11 *
12 * so the field has to be a refcount_t and this header has to provide the
13 * operations. DCL's own kref.h had the field as atomic_t, which is the older
14 * mainline layout; kref.h is rewritten below to match mainline's, and
15 * atomic_t stays as it is because tty/serial code uses it directly as a
16 * counter, not as a reference count.
17 *
18 * Saturation is the whole reason refcount_t is not just atomic_t, and it is
19 * kept here rather than simplified away. A wrapped reference count turns a
20 * live object into a freed one (or the reverse), and it does so silently,
21 * long after the mistake that caused it. Saturating trades that for a
22 * permanently stuck counter, which is at least loud in the debugger and
23 * cannot be mistaken for a working object.
24 *
25 * Two differences from mainline's implementation, neither semantic:
26 *
27 * - Ordering. Mainline picks relaxed/release/acquire per operation to save
28 * instructions. Every DCL atomic is __ATOMIC_SEQ_CST, which is stronger
29 * than all three, so the guarantees mainline documents hold here; the
30 * per-operation ordering simply is not spelled out because there is one
31 * ordering and it is the strong one.
32 *
33 * - The fetch-and-return forms. Mainline uses atomic_fetch_add_relaxed()
34 * and atomic_fetch_sub_release(), which return the value *before* the
35 * operation. DCL's atomic_add_return()/atomic_sub_return() return the
36 * value after, so the before-value is recovered by adding/subtracting i
37 * back. Same number, different arithmetic.
38 *
39 * refcount_dec_not_one() is not ported. Mainline declares it extern with the
40 * body in lib/refcount.c, and its only caller in this tree was
41 * drm_gem_shmem_helper.c, which is not vendored (see Vendored/README.md for
42 * why). It can be added with that caller.
43 */

111/*
112 * __refcount_add_not_zero() -- add @i unless the count is already zero.
113 *
114 * The zero test is inside the loop rather than before it because the count
115 * can reach zero between the read and the update; mainline expresses this as
116 * an atomic_try_cmpxchg() that refreshes its @old on failure, and the loop
117 * below is the same retry with DCL's atomic_cmpxchg(), which reports the
118 * observed value instead of updating the expected one in place.
119 *
120 * Returning the pre-add value is what makes the zero case distinguishable
121 * without a second read: old == 0 means "was already dead", and the update
122 * was not applied.
123 */
124static inline __must_check bool __refcount_add_not_zero(int i, refcount_t *r,
125 int *oldp)
126{
127 int old = atomic_read(&r->refs);
128
129 do {
130 if (!old)
131 break;
132 } while (atomic_cmpxchg(&r->refs, old, old + i) != old);
133
134 if (oldp)
135 *oldp = old;
136
137 if (old < 0 || old + i < 0)
138 refcount_warn_saturate(r, REFCOUNT_ADD_NOT_ZERO_OVF);
139
140 return old;
141}

179static inline void __refcount_add(int i, refcount_t *r, int *oldp)
180{
181 int old = atomic_add_return(i, &r->refs) - i;
182
183 if (oldp)
184 *oldp = old;
185
186 if (!old)
187 refcount_warn_saturate(r, REFCOUNT_ADD_UAF);
188 else if (old < 0 || old + i < 0)
189 refcount_warn_saturate(r, REFCOUNT_ADD_OVF);
190}

224static inline __must_check bool __refcount_sub_and_test(int i, refcount_t *r,
225 int *oldp)
226{
227 int old = atomic_sub_return(i, &r->refs) + i;
228
229 if (oldp)
230 *oldp = old;
231
232 if (old > 0 && old == i)
233 return true;
234
235 if (old <= 0 || old - i < 0)
236 refcount_warn_saturate(r, REFCOUNT_SUB_UAF);
237
238 return false;
239}
