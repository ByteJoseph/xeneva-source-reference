4/*
5 * DCL <linux/scatterlist.h> -- struct scatterlist and sg_init_one().
6 *
7 * ### Deliberately not mainline's layout
8 *
9 * mainline's struct scatterlist is
10 *
11 * unsigned long page_link; // page pointer + chain/terminate bits
12 * unsigned int offset;
13 * unsigned int length;
14 * dma_addr_t dma_address;
15 *
16 * -- a *page-based* list, whose whole job is to walk a compound page in
17 * physical segments and to carry a DMA address that the IOMMU/streaming
18 * mapper filled in. Xeneva has neither: a buffer handed to a virtio queue is
19 * a single linear-map virtual address (AuVirtioPCISetupQueue and buf_phys()
20 * in virtio_shim.c turn it into a ring address directly), and there is no
21 * DMA API to populate dma_address through. So the struct here carries exactly
22 * what the one consumer reads:
23 *
24 * struct scatterlist { void* buf; unsigned int len; };
25 *
26 * That is the shape DCL/linux_kmod_shim.c's sg_init_one() fills in, and the
27 * shape virtio_shim.c's virtqueue_add_inbuf/sgs walk -- both of which are
28 * written against this definition, not against upstream's. Keeping upstream's
29 * fields here would leave four members that every write has to remember to
30 * leave zero, for a mapping layer that does not exist.
31 *
32 * The consequence worth writing down: a driver that reaches into the sg for a
33 * page (`sg_virt()` on a compound page, DMA mapping helpers) will not build.
34 * Add those when something needs them, with the linear-map address doing the
35 * job of `page_link`.
36 *
37 * This struct used to live in <linux/virtio.h>, which included it in every
38 * file that talked to a virtqueue. It is here because that is mainline's home
39 * for it, so a ported source includes the header mainline tells it to; virtio.h
40 * now includes this file instead of redeclaring the type. Exactly one
41 * definition either way -- two would be two different `struct scatterlist`s
42 * with no way to pass one to the other.
43 */

47/*
48 * One contiguous region, as the shim sees it. buf is a *virtual* address in
49 * the linear map; the shim converts to a bus address at kick time.
50 *
51 * `page` is the one member that is not read by the queue machinery -- it is
52 * the bookkeeping a *releasing* driver walks. virtio_console.c:350-357 does
53 *
54 * for (i = 0; i < buf->sgpages; i++) {
55 * struct page *page = sg_page(&buf->sg[i]);
56 * if (!page)
57 * break;
58 * put_page(page);
59 * }
60 *
61 * and sg_set_page() at :872/:891 is what would have put the pointer there.
62 * Without a slot for it sg_page() has nothing to return but NULL, the `break`
63 * fires on the first entry, and the pages behind it are never released: a
64 * leak that no test in this tree would report, because the only path that
65 * fills sgpages > 0 is the splice one (see <linux/splice.h> on why that is
66 * not reachable here). Carrying the pointer costs 8 bytes an entry and makes
67 * the release loop do what it reads as doing.
68 *
69 * buf stays the first member: every existing walk (virtio_shim.c,
70 * linux_kmod_shim.c's sg_init_one) reads ->buf and ->len by name, and putting
71 * a pointer in front of them would move those offsets for no gain.
72 */
73struct scatterlist {
74 void* buf;
75 /*
76 * One length, two spellings. DCL's writers (sg_init_one,
77 * sg_set_page, the shmem tables) name it `len`; mainline's readers
78 * name it `length` -- virtgpu_object.c:195 copies sg->length into a
79 * virtio_gpu_mem_entry -- and vendored sources are not edited to a
80 * local spelling. A union keeps the two names on one storage, so a
81 * write through either is visible through the other.
82 */
83 union {
84 unsigned int len;
85 unsigned int length;
86 };
87 struct page* page;
88 /*
89 * mainline's two DMA fields, carried in full even though nothing
90 * fills them: there is no DMA mapping layer in this kernel, so
91 * dma_address is never written and always reads 0. drm_prime.c:889
92 * still walks it (`sg_dma_address(sgt->sgl)`) and drm_prime.c:893
93 * walks sg_dma_len(), so the *reads* have to be well-formed even
94 * where the writes never happen -- which is the same situation as
95 * iomem_resource in <linux/ioport.h>: an empty but correctly shaped
96 * answer, rather than a missing member.
97 *
98 * mainline makes these conditional on CONFIG_NEED_SG_DMA_LENGTH; here
99 * they are unconditional, so sg_dma_len() below has one arm instead
100 * of two.
101 */
102 dma_addr_t dma_address;
103 unsigned int dma_length;
104};

106/*
107 * struct sg_table -- the *list* of scatterlists, and the reason this header
108 * had to grow it is a C scoping rule rather than a missing field.
109 *
110 * drm_cache.h:41 declares `void drm_clflush_sg(struct sg_table *st);` and
111 * drm_cache.c:127 defines the same function. Both spell the parameter
112 * identically, and clang still reports "conflicting types for
113 * 'drm_clflush_sg'" -- because nothing named `struct sg_table` at file scope
114 * before those lines. C scopes a struct tag that is *first* mentioned inside a
115 * function prototype's parameter list to that declaration alone, so the two
116 * lines name two unrelated incomplete types and the signatures genuinely do
117 * differ. clang warns about the rule first ("declaration of 'struct sg_table'
118 * will not be visible outside of this function", -Wvisibility) and errors
119 * after, and reading only the error sends you hunting for a difference
120 * between two spellings that are byte-identical.
121 *
122 * mainline does not hit this because mainline's <linux/scatterlist.h>
123 * *defines* struct sg_table, so the tag is at file scope before any drm
124 * header is parsed. A forward declaration already exists here in drm_drv.h:67,
125 * but drm_cache.h does not include it -- and moving the decl there would fix
126 * only this consumer, leaving the next header that names the type to
127 * rediscover the same rule.
128 *
129 * All three of mainline's fields, whole, though nothing in this tree fills one
130 * yet: `sgl` is the array, `nents` the entries that were mapped, and
131 * `orig_nents` the entries that were handed in. A two-field version would be
132 * worse than carrying all three -- see the note on struct vm_area_struct in
133 * <linux/mm.h> for why.
134 */
135struct sg_table {
136 struct scatterlist* sgl;
137 unsigned int nents;
138 unsigned int orig_nents;
139};

197 * `struct sg_page_iter page_iter`, so the type has to be complete, not merely

200 * mainline's five fields are all here. The sixth, __sg_page_nents, is not

202 * mainline's scatterlist carries chain and terminate *bits* in page_link, so

312 * not. First, sg_page_iter_dma_address() is named for the *page* iter but is
313 * called with a `struct sg_dma_page_iter *`; it takes that type, which is
