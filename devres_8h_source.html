1/*
2 * DCL <linux/devres.h> -- managed-device resources without the lifetime.
3 *
4 * mainline ties every devm_* allocation and action to a struct device's
5 * probe-to-remove span and runs them all on device release. DCL devices
6 * have no release: put_device() is a documented no-op (DCL/linux_cdev_shim.c)
7 * and no driver in the tree ever unbinds, so the "managed" half reduces to
8 * what the call sites actually test. vkms_drv.c asks devres_open_group()
9 * for a non-NULL token before it builds its device and
10 * devres_release_group() for a "closed" answer on the way out; drm_drv.c:816
11 * returns devm_add_action_or_reset()'s status as the last step of
12 * devm_drm_dev_alloc(). Allocations hand back a real allocation -- the
13 * caller keeps the pointer on mainline too, and here it is never revoked.
14 *
15 * This header's include point is <linux/device.h>: mainline spreads these
16 * declarations over devres.h and reaches them from device users directly,
17 * while DCL's drm chain reaches device.h through drm_print.h, which makes
18 * that the one include that covers drm_drv.c, drm_bridge.c and the vkms
19 * side (via device/faux.h) without editing a vendored include list.
20 */

76/*
77 * devres_alloc() / devres_add() / devres_free() -- the three that back
78 * devres_open_group()'s siblings above, asked for by drm_drv.c:823-834.
79 *
80 * The sequence there is the one mainline documents for a private devres
81 * entry: allocate a block that carries a release function, either free it
82 * again on the failure path or hand it to the device on the success one.
83 * `devm_kfree_strarray` is the release; it is never run here, for the reason
84 * the header's opening comment gives -- no device in this tree ever releases,
85 * so the release half has nowhere to be called from.
86 *
87 * The block is a real allocation and devres_free() really frees it, because
88 * unlike the group tokens above this one is a caller's pointer: drm_drv.c
89 * stores it and drm_drv.c drops it, and an "allocation" that leaked or that
90 * outlived its free would be a defect rather than a harmless stub. devres_add()
91 * is the only empty one of the three, since it exists in mainline to link the
92 * block to a device's list and there is no list.
93 *
94 * dr_release_t is mainline's typedef, kept so a later real devres could be
95 * dropped in without the caller's declaration changing.
96 */
97typedef void (*dr_release_t)(struct device* dev, void* res);
98typedef int (*dr_match_t)(struct device* dev, void* res, void* match_data);

133/*
134 * The four "managed" calls bochs.c makes around its BARs. "Managed" for the
135 * region pair means the claim would be released when the device goes away;
136 * no device in this tree ever releases (see the header comment), so the plain
137 * claim is the whole behavior -- and requesting it again here instead of
138 * calling request_mem_region() directly is what keeps the driver's
139 * error handling meaningful: a NULL still means "someone else owns this
140 * window", which is the one answer bochs_pci_probe tests before proceeding.
141 *
142 * devm_ioremap()/devm_ioremap_wc() cannot be inlines here: their bodies need
143 * AuMapMMIO() and the page size, which live under <Mm/vmmngr.h> -- a kernel
144 * header the BaseHdr tree does not include. Bodies in DCL/pci_shim.c. They
145 * are exactly what the native drivers do with a BAR (virtgpu.cpp maps its
146 * registers the same way): the linear physical->virtual map is for RAM, PCI
147 * BARs are device memory, and a PCI window is not promised to land inside
148 * the direct map's range -- so the mapping is made explicitly, for the
149 * window, with device attributes.
